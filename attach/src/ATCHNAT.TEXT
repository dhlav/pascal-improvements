; SYSATCH natives. EXTERNAL in SYSATCH.TEXT. Zero page
; is .EQU so those addresses are not relocated; a label
; inside a procedure is procedure-relative.
;
; The linker copies these in source order. That order
; is the order of their addresses in the shipped
; 1.3 segment: DOCLI, IRQOLD, IRQNEW, STUBF4, OLDII,
; LCPATCH, RELOCAT, CALLTR.
;
; RELOCAT walks a self-relative list and adds a base.
; LCPATCH copies three bytes, flipping the language
; card when $BF22 does not already show it. DOCLI is
; CLI. IRQOLD writes the IRQ vector through the IIe
; aux/main switches. IRQNEW points IRQ at $FE02.
; STUBF4 copies a 23-byte stub to $03E7 and points
; $03FE at it. OLDII is true for a II/II+ or an
; original IIe ($FBB3<>6 or $FBC0=$EA). CALLTR
; plants its argument in its own JSR and calls it.

P00     .EQU 000
P01     .EQU 001
P04     .EQU 004
P05     .EQU 005
P06     .EQU 006
P07     .EQU 007
P08     .EQU 008
P0A     .EQU 00A
P0B     .EQU 00B
P0C     .EQU 00C
P0D     .EQU 00D

        .PROC DOCLI,0
        .TITLE "DOCLI Procedure"
        CLI
        RTS

        .PROC IRQOLD,0
        .TITLE "IRQOLD Procedure"
        LDA 0C089
        LDA 0C089
        LDA 0FFFE
        STA 0FFFE
        LDA 0FFFF
        STA 0FFFF
        STA 0C009
        LDA 0FFFE
        STA 0FFFE
        LDA 0FFFF
        STA 0FFFF
        STA 0C008
        LDA 0C08B
        LDA 0C08B
        RTS
        .BYTE 000

        .PROC IRQNEW,0
        .TITLE "IRQNEW Procedure"
        LDA 0FFFE
        STA 0FE00
        LDA 0FFFF
        STA 0FE01
        LDA #002
        STA 0FFFE
        LDA #0FE
        STA 0FFFF
        RTS
        .BYTE 000

        .PROC STUBF4,0
        .TITLE "STUBF4 Procedure"
        LDX #000
COPY    LDA PAYLD,X
        STA 003E7,X
        INX
        CPX #017
        BNE COPY
        LDA #0E7
        STA 003FE
        LDA #003
        STA 003FF
        LDA 0FFFE
        STA 003F5
        LDA 0FFFF
        STA 003F6
        RTS
PAYLD   LDA 0C08B
        LDA 0C08B
        LDA #003
        PHA
        LDA #0F7
        PHA
        PHP
        .BYTE 04C,000,000
        LDA 0C089
        LDA 0C089
        RTI
        .BYTE 000

        .FUNC OLDII,0
        .TITLE "OLDII Function"
        PLA
        STA P04
        PLA
        STA P05
        PLA
        PLA
        PLA
        PLA
        LDA #000
        PHA
        LDA 0C08A
        LDA 0FBB3
        LDY 0FBC0
        LDX 0C08B
        LDX 0C08B
        CMP #006
        BNE YES
        TYA
        CMP #0EA
        BEQ YES
        LDA #000
        BEQ STORE
YES     LDA #001
STORE   PHA
        LDA P05
        PHA
        LDA P04
        PHA
        RTS
        .BYTE 000

        .PROC LCPATCH,4
        .TITLE "LCPATCH Procedure"
        PLA
        STA P04
        PLA
        STA P05
        PLA
        STA P08
        PLA
        PLA
        STA P07
        PLA
        PLA
        STA P06
        PLA
        PLA
        STA P00
        PLA
        STA P01
        LDA 0BF22
        AND #021
        EOR #021
        BEQ COPY3
        LDA 0C083
COPY3   LDY #002
; LDA P06,Y must be absolute. Zero-page ,Y is a
; different opcode, and a label here would relocate.
ABS6    .BYTE 0B9,006,000
        STA @P00,Y
        DEY
        BPL ABS6
        LDA 0BF22
        AND #021
        EOR #021
        BEQ BACK
        LDA 0C08B
BACK    LDA P05
        PHA
        LDA P04
        PHA
        RTS
        .BYTE 000

        .FUNC RELOCAT,2
        .TITLE "RELOCAT Function"
        PLA
        STA P04
        PLA
        STA P05
        PLA
        PLA
        PLA
        PLA
        PLA
        STA P0C
        PLA
        STA P0D
        PLA
        STA P06
        PLA
        STA P07
        LDY #000
        LDA @P06,Y
        TAX
        INX
        INY
        LDA @P06,Y
        STA P08
        JMP STEP
FIX     LDY #000
        SEC
        LDA P06
        SBC @P06,Y
        STA P0A
        INY
        LDA P07
        SBC @P06,Y
        STA P0B
        CLC
        DEY
        LDA @P0A,Y
        ADC P0C
        STA @P0A,Y
        INY
        LDA @P0A,Y
        ADC P0D
        STA @P0A,Y
STEP    LDA P06
        SEC
        SBC #002
        STA P06
        BCS NODEC
        DEC P07
NODEC   DEX
        BNE FIX
        DEC P08
        BPL FIX
        CLC
        LDA P06
        ADC #002
        TAX
        LDA P07
        ADC #000
        PHA
        TXA
        PHA
        LDA P05
        PHA
        LDA P04
        PHA
        RTS
        .BYTE 000

        .PROC CALLTR,2
        .TITLE "CALLTR Procedure"
        PLA
        STA RETLO
        PLA
        STA RETHI
        PLA
        STA PLANT+1
        PLA
        STA PLANT+2
PLANT   JSR PLANT
        LDA RETHI
        PHA
        LDA RETLO
        PHA
        RTS
RETLO   .BYTE 000
RETHI   .BYTE 000

        .END
