PROGRAM ATTACHUD;

{ Apple Pascal ATTACH.DATA Creation Utility [1.3]. Builds the 16-word
  records SYSTEM.ATTACH reads at boot. One record per driver; if any
  driver uses interrupts, a final IM record is appended. }

TYPE
  DIGITS = 1..5;
  BITSET = SET OF 0..47;
  NAME8 = PACKED ARRAY [0..7] OF CHAR;
  DEVREC = PACKED RECORD
             NM: NAME8;
             INITU: BITSET;
             UNITU: BITSET;
             BOUND: 0..256;
             PAGE4: BOOLEAN;
             PAGE2: BOOLEAN;
             IMFLAG: BOOLEAN;
             TNM: NAME8;
             TBOUND: 0..256
           END;

VAR
  FNAME: STRING[80];
  BIT: BITSET;
  NAMES: ARRAY [0..16] OF NAME8;
  NDRV: INTEGER;
  UNUM: INTEGER;
  OUTF: FILE OF DEVREC;
  NEEDIM: BOOLEAN;
  LASTINIT: BOOLEAN;
  AGAIN: BOOLEAN;
  CAN4000: BOOLEAN;
  CAN2000: BOOLEAN;
  USED128: PACKED ARRAY [128..143] OF BOOLEAN;

PROCEDURE COPYNAME(VAR S: STRING; VAR D: NAME8);
VAR
  I: INTEGER;

  FUNCTION MIN(A, B: INTEGER): INTEGER;
  BEGIN
    IF A > B THEN MIN := B ELSE MIN := A
  END;

BEGIN
  D := '        ';
  FOR I := 1 TO MIN(LENGTH(S), 8) DO
    D[I - 1] := S[I]
END;

PROCEDURE UPCASE(VAR S: STRING);
VAR
  DELTA: INTEGER;
  I: INTEGER;
BEGIN
  DELTA := ORD('a') - ORD('A');
  FOR I := 1 TO LENGTH(S) DO
    IF S[I] IN ['a'..'z'] THEN
      S[I] := CHR(ORD(S[I]) - DELTA)
END;

PROCEDURE READIT(VAR S: STRING);
BEGIN
  WRITE('(<RETURN> to exit program): ');
  READLN(S);
  IF LENGTH(S) = 0 THEN
    EXIT(ATTACHUD)
  ELSE
    UPCASE(S)
END;

PROCEDURE ERRTRY(MSG: STRING; VAR S: STRING);
BEGIN
  WRITELN;
  WRITE(CHR(7));
  WRITE('ERROR => ');
  WRITE(MSG);
  WRITELN;
  WRITE('Please try again ...');
  WRITELN;
  READIT(S)
END;

FUNCTION YESNO: BOOLEAN;
VAR
  OK: BOOLEAN;
  S: STRING;
BEGIN
  REPEAT
    READLN(S);
    UPCASE(S);
    IF LENGTH(S) > 0 THEN
      OK := S[1] IN ['N','Y']
    ELSE
      OK := FALSE;
    IF NOT OK THEN
      BEGIN
        WRITE(CHR(7));
        WRITE('(Y/N)? ')
      END
  UNTIL OK;
  YESNO := S[1] = 'Y'
END;

FUNCTION GETNUM(VAR S: STRING; MAXD: DIGITS): INTEGER;
VAR
  OK: BOOLEAN;
  LIM: INTEGER;
  VAL: INTEGER;
  I: INTEGER;
BEGIN
  REPEAT
    VAL := 0;
    WHILE LENGTH(S) > MAXD DO
      ERRTRY('Number too large.', S);
    REPEAT
      I := 1;
      OK := TRUE;
      WHILE (I <= LENGTH(S)) AND OK DO
        BEGIN
          OK := S[I] IN ['0'..'9'];
          I := I + 1
        END;
      IF NOT OK THEN
        ERRTRY('We want a positive number.', S)
    UNTIL OK;
    I := 1;
    WHILE (I <= LENGTH(S)) AND OK DO
      BEGIN
        LIM := 32767 DIV 10;
        IF (VAL > LIM) OR
           ((VAL = LIM) AND (S[I] > '7')) THEN
          BEGIN
            OK := FALSE;
            S := '      '
          END
        ELSE
          VAL := VAL * 10 + (ORD(S[I]) - ORD('0'));
        I := I + 1
      END
  UNTIL OK;
  GETNUM := VAL
END;

PROCEDURE OPENDAT;
VAR
  OK: BOOLEAN;
  I: INTEGER;

  PROCEDURE ASKHIRES;
  BEGIN
    WRITELN;
    WRITE('NOTE: These next questions will determine if');
    WRITELN;
    WRITE('any attached drivers can reside in the HIRES');
    WRITELN;
    WRITE('pages.  It will be assumed that they can for');
    WRITELN;
    WRITE('the page in question if your program will not');
    WRITELN;
    WRITE('ever use the HIRES page and you answer no to');
    WRITELN;
    WRITE('the prompt for that page.');
    WRITELN;
    WRITELN;
    WRITE('Will you use the 2000.3FFF HIRES page? (Y/N) ');
    CAN2000 := NOT YESNO;
    WRITELN;
    WRITE('Will you use the 4000.5FFF HIRES page? (Y/N) ');
    CAN4000 := NOT YESNO
  END;

BEGIN
  WRITE('Apple Pascal ATTACH.DATA Creation Utility [1.3]');
  WRITELN;
  WRITE('Copyright Apple Computer, Inc. 1983,1985');
  WRITELN;
  WRITELN;
  WRITE('What will be the name of the attach data file?');
  WRITELN;
  READIT(FNAME);
  REPEAT
    (*$I-*)
    REWRITE(OUTF, FNAME);
    OK := IORESULT = 0;
    (*$I+*)
    IF NOT OK THEN
      BEGIN
        WRITELN;
        WRITE(CHR(7));
        WRITE('ERROR => Open failed upon attach data file.');
        WRITELN;
        WRITE('Please try again ...');
        WRITELN;
        WRITE('(<RETURN> to exit program): ');
        READLN(FNAME);
        IF LENGTH(FNAME) = 0 THEN
          EXIT(ATTACHUD)
        ELSE
          UPCASE(FNAME)
      END
  UNTIL OK;
  ASKHIRES;
  FOR I := 128 TO 143 DO
    USED128[I] := FALSE;
  NEEDIM := FALSE;
  NDRV := 0
END;

PROCEDURE ADDRIVER;
VAR
  OK: BOOLEAN;
  U: INTEGER; { unused; WITH OUTF^ must sit at word 3 }

  FUNCTION VALIDNM(VAR S: STRING): BOOLEAN;
  VAR
    LEN: INTEGER;
    I: INTEGER;
  BEGIN
    VALIDNM := FALSE;
    LEN := LENGTH(S);
    IF (LEN < 1) OR (LEN > 8) THEN
      EXIT(VALIDNM);
    FOR I := 1 TO LEN DO
      IF NOT ((S[I] IN ['A'..'Z']) OR (S[I] IN ['0'..'9'])) THEN
        EXIT(VALIDNM);
    VALIDNM := TRUE
  END;

  FUNCTION UNIQNAME(VAR S: STRING): BOOLEAN;
  VAR
    I: INTEGER;
    NM: NAME8;
  BEGIN
    UNIQNAME := FALSE;
    COPYNAME(S, NM);
    FOR I := 1 TO NDRV DO
      IF NAMES[I] = NM THEN
        EXIT(UNIQNAME);
    IF NDRV < 16 THEN
      NDRV := NDRV + 1
    ELSE
      BEGIN
        WRITELN;
        WRITE(CHR(7));
        WRITE('ERROR => A maximum of 16 drivers');
        WRITELN;
        WRITE('and transients are allowed per file.');
        WRITELN;
        WRITE('(<RETURN> to exit program)');
        READLN;
        EXIT(ATTACHUD)
      END;
    NAMES[NDRV] := NM;
    UNIQNAME := TRUE
  END;

  FUNCTION MIXOK: BOOLEAN;
  BEGIN
    WITH OUTF^ DO
      IF ([1,6,7] * UNITU <> []) AND
         (UNITU - [1,6,7] <> []) THEN
        BEGIN
          WRITELN;
          WRITE(CHR(7));
          WRITE('ERROR => Character units and block');
          WRITELN;
          WRITE('units cannot be attached to the same');
          WRITELN;
          WRITE('device driver.  Your last entry will');
          WRITELN;
          WRITE('be removed.');
          WRITELN;
          WRITE('(<RETURN> to continue program)');
          READLN;
          UNITU := UNITU - BIT;
          MIXOK := FALSE
        END
      ELSE
        MIXOK := TRUE
  END;

  PROCEDURE ASKINIT;
  BEGIN
    WRITELN;
    WRITE('Do you want this unit to be initialized');
    WRITELN;
    WRITE('at boot time? (Y/N) ');
    LASTINIT := YESNO;
    IF LASTINIT THEN
      WITH OUTF^ DO
        INITU := INITU + BIT
  END;

  FUNCTION GETALIGN: INTEGER;
  BEGIN
    WRITELN;
    WRITE('The boundary can be between 0 and 256.');
    WRITELN;
    WRITE('   0=>Driver can start anywhere.(default)');
    WRITELN;
    WRITE('   8=>Driver starts on 8 byte boundary.');
    WRITELN;
    WRITE('   N=>Driver starts on N byte boundary.');
    WRITELN;
    WRITE(' 256=>Driver starts on PAGE boundary.');
    WRITELN;
    REPEAT
      WRITELN;
      WRITE('What boundary do you want? (0..256)');
      WRITELN;
      READIT(FNAME);
      UNUM := GETNUM(FNAME, 3)
    UNTIL UNUM <= 256;
    GETALIGN := UNUM
  END;

  PROCEDURE ASKIRQ;
  BEGIN
    WRITELN;
    WRITE('Will this driver use interrupts? (Y/N) ');
    WITH OUTF^ DO
      IF YESNO THEN
        IF LASTINIT THEN
          BEGIN
            IMFLAG := TRUE;
            NEEDIM := TRUE
          END
        ELSE
          BEGIN
            WRITELN;
            WRITE(CHR(7));
            WRITE('ERROR => Drivers using interrupts');
            WRITELN;
            WRITE('must be initialized at boot time.');
            WRITELN;
            WRITE('(<RETURN> to exit program)');
            READLN;
            IMFLAG := FALSE;
            EXIT(ATTACHUD)
          END
      ELSE
        IMFLAG := FALSE
  END;

  PROCEDURE ASKTRANS;
  BEGIN
    WRITELN;
    WRITE('Do you want this driver to have a');
    WRITELN;
    WRITE('transient initialization section? (Y/N) ');
    WITH OUTF^ DO
      IF YESNO THEN
        BEGIN
          WRITELN;
          WRITE('What is the transient''s name?');
          WRITELN;
          WRITE('It will be the assembly''s .PROC name.');
          WRITELN;
          READIT(FNAME);
          OK := FALSE;
          REPEAT
            IF NOT VALIDNM(FNAME) THEN
              ERRTRY('Alphanumeric names of 1..8 characters.',
                     FNAME)
            ELSE
              IF NOT UNIQNAME(FNAME) THEN
                ERRTRY('Driver/transient names must be unique.',
                       FNAME)
              ELSE
                OK := TRUE
          UNTIL OK;
          COPYNAME(FNAME, TNM);
          WRITELN;
          WRITE('Do you want this transient aligned on');
          WRITELN;
          WRITE('a particular byte boundary? (Y/N) ');
          IF YESNO THEN
            TBOUND := GETALIGN
          ELSE
            TBOUND := 0
        END
      ELSE
        BEGIN
          TNM := '        ';
          TBOUND := 0
        END
  END;

BEGIN
  WRITELN;
  WRITE('What is this device driver''s name?');
  WRITELN;
  WRITE('It will be the assembly''s .PROC name.');
  WRITELN;
  READIT(FNAME);
  OK := FALSE;
  REPEAT
    IF NOT VALIDNM(FNAME) THEN
      ERRTRY('Alphanumeric names of 1..8 characters.', FNAME)
    ELSE
      IF NOT UNIQNAME(FNAME) THEN
        ERRTRY('Driver/transient names must be unique.', FNAME)
      ELSE
        OK := TRUE
  UNTIL OK;
  WITH OUTF^ DO
    BEGIN
      COPYNAME(FNAME, NM);
      WRITELN;
      WRITE('NOTE: The next questions will be repeated for');
      WRITELN;
      WRITE('each unit number to be attached to this driver.');
      WRITELN;
      WRITE('As many unit numbers as desired may be attached');
      WRITELN;
      WRITE('to the driver.  The units should be either');
      WRITELN;
      WRITE('all character devices or all block devices.');
      WRITELN;
      AGAIN := TRUE;
      PAGE2 := CAN2000;
      PAGE4 := CAN4000;
      UNITU := [];
      INITU := [];
      WHILE AGAIN DO
        BEGIN
          WRITELN;
          WRITE('Assign what unit number to this driver?');
          WRITELN;
          WRITE('Valid units are 1, 2, 4..20, 128..143.');
          WRITELN;
          READIT(FNAME);
          REPEAT
            UNUM := GETNUM(FNAME, 3);
            OK := UNUM IN ([128..143] + [1,2,4..20]);
            IF NOT OK THEN
              ERRTRY('Invalid unit number.', FNAME)
          UNTIL OK;
          IF UNUM IN [1,2,4..20] THEN
            CASE UNUM OF
              1: BIT := [1];
              2: BIT := [1];
              4: BIT := [4];
              5: BIT := [5];
              6: BIT := [6];
              7: BIT := [7];
              8: BIT := [7];
              9: BIT := [9];
              10: BIT := [10];
              11: BIT := [11];
              12: BIT := [12];
              13: BIT := [13];
              14: BIT := [14];
              15: BIT := [15];
              16: BIT := [16];
              17: BIT := [17];
              18: BIT := [18];
              19: BIT := [19];
              20: BIT := [20]
            END
          ELSE
            IF UNUM IN [128..143] THEN
              BEGIN
                USED128[UNUM] := TRUE;
                CASE UNUM OF
                  128: BIT := [21];
                  129: BIT := [22];
                  130: BIT := [23];
                  131: BIT := [24];
                  132: BIT := [25];
                  133: BIT := [26];
                  134: BIT := [27];
                  135: BIT := [28];
                  136: BIT := [29];
                  137: BIT := [30];
                  138: BIT := [31];
                  139: BIT := [32];
                  140: BIT := [33];
                  141: BIT := [34];
                  142: BIT := [35];
                  143: BIT := [36]
                END
              END;
          UNITU := UNITU + BIT;
          IF MIXOK THEN
            ASKINIT;
          WRITELN;
          WRITE('Do you want another unit number to');
          WRITELN;
          WRITE('refer to this device driver? (Y/N) ');
          AGAIN := YESNO
        END;
      WRITELN;
      WRITE('Do you want this driver aligned on');
      WRITELN;
      WRITE('a particular byte boundary? (Y/N) ');
      IF YESNO THEN
        BOUND := GETALIGN
      ELSE
        BOUND := 0;
      ASKTRANS;
      ASKIRQ;
      PUT(OUTF)
    END
END;

PROCEDURE ADDIM;
VAR
  U: INTEGER;
  BIT: BITSET;

  PROCEDURE FINDUNIT;
  VAR
    FOUND: BOOLEAN;
  BEGIN
    U := 144;
    FOUND := FALSE;
    REPEAT
      U := U - 1;
      FOUND := NOT USED128[U]
    UNTIL FOUND OR (U = 127);
    IF FOUND THEN
      CASE U OF
        128: BIT := [21];
        129: BIT := [22];
        130: BIT := [23];
        131: BIT := [24];
        132: BIT := [25];
        133: BIT := [26];
        134: BIT := [27];
        135: BIT := [28];
        136: BIT := [29];
        137: BIT := [30];
        138: BIT := [31];
        139: BIT := [32];
        140: BIT := [33];
        141: BIT := [34];
        142: BIT := [35];
        143: BIT := [36]
      END
    ELSE
      BEGIN
        WRITELN;
        WRITE(CHR(7));
        WRITE('ERROR => No user unit number available');
        WRITELN;
        WRITE('for Interrupt Manager puesdo-driver.');
        WRITELN;
        WRITE('(<RETURN> to exit program)');
        WRITELN;
        READLN;
        EXIT(ATTACHUD)
      END
  END;

BEGIN
  FINDUNIT;
  WITH OUTF^ DO
    BEGIN
      NM := 'IM      ';
      BOUND := 0;
      IMFLAG := TRUE;
      UNITU := BIT;
      INITU := BIT;
      PAGE2 := CAN2000;
      PAGE4 := CAN4000
    END;
  PUT(OUTF)
END;

BEGIN
  OPENDAT;
  REPEAT
    ADDRIVER;
    WRITELN;
    WRITE('Do you want to attach another driver? (Y/N) ')
  UNTIL NOT YESNO;
  IF NEEDIM THEN
    ADDIM;
  CLOSE(OUTF, LOCK);
  WRITELN;
  WRITE('Attach data creation complete.');
  WRITELN
END.
