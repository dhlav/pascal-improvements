{ LIBRARY -- Apple Pascal 1.3 LIBRARY.CODE, the Librarian,
  reconstructed.

  One segment, LIBRARIA, and 16 procedures. The shape is the
  linker's: a (*$U-*) compilation whose first SEGMENT PROCEDURE
  is segment 1, finished by the Librarian itself, which leaves
  slot 0 of the shipped file blank (finding 267).

  The ancestor is UCSD's I.5 Librarian, reference_source/ucsd_15,
  by way of its II.0 revision (Gary Dismukes, March 1979, in
  github.com/dhlav/ucsd-psystem-os, librarian/), which added the
  byte-flipped dictionaries: FIRSTFILE, FLIPPED, BYTESWAP,
  TABLEFLIPPED and FLIPTABLE are its names (finding 292).
  Apple moved NEWLINKER's body into LIBRARIAN, made its two
  dictionaries global records rather than heap blocks, gave
  LINKIT its slots as parameters, and replaced COPYINTERFACE.
  The ancestors' names are used wherever the binary confirms
  the role. What Apple added has no name to borrow and carries
  a placeholder: LB<n> for procedure n, G<n> for global word n,
  L<n> or M<n> for local word n.

  The byte-sex routines are the linker's own (BYTESWAP and
  FLIPTABLE in LINKER.text), and so are the dictionary types
  they need.

  (*$U-*) leaves range and I/O checking off, and the binary has
  no CHK and no CSP 0 anywhere. }

(*$U-*)
(*$G+*)
PROGRAM PLIBRARIAN;

CONST
     MAXSEG = 15;       (*MAX CODE SEGMENT NUMBER*)

TYPE
                                        (*CODE SEGMENT LAYOUTS*)

     SEGRANGE = 0..MAXSEG;
     SEGDESC = RECORD
                 DISKADDR: INTEGER;     (*REL BLK IN CODE...ABS IN SYSCOM^*)
                 CODELENG: INTEGER      (*# BYTES TO READ IN*)
               END (*SEGDESC*) ;
                                        (*SYSTEM COMMUNICATION AREA*)
                                        (*SEE INTERPRETERS...NOTE  *)
                                        (*THAT WE ASSUME BACKWARD  *)
                                        (*FIELD ALLOCATION IS DONE *)
     SYSCOMREC = RECORD
                   IORSLT: INTEGER ;    (*RESULT OF LAST IO CALL*)
                   XEQERR: INTEGER;     (*REASON FOR EXECERROR CALL*)
                   SYSUNIT: INTEGER;    (*PHYSICAL UNIT OF BOOTLOAD*)
                   BUGSTATE: INTEGER;   (*DEBUGGER INFO*)
                   GDIRP: INTEGER;
                   LASTMP,STKBASE,BOMBP: INTEGER;
                   MEMTOP,SEG,JTAB: INTEGER;
                   BOMBIPC: INTEGER;    (*WHERE XEQERR BLOWUP WAS*)
                   HLTLINE: INTEGER;    (*MORE DEBUGGER STUFF*)
                   BRKPTS: ARRAY [0..3] OF INTEGER;
                   RETRIES: INTEGER;    (*DRIVERS PUT RETRY COUNTS*)
                   EXPANSION: ARRAY [0..8] OF INTEGER;
                   HIGHTIME,LOWTIME: INTEGER;
                   MISCINFO: PACKED RECORD
                               NOBREAK,STUPID,SLOWTERM,
                               HASXYCRT,HASLCCRT,HAS8510A,HASCLOCK: BOOLEAN
                             END;
                   CRTTYPE: INTEGER;
                   CRTCTRL: PACKED RECORD
                              RLF,NDFS,ERASEEOL,ERASEEOS,HOME,ESCAPE: CHAR;
                              BACKSPACE: CHAR;
                              FILLCOUNT: 0..255;
                              EXPANSION: PACKED ARRAY [0..3] OF CHAR
                            END;
                   CRTINFO: PACKED RECORD
                              WIDTH,HEIGHT: INTEGER;
                              RIGHT,LEFT,DOWN,UP: CHAR;
                              BADCH,CHARDEL,STOP,BREAK,FLUSH,EOF: CHAR;
                              ALTMODE,LINEDEL: CHAR;
                              EXPANSION: PACKED ARRAY [0..5] OF CHAR
                            END;
                   SEGTABLE: ARRAY [SEGRANGE] OF
                               RECORD
                                 CODEUNIT: INTEGER;
                                 CODEDESC: SEGDESC
                               END
                 END (*SYSCOM*);

VAR
    SYSCOM: ^SYSCOMREC;                 (*MAGIC PARAM...SET UP IN BOOT*)

SEGMENT PROCEDURE LIBRARIAN(III,JJJ: INTEGER);

CONST
  WINDOW = 2;
  ERROR = 23;
  MARKCODE = 15;
  MARKIN = 5;

TYPE
  SEGKINDS = (LINKED, HOSTSEG, SEGPROC, UNITSEG, SEPRTSEG,
              UNLINKED_INTRINS, LINKED_INTRINS, DATASEG);

  MTYPES = (UNDEFINED, PCODEMOST, PCODELEAST, PDP11, M8080, Z80,
            GA440, M6502, M6800, TI9900);
  SIREC = PACKED RECORD
            SEGNUM: 0..255;
            MTYPE: MTYPES;
            SIFILLER: 0..1;
            VERSION: 0..7
          END;

  { I.5's block 0 with Apple's SEGINFO word added after EXTRA:
    DISPLAY reads its top three bits and its low byte, and LINKIT
    copies it with MOV 1. NOTICE is at word 216, so the filler is
    72 words where I.5's was 88. }
  BLOCK0P = ^BLOCK0;
  BLOCK0 = RECORD
             SEGDSC: ARRAY [SEGRANGE] OF SEGDESC;
             SEGNAME: ARRAY [SEGRANGE] OF
                        PACKED ARRAY [0..7] OF CHAR;
             SEGKIND: ARRAY [SEGRANGE] OF SEGKINDS;
             EXTRA: ARRAY [SEGRANGE] OF INTEGER;
             SEGINFO: ARRAY [SEGRANGE] OF SIREC;
             FILLER: ARRAY [1..72] OF INTEGER;
             NOTICE: STRING[79]
           END;

  SWAPREC = PACKED RECORD
              CASE BOOLEAN OF
                TRUE: (INTEG: INTEGER);
                FALSE: (LOWBYTE: 0..255;
                        HIGHBYTE: 0..255)
            END;

  WORDREC = RECORD
              CASE INTEGER OF
                0: (INTEG: INTEGER);
                1: (KIND: SEGKINDS);
                2: (MISC: SIREC);
                3: (BYTES: PACKED ARRAY [0..1] OF 0..255)
            END;

VAR NBLOCKS,OUTBLOCK,RSLT: INTEGER;
    DSEG,SSEG: SEGRANGE;
    PL,TITLE,NTITLE: STRING;
    NOTICEIN: STRING[79];           { the notice as typed }
    BUF: BLOCK0P;
    CODE,INFILE: FILE;
    CODETABLE,INTBL: BLOCK0;
    FIRSTFILE,FLIPPED: BOOLEAN; { no slot copied yet;
                                  dictionaries are byte-flipped }
    CCH: CHAR;

{ Swap the two bytes of a word. }

PROCEDURE BYTESWAP(VAR WORD: INTEGER);
  VAR TEMP1, TEMP2: SWAPREC;
BEGIN
  TEMP1.INTEG := WORD;
  TEMP2.LOWBYTE := TEMP1.HIGHBYTE;
  TEMP2.HIGHBYTE := TEMP1.LOWBYTE;
  WORD := TEMP2.INTEG
END;

{ TRUE if any segment address has a high byte, which a
  dictionary in this machine's byte order cannot. }

FUNCTION TABLEFLIPPED(VAR TABLE: BLOCK0): BOOLEAN;
  LABEL 1;
  VAR S: INTEGER;
      HIGHBYTE: 0..1;    { index of a word's high byte }
      INT: WORDREC;
BEGIN
  TABLEFLIPPED := FALSE;
  INT.INTEG := 1;
  HIGHBYTE := ORD(INT.BYTES[0] = 1);
  FOR S := 0 TO MAXSEG DO
    WITH TABLE.SEGDSC[S] DO
      BEGIN
        INT.INTEG := DISKADDR;
        IF INT.BYTES[HIGHBYTE] <> 0 THEN
          BEGIN
            TABLEFLIPPED := TRUE;
            GOTO 1
          END
      END;
1:
END;

{ Byte-swap a whole dictionary. }

PROCEDURE FLIPTABLE(VAR TABLE: BLOCK0);
  VAR S: INTEGER;
      WORD: WORDREC;
BEGIN
  FOR S := 0 TO MAXSEG DO
    WITH TABLE DO
      BEGIN
        WITH SEGDSC[S] DO
          BEGIN
            BYTESWAP(DISKADDR);
            BYTESWAP(CODELENG)
          END;
        WORD.KIND := SEGKIND[S];
        BYTESWAP(WORD.INTEG);
        SEGKIND[S] := WORD.KIND;
        BYTESWAP(EXTRA[S]);
        WORD.MISC := SEGINFO[S];
        BYTESWAP(WORD.INTEG);
        SEGINFO[S] := WORD.MISC
      END
END;

PROCEDURE PROMPT(AT: INTEGER);
  VAR L2: CHAR;
BEGIN
  GOTOXY(0,AT);
  IF AT = ERROR THEN WRITE(CHR(7));
  WRITE(PL);
  WITH SYSCOM^.CRTCTRL DO WRITE(ESCAPE,ERASEEOL);
  IF AT = ERROR THEN
    BEGIN
      WRITE(' ':5,'Type <space> to continue');
      REPEAT
        READ(KEYBOARD,L2)
      UNTIL L2 = ' ';
      GOTOXY(0,AT);
      WITH SYSCOM^.CRTCTRL DO WRITE(ESCAPE,ERASEEOL)
    END
END;

{ PROMPT with a number after the message, always on the error
  line's terms: bell, then wait for a space. }

PROCEDURE BELLWAIT(AT, N: INTEGER);
  VAR L3: CHAR;
BEGIN
  GOTOXY(0,AT);
  WRITE(CHR(7));
  WRITE(PL);
  WRITE(N);
  WITH SYSCOM^.CRTCTRL DO WRITE(ESCAPE,ERASEEOL);
  WRITE(' ':5,'Type <space> to continue');
  REPEAT
    READ(KEYBOARD,L3)
  UNTIL L3 = ' ';
  GOTOXY(0,AT);
  WITH SYSCOM^.CRTCTRL DO WRITE(ESCAPE,ERASEEOL)
END;

FUNCTION CHECKIO:BOOLEAN;
VAR RSLT:INTEGER;
BEGIN
  CHECKIO:=IORESULT=0;
  IF IORESULT <> 0 THEN
    BEGIN
      RSLT:=IORESULT;
      PL := 'I/O error # ';
      BELLWAIT(ERROR,RSLT)
    END;
END; (* CHECKIO *)

PROCEDURE OPENFILE;
BEGIN
  REPEAT
    PL := 'Input File -> ';
    PROMPT(MARKIN-1);
    READLN(INPUT,NTITLE);
    IF NTITLE = '*' THEN
      NTITLE := '*SYSTEM.LIBRARY';
    IF LENGTH(NTITLE) > 0 THEN
      BEGIN
        RESET(INFILE,NTITLE);
        IF IORESULT <> 0 THEN
          RESET(INFILE,CONCAT(NTITLE,'.CODE'))
      END;
  UNTIL (CHECKIO) OR (LENGTH(NTITLE) = 0);
END (*OPENFILE*) ;

PROCEDURE DISPLAY(AT: INTEGER; VAR WHAT: BLOCK0);
VAR
  T: INTEGER;
  L4: INTEGER;
BEGIN
  GOTOXY(0,AT);
  WITH WHAT DO
    FOR T := 0 TO MAXSEG DO
      BEGIN
        L4 := (T+1) DIV 2;
        IF ODD(T) THEN
          L4 := L4+7;
        WRITE(' ':11,L4:2,'-');
        IF (SEGINFO[L4].VERSION >= 1) AND
           (SEGNAME[L4] <> '        ') THEN
          WRITE('(',SEGINFO[L4].SEGNUM:2,') ')
        ELSE
          WRITE(' ':5);
        WRITE(SEGNAME[L4],SEGDSC[L4].CODELENG:6);
        IF ODD(T) THEN
          WRITELN
      END;
  PL := 'File length - ';
  PROMPT(MARKCODE-1);
  WRITE(OUTPUT,OUTBLOCK);
END;


PROCEDURE LINKCODE;
  LABEL 1;
  VAR NBLOCKS: INTEGER;
      M2: INTEGER;
      OUTFULL: BOOLEAN;      { set by LINKIT when the output is full }

  PROCEDURE LINKIT(SSEG, DSEG: INTEGER);
    VAR L3: CHAR;
        L4: ^INTEGER;
        L5: INTEGER;

    PROCEDURE COPYLINKINFO(INFOBLK: INTEGER);
      VAR N, NRECS: INTEGER;
          DONE: BOOLEAN;
          REC: ARRAY [0..7] OF INTEGER;
          BUF: ARRAY [0..31, 0..7] OF INTEGER;

        PROCEDURE GETREC;
        BEGIN
          IF NRECS = 0 THEN
            IF BLOCKREAD(INFILE, BUF, 1, INFOBLK) <> 1 THEN
              BEGIN
                PL := 'Link info read error';
                PROMPT(ERROR);
                DONE := TRUE
              END
            ELSE
              IF BLOCKWRITE(CODE, BUF, 1, OUTBLOCK) <> 1 THEN
                BEGIN
                  PL := 'Code file overflow';
                  PROMPT(ERROR);
                  DONE := TRUE
                END
              ELSE
                BEGIN
                  OUTBLOCK := OUTBLOCK+1;
                  INFOBLK := INFOBLK+1;
                  NRECS := 32
                END;
          IF NOT DONE THEN
            REC := BUF[32-NRECS];
          NRECS := NRECS-1
        END { GETREC } ;

    BEGIN { COPYLINKINFO }
      NRECS := 0; DONE := FALSE;
      REPEAT
        GETREC;
        IF FLIPPED THEN
          BYTESWAP(REC[4]);
        IF NOT (REC[4] IN [0..14]) THEN
          BEGIN
            PL := 'bad link info';
            PROMPT(ERROR);
            REC[4] := 0
          END;
        DONE := REC[4] = 0;
        IF NOT DONE THEN
          IF REC[4] IN [1..5,13,14] THEN
            BEGIN { COPY REF LIST }
              IF FLIPPED THEN
                BYTESWAP(REC[6]);
              N := (REC[6]+7) DIV 8;
              WHILE N > 0 DO
                BEGIN GETREC; N := N-1 END
            END
      UNTIL DONE
    END { COPYLINKINFO } ;

    { Copy the source slot's interface text. Its length is the
      distance to the segment's own code, which follows it; when
      the code comes first, copy a block at a time to the end of
      the file instead. }

    PROCEDURE COPYINTERFACE(START: INTEGER);
      VAR L3, IFSIZE: INTEGER;

      PROCEDURE IFWRERR;
      BEGIN
        PL := 'Interface write error';
        PROMPT(ERROR)
      END;

    BEGIN
      IF START <= 0 THEN
        EXIT(COPYINTERFACE);
      IFSIZE := INTBL.SEGDSC[SSEG].DISKADDR-START;
      CODETABLE.EXTRA[DSEG] := OUTBLOCK;
      IF IFSIZE < 0 THEN
        BEGIN
          IFSIZE := START;
          WHILE BLOCKREAD(INFILE, BUF^, 1, START) = 1 DO
            BEGIN
              START := START+1;
              IF BLOCKWRITE(CODE, BUF^, 1, OUTBLOCK) <> 1 THEN
                BEGIN
                  IFWRERR;
                  EXIT(COPYINTERFACE)
                END;
              OUTBLOCK := OUTBLOCK+1
            END;
          IF IFSIZE = START THEN
            IFSIZE := 0
          ELSE
            EXIT(COPYINTERFACE)
        END;
      IF IFSIZE = 0 THEN
        BEGIN
          PL := 'Interface error';
          PROMPT(ERROR);
          EXIT(COPYINTERFACE)
        END;
      MARK(L4);
      FOR L3 := 1 TO IFSIZE DO
        NEW(BUF);
      IF BLOCKREAD(INFILE, L4^, IFSIZE, START) <> IFSIZE THEN
        BEGIN
          PL := 'Interface read error';
          PROMPT(ERROR)
        END
      ELSE
        IF BLOCKWRITE(CODE, L4^, IFSIZE, OUTBLOCK) <> IFSIZE THEN
          IFWRERR
        ELSE
          OUTBLOCK := OUTBLOCK+IFSIZE;
      RELEASE(L4)
    END { COPYINTERFACE } ;

  BEGIN { LINKIT }
    OUTFULL := FALSE;
    IF INTBL.SEGNAME[SSEG] <> '        ' THEN
      BEGIN
        IF CODETABLE.SEGDSC[DSEG].CODELENG <> 0 THEN
          BEGIN
            PL := 'WARNING - Slot ';
            PROMPT(WINDOW);
            WRITE(DSEG);
            WRITE(' already linked.  Please reconfirm (y/n) - ');
            WRITE(CHR(7));
            READ(INPUT,L3);
            WRITELN(OUTPUT);
            PL := '';
            PROMPT(WINDOW);
            IF NOT (L3 IN ['Y','y']) THEN
              EXIT(LINKIT)
          END;
        WITH INTBL.SEGDSC[SSEG] DO
          BEGIN
            IF DISKADDR = 0 THEN
              NBLOCKS := 0
            ELSE
              NBLOCKS := (CODELENG+511) DIV 512;
            IF (MEMAVAIL+255) DIV 256 < NBLOCKS THEN
              BEGIN
                PL := 'Not enough room to copy slot';
                PROMPT(ERROR)
              END
            ELSE
              BEGIN
                CODETABLE.EXTRA[DSEG] := 0;
                IF INTBL.EXTRA[SSEG] <> 0 THEN
                  COPYINTERFACE(INTBL.EXTRA[SSEG]);
                MARK(L4);
                FOR L5 := 1 TO NBLOCKS DO
                  NEW(BUF);
                IF BLOCKREAD(INFILE,L4^,NBLOCKS,DISKADDR) <> NBLOCKS THEN
                  BEGIN
                    PL := 'Error reading slot ';
                    BELLWAIT(ERROR,SSEG)
                  END
                ELSE
                  IF BLOCKWRITE(CODE,L4^,NBLOCKS,OUTBLOCK) <> NBLOCKS THEN
                    BEGIN
                      PL := 'I/O error - no room on disk';
                      OUTFULL := TRUE;
                      PROMPT(ERROR)
                    END
                  ELSE
                    BEGIN
                      CODETABLE.SEGNAME[DSEG] := INTBL.SEGNAME[SSEG];
                      CODETABLE.SEGDSC[DSEG].CODELENG := CODELENG;
                      IF DISKADDR <> 0 THEN
                        CODETABLE.SEGDSC[DSEG].DISKADDR := OUTBLOCK;
                      OUTBLOCK := OUTBLOCK+NBLOCKS;
                      IF (INTBL.SEGKIND[SSEG] < LINKED) OR
                         (INTBL.SEGKIND[SSEG] > DATASEG) THEN
                        INTBL.SEGKIND[SSEG] := LINKED;
                      CODETABLE.SEGKIND[DSEG] := INTBL.SEGKIND[SSEG];
                      CODETABLE.SEGINFO[DSEG] := INTBL.SEGINFO[SSEG];
                      IF NOT (INTBL.SEGKIND[SSEG] IN
                              [LINKED, LINKED_INTRINS, DATASEG]) THEN
                        COPYLINKINFO(DISKADDR+NBLOCKS)
                    END;
                RELEASE(L4);
                DISPLAY(MARKCODE,CODETABLE)
              END
          END
      END
  END { LINKIT } ;

  FUNCTION CONFIRM: BOOLEAN;
    VAR
      N: INTEGER;
    BEGIN
      CONFIRM:=FALSE;
      (*get segment*)
      N:= 0;
      PL := '';
      PROMPT(WINDOW);
      REPEAT
        READ(CCH);
        IF CCH = CHR(8) THEN
          N := N DIV 10;
        IF CCH IN ['0'..'9'] THEN
          N := N*10 + ORD(CCH)-ORD('0')
      UNTIL NOT (CCH IN [CHR(8),'0'..'9']);
      IF CCH IN ['=','?'] THEN
        BEGIN
          CONFIRM := TRUE;
          EXIT(CONFIRM)
        END;
      IF CCH <> ' ' THEN (*probably N or Q*)
        EXIT(CONFIRM);
      IF N IN [0..MAXSEG] THEN (*good segment number*)
        IF (INTBL.SEGDSC[N].CODELENG > 0) OR
           (INTBL.SEGKIND[N] = DATASEG) THEN
          BEGIN
            SSEG := N;
            REPEAT
              PL := 'Slot to copy into? ';
              PROMPT(WINDOW);
              READ(DSEG)
            UNTIL DSEG IN [0..MAXSEG];
            READ(CCH); { EAT XTRA CHAR }
            CONFIRM := TRUE
          END;
    END; (* CONFIRM *)

BEGIN { LINKCODE }
  IF LENGTH(NTITLE)>0 THEN
    IF BLOCKREAD(INFILE,INTBL,1,0) = 1 THEN
      DISPLAY(MARKIN,INTBL)
    ELSE
      BEGIN
        RSLT:=IORESULT;
        PL := 'Read error # ';
        BELLWAIT(ERROR,RSLT)
      END;
  IF FIRSTFILE THEN
    BEGIN
      IF TABLEFLIPPED(INTBL) THEN
        BEGIN
          FLIPTABLE(INTBL);
          FLIPPED := TRUE
        END
      ELSE
    END
  ELSE
    IF FLIPPED <> TABLEFLIPPED(INTBL) THEN
      BEGIN
        PL := 'Incompatible byte sex';
        PROMPT(ERROR)
      END
    ELSE
      IF FLIPPED THEN
        FLIPTABLE(INTBL);
  PL :=
'Slot # to copy and <space>, = for all, ? for select, N(ew file, Q(uit, A(bort';
  PROMPT(0);
  REPEAT
    IF CONFIRM THEN
      BEGIN
        FIRSTFILE := FALSE;
        IF CCH = '=' THEN
          BEGIN
            PL := 'Copying all slots...';
            PROMPT(WINDOW);
            FOR M2 := 0 TO MAXSEG DO
              BEGIN
                LINKIT(M2,M2);
                IF OUTFULL THEN
                  GOTO 1
              END
          END
        ELSE
          IF CCH = '?' THEN
            BEGIN
              FOR M2 := 0 TO MAXSEG DO
                IF INTBL.SEGDSC[M2].CODELENG > 0 THEN
                  BEGIN
                    PL := 'Copy slot ';
                    PROMPT(WINDOW);
                    WRITE(M2);
                    WRITE('? ');
                    READ(INPUT,CCH);
                    WRITELN(OUTPUT);
                    IF CCH IN ['Y','y'] THEN
                      BEGIN
                        LINKIT(M2,M2);
                        IF OUTFULL THEN
                          GOTO 1
                      END
                  END;
              CCH := 'x'
            END
          ELSE
            LINKIT(SSEG,DSEG);
      { Inside the BEGIN: each label takes one jump-table slot as
        it is placed, and Apple's GOTO has the slot before the IF's }
      1:
      END;
  UNTIL CCH IN ['N','Q','A','n','q','a'];
  CLOSE(INFILE)
END (*LINKCODE*) ;

BEGIN { LIBRARIAN }
  PAGE(OUTPUT);
  PL := 'Apple Pascal Librarian [1.3]';
  PROMPT(0);
  REPEAT
    PL := 'Output file -> ';
    PROMPT(MARKCODE-2);
    READLN(INPUT,TITLE);
    IF TITLE = '*' THEN
      TITLE := '*SYSTEM.LIBRARY';
    IF LENGTH(TITLE) = 0 THEN EXIT(LIBRARIAN)
    ELSE REWRITE(CODE,TITLE)
  UNTIL (LENGTH(TITLE) = 0) OR (CHECKIO);
  OUTBLOCK := 1; NEW(BUF);
  FIRSTFILE := TRUE;
  FLIPPED := FALSE;
  FILLCHAR(CODETABLE, SIZEOF(BLOCK0), 0);
  WITH CODETABLE DO
    FOR DSEG := 0 TO MAXSEG DO
      SEGNAME[DSEG] := '        ';
  REPEAT
    OPENFILE;
    LINKCODE;
  UNTIL CCH IN ['Q','q','A','a'];
  IF CCH IN ['A','a'] THEN EXIT(LIBRARIAN);
  GOTOXY(0,ERROR);
  WRITE('Notice? ');
  WRITE(CHR(7));
  WITH SYSCOM^.CRTCTRL DO WRITE(ESCAPE,ERASEEOL);
  READLN(INPUT,NOTICEIN);
  IF NOTICEIN <> '' THEN
    CODETABLE.NOTICE := NOTICEIN;
  IF FLIPPED THEN
    FLIPTABLE(CODETABLE);
  IF BLOCKWRITE(CODE,CODETABLE,1,0) = 1 THEN
    CLOSE(CODE,LOCK)
  ELSE
    WRITELN(OUTPUT,'Code write error')
END { LIBRARIAN } ;

BEGIN END.
