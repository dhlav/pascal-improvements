{ Segment 11. II.0's STATEMENT is one procedure of BODYPART with nine
  nested ones. Apple's is a SEGMENT PROCEDURE, and two of the nine --
  CASESTATEMENT and FORSTATEMENT -- are segments of their own.

  That is what fixes where those two are declared. A segment procedure's
  procedure 1 comes out at the lexical level of its declaration, and
  CASESTAT.1 and FORSTATE.1 are both lex 3, so each is declared inside a
  procedure of a phase rather than beside one; the `CXP 12,1` and `CXP 13,1`
  in the CASE below say which procedure that is. Everything the seven local
  procedures call in BODYPART reaches them as `CXP 9,n`, which BODYPART's
  forward block already provides.

  The declaration order is read off the code. Bodies are emitted in
  declaration order, so ASSIGNMENT at $0000 through WITHSTATEMENT at $0410,
  with STATEMENT's own body last at $04EE, is the order they were written
  in. CASESTAT and FORSTATE take no procedure number here -- a segment does
  not -- so the seven numbered procedures below are unaffected by where the
  two segments sit, and their own numbers, 12 before 13, are the whole of
  what fixes their order relative to each other.

  Where they sit is not a choice either, and the compiler is the one that
  says so: PROCDECLARATION opens with

      IF SEGDEC THEN IF CODEINSEG THEN ERROR(399)

  and CODEINSEG goes true in BODY3 the moment any procedure body is written
  into the current segment. So a segment procedure has to be declared before
  the first ordinary procedure of the block that encloses it -- which is
  where BODYPART puts ROUTINE, STATEMENT, BODY1 and BODY3, and where these
  two have to go. II.0, having no segments here, declares CASESTATEMENT
  between IFSTATEMENT and REPEATSTATEMENT; Apple could not. }

  LABEL 1;

  VAR LCP: CTP;
      LDISX: DISPRANGE;
      LLP: LABELP;
      LMARKP: ^INTEGER;
      LLC: ADDRRANGE;

{SEGMENT CASESTAT}

{SEGMENT FORSTATE}

  PROCEDURE ASSIGNMENT(FCP: CTP);
    VAR LATTR: ATTR;
	LSTRG: BOOLEAN;
	LSTRGCST: BOOLEAN;
	LMIN,LMAX: INTEGER;
  BEGIN
    SELECTOR(FSYS + [BECOMES],FCP);
    IF SY = BECOMES THEN
      BEGIN
	LMAX := 0; LSTRGCST := FALSE;
	IF GATTR.TYPTR <> NIL THEN
	  IF (GATTR.ACCESS = INDRCT) OR (GATTR.TYPTR^.FORM > POWER) THEN
	    LOADADDRESS;
	LSTRG := PAOFCHAR(GATTR.TYPTR);
	LATTR := GATTR;
	INSYMBOL; EXPRESSION(FSYS);
	IF GATTR.KIND = CST THEN
	  LSTRGCST := (GATTR.TYPTR = CHARPTR) OR STRGTYPE(GATTR.TYPTR);
	IF GATTR.TYPTR <> NIL THEN
	  IF GATTR.TYPTR^.FORM <= POWER THEN LOAD
	  ELSE LOADADDRESS;
	IF (LATTR.TYPTR <> NIL) AND (GATTR.TYPTR <> NIL) THEN
	  BEGIN
	    IF GATTR.TYPTR = INTPTR THEN
	      IF COMPTYPES(REALPTR,LATTR.TYPTR) THEN
		BEGIN GEN0(10(*FLT*)); GATTR.TYPTR := REALPTR END;
	    IF COMPTYPES(LONGINTPTR,LATTR.TYPTR) THEN
	      BEGIN
		IF GATTR.TYPTR = INTPTR THEN
		  BEGIN GENLDC(18); GENNR(30,4);
		    GATTR.TYPTR := LONGINTPTR
		  END;
		IF GATTR.TYPTR^.FORM <> LONGINT THEN
		  BEGIN ERROR(129); GATTR.TYPTR := LONGINTPTR END
	      END;
	    IF LSTRG THEN
	      IF LATTR.TYPTR^.AISSTRNG THEN
  { The empty ELSE is in the binary: the branch ends with a jump over an
    else part that is not there, and a plain IF-THEN does not emit one. It
    is what keeps the ELSE below with the outer IF. }
		IF LSTRGCST AND (GATTR.TYPTR = CHARPTR) THEN
		  GATTR.TYPTR := STRGPTR
		ELSE
	      ELSE
		BEGIN
		  IF LATTR.TYPTR = BYTEPTR THEN
		    IF COMPTYPES(LATTR.TYPTR,GATTR.TYPTR) THEN
		      BEGIN
			IF GATTR.TYPTR = BYTEPTR THEN ERROR(129)
			ELSE LATTR.TYPTR := GATTR.TYPTR
		      END
		    ELSE
		      IF LSTRGCST AND (GATTR.TYPTR <> CHARPTR) THEN
			BEGIN STRGTOPA(STRGCSTIC);
			  GEN1(40(*MOV*),
			       (GATTR.TYPTR^.MAXLENG + 1) DIV 2);
			  EXIT(ASSIGNMENT)
			END;
		  IF LATTR.TYPTR^.INXTYPE <> NIL THEN
		    BEGIN
		      GETBOUNDS(LATTR.TYPTR^.INXTYPE,LMIN,LMAX);
		      LMAX := LMAX - LMIN + 1;
		      IF LSTRGCST AND (GATTR.TYPTR <> CHARPTR) THEN
			BEGIN STRGTOPA(STRGCSTIC);
			  IF LMAX <> GATTR.TYPTR^.MAXLENG THEN ERROR(129);
			  GATTR.TYPTR := LATTR.TYPTR
			END
		    END
		  ELSE GATTR.TYPTR := LATTR.TYPTR
		END;
	    IF COMPTYPES(LATTR.TYPTR,GATTR.TYPTR) THEN
	      CASE LATTR.TYPTR^.FORM OF
		SUBRANGE:
		  BEGIN
		    IF RANGECHECK THEN
		      BEGIN GENLDC(LATTR.TYPTR^.MIN.IVAL);
			GENLDC(LATTR.TYPTR^.MAX.IVAL);
			GEN0(8(*CHK*))
		      END;
		    STORE(LATTR)
		  END;
		POWER:
		  BEGIN GEN1(32(*ADJ*),LATTR.TYPTR^.SIZE);
		    STORE(LATTR)
		  END;
		SCALAR,POINTER:
		  STORE(LATTR);
		LONGINT:
		  BEGIN GENLDC(LATTR.TYPTR^.SIZE); GENLDC(0);
		    GENNR(30,4); STORE(LATTR)
		  END;
		ARRAYS:
		  IF LSTRG THEN
		    IF LATTR.TYPTR^.AISSTRNG THEN
		      GEN1(42(*SAS*),LATTR.TYPTR^.MAXLENG)
		    ELSE GEN1(40(*MOV*),(LMAX + 1) DIV 2)
		  ELSE
		    BEGIN
		      IF LATTR.TYPTR = WORDPTR THEN
			IF GATTR.TYPTR = WORDPTR THEN ERROR(129)
			ELSE LATTR.TYPTR := GATTR.TYPTR;
		      GEN1(40(*MOV*),LATTR.TYPTR^.SIZE)
		    END;
		RECORDS:
		  GEN1(40(*MOV*),LATTR.TYPTR^.SIZE);
		FILES:
		  ERROR(146)
	      END (*CASE*)
	    ELSE ERROR(129)
	  END
      END
    ELSE ERROR(51)
  END (*ASSIGNMENT*) ;

  PROCEDURE GOTOSTATEMENT;
    VAR LLP: LABELP;
	LFOUND: BOOLEAN;
	LDISX: DISPRANGE;
  BEGIN
    IF NOT GOTOOK THEN ERROR(6);
    IF SY = INTCONST THEN
      BEGIN LFOUND := FALSE;
	LDISX := TOP;
	WHILE DISPLAY[LDISX].OCCUR <> BLCK DO LDISX := LDISX - 1;
	LLP := DISPLAY[LDISX].FLABEL;
	WHILE (LLP <> NIL) AND NOT LFOUND DO
	  WITH LLP^ DO
	    IF LABVAL = VAL.IVAL THEN
	      BEGIN LFOUND := TRUE; GENJMP(57(*UJP*),CODELBP) END
	    ELSE LLP := NEXTLAB;
	IF NOT LFOUND THEN ERROR(167);
	INSYMBOL
      END
    ELSE ERROR(15)
  END (*GOTOSTATEMENT*) ;

  PROCEDURE COMPOUNDSTATEMENT;
  BEGIN
    REPEAT
      REPEAT STATEMENT(FSYS + [SEMICOLON,ENDSY])
      UNTIL NOT (SY IN STATBEGSYS);
      TEST := SY <> SEMICOLON;
      IF NOT TEST THEN INSYMBOL
    UNTIL TEST;
    IF SY = ENDSY THEN INSYMBOL ELSE ERROR(13)
  END (*COMPOUNDSTATEMENT*) ;

  PROCEDURE IFSTATEMENT;
{ LCONST is set FALSE and never set again, and nothing anywhere assigns
  LVAL, so the three IF LCONST arms below cannot be reached. They are in
  the binary all the same and the frame carries both words. }
    VAR LCIX1,LCIX2: LBP;
	LOLDIC: ADDRRANGE;
	LCONST,LVAL: BOOLEAN;
  BEGIN
    LCONST := FALSE;
    EXPRESSION(FSYS + [THENSY]);
    IF NOT LCONST THEN BEGIN GENLABEL(LCIX1); GENFJP(LCIX1) END;
    IF SY = THENSY THEN INSYMBOL ELSE ERROR(52);
    STATEMENT(FSYS + [ELSESY]);
    IF LCONST THEN
      IF LVAL THEN IC := LOLDIC
      ELSE LOLDIC := IC;
    IF SY = ELSESY THEN
      BEGIN
	IF NOT LCONST THEN
	  BEGIN GENLABEL(LCIX2); GENJMP(57(*UJP*),LCIX2);
	    PUTLABEL(LCIX1)
	  END;
	INSYMBOL; STATEMENT(FSYS);
	IF LCONST THEN
	  BEGIN IF NOT LVAL THEN IC := LOLDIC END
	ELSE PUTLABEL(LCIX2)
      END
    ELSE
      IF NOT LCONST THEN PUTLABEL(LCIX1)
  END (*IFSTATEMENT*) ;

  PROCEDURE REPEATSTATEMENT;
    VAR LADDR: LBP;
  BEGIN
    GENLABEL(LADDR); PUTLABEL(LADDR);
    REPEAT
      REPEAT STATEMENT(FSYS + [SEMICOLON,UNTILSY])
      UNTIL NOT (SY IN STATBEGSYS);
      TEST := SY <> SEMICOLON;
      IF NOT TEST THEN INSYMBOL
    UNTIL TEST;
    IF SY = UNTILSY THEN
      BEGIN INSYMBOL; EXPRESSION(FSYS); GENFJP(LADDR) END
    ELSE ERROR(53)
  END (*REPEATSTATEMENT*) ;

  PROCEDURE WHILESTATEMENT;
    VAR LADDR,LCIX: LBP;
  BEGIN
    GENLABEL(LADDR); PUTLABEL(LADDR);
    EXPRESSION(FSYS + [DOSY]);
    GENLABEL(LCIX); GENFJP(LCIX);
    IF SY = DOSY THEN INSYMBOL ELSE ERROR(54);
    STATEMENT(FSYS);
    GENJMP(57(*UJP*),LADDR); PUTLABEL(LCIX)
  END (*WHILESTATEMENT*) ;

  PROCEDURE WITHSTATEMENT;
{ LLC is declared and never used. II.0's WITHSTATEMENT saves LC here and
  restores it at the end; Apple moved that up into STATEMENT, where it now
  brackets every statement rather than only a WITH, and the word stayed. }
    VAR LCP: CTP;
	LLC: ADDRRANGE;
	LCNT: DISPRANGE;
  BEGIN LCNT := 0;
    REPEAT
      IF SY = IDENT THEN
	BEGIN SEARCHID(VARS + [FIELD],LCP); INSYMBOL END
      ELSE BEGIN ERROR(2); LCP := UVARPTR END;
      SELECTOR(FSYS + [COMMA,DOSY],LCP);
      IF GATTR.TYPTR <> NIL THEN
	IF GATTR.TYPTR^.FORM = RECORDS THEN
	  IF TOP < DISPLIMIT THEN
	    BEGIN TOP := TOP + 1; LCNT := LCNT + 1;
	      WITH DISPLAY[TOP] DO FNAME := GATTR.TYPTR^.FSTFLD;
	      IF GATTR.ACCESS = DRCT THEN
		WITH DISPLAY[TOP] DO
		  BEGIN OCCUR := CREC; CLEV := GATTR.VLEVEL;
		    CDSPL := GATTR.DPLMT
		  END
	      ELSE
		BEGIN LOADADDRESS;
		  GEN2(56(*STR*),LEVEL,LC);
		  WITH DISPLAY[TOP] DO
		    BEGIN OCCUR := VREC; VDSPL := LC END;
		  LC := LC + 1
		END
	    END
	  ELSE ERROR(250)
	ELSE ERROR(140);
      TEST := SY <> COMMA;
      IF NOT TEST THEN INSYMBOL
    UNTIL TEST;
    IF SY = DOSY THEN INSYMBOL ELSE ERROR(54);
    STATEMENT(FSYS);
    TOP := TOP - LCNT
  END (*WITHSTATEMENT*) ;

BEGIN (*STATEMENT*)
  STMTLEV := STMTLEV + 1;
  IF SY = INTCONST THEN
    BEGIN (*LABEL*)
      LDISX := TOP;
      WHILE DISPLAY[LDISX].OCCUR <> BLCK DO LDISX := LDISX - 1;
      LLP := DISPLAY[LDISX].FLABEL;
      WHILE LLP <> NIL DO
	WITH LLP^ DO
	  IF LABVAL = VAL.IVAL THEN
	    BEGIN
	      IF CODELBP^.DEFINED THEN ERROR(165);
	      PUTLABEL(CODELBP); GOTO 1
	    END
	  ELSE LLP := NEXTLAB;
      ERROR(167);
  1:  INSYMBOL;
      IF SY = COLON THEN INSYMBOL ELSE ERROR(5)
    END;
  IF DEBUGGING THEN
    BEGIN GEN1(85(*BPT*),SCREENDOTS + 1); BPTONLINE := TRUE END;
  IF NOT (SY IN FSYS + [IDENT]) THEN
    BEGIN ERROR(6); SKIP(FSYS) END;
  IF SY IN STATBEGSYS + [IDENT] THEN
    BEGIN
      MARK(LMARKP); LLC := LC;
      CASE SY OF
	IDENT:
	  BEGIN SEARCHID(VARS + [FIELD,PROC,FUNC],LCP); INSYMBOL;
	    IF LCP^.KLASS = PROC THEN CALL(FSYS,LCP)
	    ELSE ASSIGNMENT(LCP)
	  END;
	BEGINSY:  BEGIN INSYMBOL; COMPOUNDSTATEMENT END;
	GOTOSY:   BEGIN INSYMBOL; GOTOSTATEMENT END;
	IFSY:     BEGIN INSYMBOL; IFSTATEMENT END;
	CASESY:   BEGIN INSYMBOL; CASESTATEMENT END;
	WHILESY:  BEGIN INSYMBOL; WHILESTATEMENT END;
	REPEATSY: BEGIN INSYMBOL; REPEATSTATEMENT END;
	FORSY:    BEGIN INSYMBOL; FORSTATEMENT END;
	WITHSY:   BEGIN INSYMBOL; WITHSTATEMENT END
      END (*CASE*);
      IF LC > LCMAX THEN LCMAX := LC;
      LC := LLC; RELEASE(LMARKP);
      IF IC + 100 > MAXCODE THEN BEGIN ERROR(253); IC := 0 END;
      IF NOT (SY IN [SEMICOLON,ENDSY,UNTILSY,ELSESY,SEPARATSY]) THEN
	BEGIN ERROR(6); SKIP(FSYS) END
    END;
  STMTLEV := STMTLEV - 1
END;
