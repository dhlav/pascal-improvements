{ Segment 7. UCSD II.0's COMPINIT, which is compinit.text, with four
  additions and one deletion.

  The deletion is the transcendentals. II.0's ENTSTDPROCS enters SIN, COS,
  LOG, ATAN, LN, EXP and SQRT as standard functions; Apple moved them into
  the TRANSCEND unit, so nineteen names become thirteen.

  The additions are Apple's: a version check against SYSTEM.PASCAL before
  anything else runs, a prompt for the listing file, and -- the one that
  changes the shape of two procedures -- the standard identifier names are
  no longer forty-three assignments to an ARRAY OF ALPHA. They are five
  dot-separated string literals appended into one 512-byte buffer, and
  NEXTNAME cuts the next name out of it with SCAN. The buffer and its two
  cursors live in COMPINIT's own frame, which is why both helpers reach
  them with LOD 1,n. }

  VAR LNAMES: PACKED ARRAY [0..511] OF CHAR;
      NPOS: INTEGER;
      NCHARS: INTEGER;
      LTITLE: STRING;
      LOK: BOOLEAN;
      LERR: INTEGER;
      LLENG: INTEGER;
      LI: INTEGER;

  PROCEDURE ENTSTDTYPES;
    VAR LSP: STP;
  BEGIN
    NEW(INTPTR,SCALAR,STANDARD);
    WITH INTPTR^ DO
      BEGIN SIZE := INTSIZE; FORM := SCALAR; SCALKIND := STANDARD END;
    NEW(REALPTR,SCALAR,STANDARD);
    WITH REALPTR^ DO
      BEGIN SIZE := REALSIZE; FORM := SCALAR; SCALKIND := STANDARD END;
    NEW(LONGINTPTR,LONGINT);
    WITH LONGINTPTR^ DO
      BEGIN SIZE := DECSIZE((BITSPERWD-1)*100 DIV 332 + 1); FORM := LONGINT END;
    NEW(CHARPTR,SCALAR,STANDARD);
    WITH CHARPTR^ DO
      BEGIN SIZE := CHARSIZE; FORM := SCALAR; SCALKIND := STANDARD END;
    NEW(BOOLPTR,SCALAR,DECLARED);
    WITH BOOLPTR^ DO
      BEGIN SIZE := BOOLSIZE; FORM := SCALAR; SCALKIND := DECLARED END;
    NEW(NILPTR,POINTER);
    WITH NILPTR^ DO
      BEGIN SIZE := PTRSIZE; FORM := POINTER; ELTYPE := NIL END;
    NEW(TEXTPTR,FILES);
    WITH TEXTPTR^ DO
      BEGIN SIZE := FILESIZE+CHARSIZE; FORM := FILES; FILTYPE := CHARPTR END;
    NEW(INTRACTVPTR,FILES);
    WITH INTRACTVPTR^ DO
      BEGIN SIZE := FILESIZE+CHARSIZE; FORM := FILES; FILTYPE := CHARPTR END;
    NEW(STRGPTR,ARRAYS,TRUE,TRUE);
    WITH STRGPTR^ DO
      BEGIN FORM := ARRAYS; SIZE := (DEFSTRGLGTH + CHRSPERWD) DIV CHRSPERWD;
	AISPACKD := TRUE; AISSTRNG := TRUE; INXTYPE := INTPTR;
	ELWIDTH := BITSPERCHR; ELSPERWD := CHRSPERWD;
	AELTYPE := CHARPTR; MAXLENG := DEFSTRGLGTH
      END;
{ Apple's two. BYTESTREAM and WORDSTREAM are the types a UNITREAD or a
  BLOCKREAD buffer is checked against, and both are an array over the whole
  address space -- packed characters for one, words for the other. }
    NEW(LSP,SUBRANGE);
    WITH LSP^ DO
      BEGIN FORM := SUBRANGE; SIZE := 1; RANGETYPE := INTPTR;
	MIN.IVAL := 0; MAX.IVAL := MAXINT
      END;
    NEW(BYTEPTR,ARRAYS,TRUE,FALSE);
    WITH BYTEPTR^ DO
      BEGIN FORM := ARRAYS; SIZE := 1;
	AISPACKD := TRUE; AISSTRNG := FALSE; INXTYPE := LSP;
	ELWIDTH := BITSPERCHR; ELSPERWD := CHRSPERWD;
	AELTYPE := CHARPTR
      END;
    NEW(WORDPTR,ARRAYS,FALSE);
    WITH WORDPTR^ DO
      BEGIN FORM := ARRAYS; SIZE := 1;
	AISPACKD := FALSE; INXTYPE := LSP; AELTYPE := INTPTR
      END
  END (*ENTSTDTYPES*) ;

  PROCEDURE ENTSTDNAMES;
    VAR CP,CP1: CTP; I: INTEGER;
  BEGIN
    NEW(CP,TYPES);
    WITH CP^ DO
      BEGIN NAME := 'INTEGER '; IDTYPE := INTPTR; KLASS := TYPES END;
    ENTERID(CP);
    NEW(CP,TYPES);
    WITH CP^ DO
      BEGIN NAME := 'REAL    '; IDTYPE := REALPTR; KLASS := TYPES END;
    ENTERID(CP);
    NEW(CP,TYPES);
    WITH CP^ DO
      BEGIN NAME := 'CHAR    '; IDTYPE := CHARPTR; KLASS := TYPES END;
    ENTERID(CP);
    NEW(CP,TYPES);
    WITH CP^ DO
      BEGIN NAME := 'BOOLEAN '; IDTYPE := BOOLPTR; KLASS := TYPES END;
    ENTERID(CP);
    NEW(CP,TYPES);
    WITH CP^ DO
      BEGIN NAME := 'STRING  '; IDTYPE := STRGPTR; KLASS := TYPES END;
    ENTERID(CP);
    NEW(CP,TYPES);
    WITH CP^ DO
      BEGIN NAME := 'TEXT    '; IDTYPE := TEXTPTR; KLASS := TYPES END;
    ENTERID(CP);
    NEW(CP,TYPES);
    WITH CP^ DO
      BEGIN NAME := 'INTERACT'; IDTYPE := INTRACTVPTR; KLASS := TYPES END;
    ENTERID(CP);
    NEW(INPUTPTR,FORMALVARS,FALSE);
    WITH INPUTPTR^ DO
      BEGIN NAME := 'INPUT   '; IDTYPE := TEXTPTR; KLASS := FORMALVARS;
	VLEV := 0; VADDR := 2
      END;
    ENTERID(INPUTPTR);
    NEW(OUTPUTPTR,FORMALVARS,FALSE);
    WITH OUTPUTPTR^ DO
      BEGIN NAME := 'OUTPUT  '; IDTYPE := TEXTPTR; KLASS := FORMALVARS;
	VLEV := 0; VADDR := 3
      END;
    ENTERID(OUTPUTPTR);
    NEW(CP,FORMALVARS,FALSE);
    WITH CP^ DO
      BEGIN NAME := 'KEYBOARD'; IDTYPE := TEXTPTR; KLASS := FORMALVARS;
	VLEV := 0; VADDR := 4
      END;
    ENTERID(CP);
    CP1 := NIL;
    FOR I := 0 TO 1 DO
      BEGIN NEW(CP,KONST);
	WITH CP^ DO
	  BEGIN IDTYPE := BOOLPTR;
	    IF I = 0 THEN NAME := 'FALSE   '
	    ELSE NAME := 'TRUE    ';
	    NEXT := CP1; VALUES.IVAL := I; KLASS := KONST
	  END;
	ENTERID(CP); CP1 := CP
      END;
    BOOLPTR^.FCONST := CP;
    NEW(CP,KONST);
    WITH CP^ DO
      BEGIN NAME := 'NIL     '; IDTYPE := NILPTR;
	NEXT := NIL; VALUES.IVAL := 0; KLASS := KONST
      END;
    ENTERID(CP);
    NEW(CP,KONST);
    WITH CP^ DO
      BEGIN NAME := 'MAXINT  '; IDTYPE := INTPTR;
	KLASS := KONST; VALUES.IVAL := MAXINT
      END;
    ENTERID(CP);
    NEW(CP,TYPES);
    WITH CP^ DO
      BEGIN NAME := 'BYTESTRE'; IDTYPE := BYTEPTR; NEXT := NIL;
	KLASS := TYPES
      END;
    ENTERID(CP);
    NEW(CP,TYPES);
    WITH CP^ DO
      BEGIN NAME := 'WORDSTRE'; IDTYPE := WORDPTR; NEXT := NIL;
	KLASS := TYPES
      END;
    ENTERID(CP)
  END (*ENTSTDNAMES*) ;

  PROCEDURE ENTUNDECL;
{ II.0 repeats the eight blanks at every one of the six records. Apple
  assigns it to a local once and moves that. }
    VAR LNAME: ALPHA;
  BEGIN
    LNAME := '        ';
    NEW(UTYPPTR,TYPES);
    WITH UTYPPTR^ DO
      BEGIN NAME := LNAME; IDTYPE := NIL; KLASS := TYPES END;
    NEW(UCSTPTR,KONST);
    WITH UCSTPTR^ DO
      BEGIN NAME := LNAME; IDTYPE := NIL; NEXT := NIL;
	VALUES.IVAL := 0; KLASS := KONST
      END;
    NEW(UVARPTR,ACTUALVARS,FALSE);
    WITH UVARPTR^ DO
      BEGIN NAME := LNAME; IDTYPE := NIL;
	NEXT := NIL; VLEV := 0; VADDR := 0; KLASS := ACTUALVARS
      END;
    NEW(UFLDPTR,FIELD);
    WITH UFLDPTR^ DO
      BEGIN NAME := LNAME; IDTYPE := NIL; NEXT := NIL;
	FLDADDR := 0; KLASS := FIELD
      END;
    NEW(UPRCPTR,PROC,DECLARED,ACTUAL,FALSE);
    WITH UPRCPTR^ DO
      BEGIN NAME := LNAME; IDTYPE := NIL; FORWDECL := FALSE;
	NEXT := NIL; INSCOPE := FALSE; LOCALLC := 0; EXTURNAL := FALSE;
	PFLEV := 0; PFNAME := 0; PFSEG := 0;
	KLASS := PROC; PFDECKIND := DECLARED; PFKIND := ACTUAL
      END;
    NEW(UFCTPTR,FUNC,DECLARED,ACTUAL,FALSE);
    WITH UFCTPTR^ DO
      BEGIN NAME := LNAME; IDTYPE := NIL; NEXT := NIL;
	FORWDECL := FALSE; EXTURNAL := FALSE; INSCOPE := FALSE; LOCALLC := 0;
	PFLEV := 0; PFNAME := 0; PFSEG := 0;
	KLASS := FUNC; PFDECKIND := DECLARED; PFKIND := ACTUAL
      END
  END (*ENTUNDECL*) ;

  PROCEDURE ADDNAMES(S: STRING);
  BEGIN
    IF NCHARS + LENGTH(S) < 511 THEN
      MOVELEFT(S[1],LNAMES[NCHARS],LENGTH(S));
    NCHARS := NCHARS + LENGTH(S)
  END (*ADDNAMES*) ;

  PROCEDURE NEXTNAME(VAR A: ALPHA);
    VAR L: INTEGER;
  BEGIN
    A := '        ';
    L := SCAN(8,='.',LNAMES[NPOS]);
    MOVELEFT(LNAMES[NPOS],A,L);
    NPOS := NPOS + L + 1
  END (*NEXTNAME*) ;

  PROCEDURE ENTSPCPROCS;
    LABEL 1;
    VAR LCP: CTP; I: INTEGER; ISFUNC: BOOLEAN; LNAME: ALPHA;
  BEGIN
    NCHARS := 0;
    ADDNAMES('READ.READLN.WRITE.WRITELN.EOF.EOLN.PRED.SUCC.ORD.SQR.ABS.NEW.');
    ADDNAMES('UNITREAD.UNITWRIT.CONCAT.LENGTH.INSERT.DELETE.COPY.POS.');
    ADDNAMES('MOVELEFT.MOVERIGH.EXIT.IDSEARCH.TREESEAR.TIME.FILLCHAR.');
    ADDNAMES('OPENNEW.OPENOLD.REWRITE.CLOSE.SEEK.RESET.GET.PUT.SCAN.');
    ADDNAMES('BLOCKREA.BLOCKWRI.TRUNC.PAGE.SIZEOF.STR.GOTOXY.UNITSTAT.');
    NPOS := 0;
    I := 0;
    WHILE NPOS < NCHARS DO
      BEGIN
	NEXTNAME(LNAME);
	I := I + 1;
	IF TINY THEN
	  IF I IN [2,7,10,13,17,18,19,20,32,34,35,40,42,43,44] THEN
	    GOTO 1;
	ISFUNC := I IN [5,6,7,8,9,10,11,15,16,19,20,25,36,37,38,39,41];
	IF ISFUNC THEN NEW(LCP,FUNC,SPECIAL)
	ELSE NEW(LCP,PROC,SPECIAL);
	WITH LCP^ DO
	  BEGIN NAME := LNAME; NEXT := NIL; IDTYPE := NIL;
	    IF ISFUNC THEN KLASS := FUNC ELSE KLASS := PROC;
	    PFDECKIND := SPECIAL; KEY := I
	  END;
	ENTERID(LCP);
1:    END
  END (*ENTSPCPROCS*) ;

  PROCEDURE ENTSTDPROCS;
    LABEL 1;
    VAR LCP,PARAM: CTP; LSP,FTYPE: STP; I: INTEGER; ISPROC: BOOLEAN;
	LNAME: ALPHA;
  BEGIN
    NCHARS := 0;
    ADDNAMES('ODD.CHR.XXX.ROUND.MARK.RELEASE.');
    ADDNAMES('IORESULT.UNITBUSY.PWROFTEN.UNITWAIT.UNITCLEA.HALT.MEMAVAIL.');
    NPOS := 0;
    I := 0;
    WHILE NPOS < NCHARS DO
      BEGIN
	NEXTNAME(LNAME);
	I := I + 1;
	ISPROC := I IN [5,6,10,11,12];
	CASE I OF
	  1:  BEGIN FTYPE := BOOLPTR; NEW(PARAM,ACTUALVARS,FALSE);
		WITH PARAM^ DO
		  BEGIN IDTYPE := INTPTR; KLASS := ACTUALVARS END
	      END;
	  2:  FTYPE := CHARPTR;
	  3:  GOTO 1;
	  4:  BEGIN FTYPE := INTPTR; NEW(PARAM,ACTUALVARS,FALSE);
		WITH PARAM^ DO BEGIN IDTYPE := REALPTR; KLASS := ACTUALVARS END
	      END;
	  5:  BEGIN FTYPE := NIL; NEW(PARAM,FORMALVARS,FALSE); NEW(LSP,POINTER);
		WITH LSP^ DO
		  BEGIN SIZE := PTRSIZE; FORM := POINTER; ELTYPE := NIL END;
		WITH PARAM^ DO BEGIN IDTYPE := LSP; KLASS := FORMALVARS END
	      END;
	  7:  BEGIN FTYPE := INTPTR; PARAM := NIL END;
	  8:  BEGIN FTYPE := BOOLPTR; NEW(PARAM,ACTUALVARS,FALSE);
		WITH PARAM^ DO
		  BEGIN IDTYPE := INTPTR; KLASS := ACTUALVARS END
	      END;
	  9:  FTYPE := REALPTR;
	 10:  FTYPE := NIL;
	 12:  BEGIN FTYPE := NIL; PARAM := NIL END;
	 13:  FTYPE := INTPTR
	END (*PARAM AND TYPE CASES*) ;
	IF ISPROC THEN NEW(LCP,PROC,STANDARD)
	ELSE NEW(LCP,FUNC,STANDARD);
	WITH LCP^ DO
	  BEGIN NAME := LNAME; PFDECKIND := STANDARD;
	    IF I < 5 THEN CSPNUM := I + 20 ELSE CSPNUM := I + 27;
	    IF ISPROC THEN KLASS := PROC ELSE KLASS := FUNC;
	    IF PARAM <> NIL THEN PARAM^.NEXT := NIL;
	    IDTYPE := FTYPE; NEXT := PARAM
	  END;
	ENTERID(LCP);
1:    END
  END (*ENTSTDPROCS*) ;

  PROCEDURE INITSCALARS;
  BEGIN
    IF (MEMAVAIL < 0) OR (MEMAVAIL > 9950) THEN
      SWAPPING := FALSE ELSE SWAPPING := TRUE;
    SWAPMORE := FALSE;
    FWPTR := NIL; MODPTR := NIL; GLOBTESTP := NIL;
    LINESTART := 0; LINEINFO := LCAFTERMARKSTACK; LIST := FALSE;
    SYMBLK := 2; SCREENDOTS := 0; STARTDOTS := 0;
    FOR SEG := 0 TO MAXSEG DO
      WITH SEGTABLE[SEG] DO
	BEGIN DISKADDR := 0; CODELENG := 0; SEGNAME := '        ';
	  SEGKIND := 0; TEXTADDR := 0; SEGNUM := 0
	END;
    USINGLIST := NIL;
    SYSTEMLIB := '*';
    LC := LCAFTERMARKSTACK; IOCHECK := TRUE; DP := TRUE;
    SEGSLOT := 0; SEGINX := 0; NEXTJTAB := 1; NEXTPROC := 2; CURPROC := 1;
    NEW(SCONST); NEW(SYMBUFP); CODEP := NIL;
    LINKINFO := FALSE; INTRINSIC := FALSE; LSEPPROC := FALSE;
    VARSTRG := TRUE;
    SEG := 1; NEXTSEG := 7; CURBLK := 1; CURBYTE := 0;
    NOISY := NOT USERINFO.SLOWTERM; OPT_E := FALSE;
    DEBUGGING := FALSE; BPTONLINE := FALSE; INMODULE := FALSE;
    NOLOAD := FALSE;
    GOTOOK := FALSE; RANGECHECK := TRUE; SYSCOMP := FALSE; TINY := FALSE;
    CODEINSEG := FALSE; PRTERR := TRUE; INCLUDING := FALSE; USING := FALSE;
    COMMENT := NIL; LIBNOTOPEN := TRUE;
    GETSTMTLEV := TRUE; BEGSTMTLEV := 0;
    FLIPBYTES := FALSE;
    SEGSUSED := []
  END (*INITSCALARS*) ;

  PROCEDURE INITSETS;
  BEGIN
    CONSTBEGSYS := [ADDOP,INTCONST,REALCONST,STRINGCONST,IDENT,LONGCONST];
    SIMPTYPEBEGSYS := [LPARENT] + CONSTBEGSYS;
    TYPEBEGSYS := [ARROW,PACKEDSY,ARRAYSY,RECORDSY,SETSY,FILESY]
		  + SIMPTYPEBEGSYS;
    TYPEDELS := [ARRAYSY,RECORDSY,SETSY,FILESY];
    BLOCKBEGSYS := [USESSY,LABELSY,CONSTSY,TYPESY,VARSY,
		    PROCSY,FUNCSY,PROGSY,BEGINSY];
    SELECTSYS := [ARROW,PERIOD,LBRACK];
    FACBEGSYS := [INTCONST,REALCONST,LONGCONST,STRINGCONST,IDENT,
		  LPARENT,LBRACK,NOTSY];
    STATBEGSYS := [BEGINSY,GOTOSY,IFSY,WHILESY,REPEATSY,FORSY,WITHSY,CASESY];
    VARS := [FORMALVARS,ACTUALVARS]
  END (*INITSETS*) ;

  PROCEDURE CHECKVERS;
{ Apple's, and it runs before anything else. SYSTEM.PASCAL keeps its
  version byte at $BF21 and a machine-configuration bit at $BF22, and
  neither is reachable through a declaration -- hence the variant record,
  which is the only way to put an address in a pointer.

  The two arms are separate types because the two reads are different
  widths: a byte for the version, a bit for the 128K flag, which the
  binary takes with IXP 16,1. }
    TYPE VBYTES = PACKED ARRAY [0..1] OF CHAR;
	 VBITS = PACKED ARRAY [0..15] OF BOOLEAN;
    VAR LPTR: RECORD
		CASE BOOLEAN OF
		  TRUE:  (LADDR: INTEGER);
		  FALSE: (CASE BOOLEAN OF
			    TRUE:  (LBYTE: ^ VBYTES);
			    FALSE: (LBIT: ^ VBITS))
	      END;
  BEGIN
    LPTR.LADDR := -16607;
    IF LPTR.LBYTE^[0] <> CHR(4) THEN
      BEGIN
	WRITELN(OUTPUT); WRITELN(OUTPUT);
	WRITE(OUTPUT,'Version 1.3 of SYSTEM.COMPILER cannot run');
	WRITELN(OUTPUT);
	WRITE(OUTPUT,'with a non-1.3 version of SYSTEM.PASCAL');
	WRITELN(OUTPUT);
	EXIT(PASCALCOMPILER)
      END;
    LPTR.LADDR := -16606;
    HAS128K := LPTR.LBIT^[6]
  END (*CHECKVERS*) ;

BEGIN (*COMPINIT*)
  CHECKVERS;
  INITSCALARS; INITSETS;
  LEVEL := 0; TOP := 0;
  CONLIST := FALSE; LSTOPEN := FALSE;
{ Apple asks for the listing file before it reads a line of source. The
  prompt repeats until the file opens or the answer is empty. }
  IF NOISY THEN
    REPEAT
      LOK := TRUE;
      WRITELN(OUTPUT);
      WRITE(OUTPUT,'Listing file (<ret> for none or option in source): ');
      READ(INPUT,LTITLE); READLN(INPUT);
      IF LTITLE <> '' THEN
	IF LTITLE[1] = CHR(16) THEN DELETE(LTITLE,1,2);
      IF LTITLE <> '' THEN
	BEGIN
	  IF LTITLE[1] = CHR(27) THEN EXIT(PASCALCOMPILER);
	  OSPROC43(LTITLE,1,80);
	  LLENG := LENGTH(LTITLE);
	  IF LTITLE[LLENG] = ':' THEN
	    FOR LI := 1 TO LLENG DO
	      IF LTITLE[LI] IN ['a'..'z'] THEN
		LTITLE[LI] := CHR(ORD(LTITLE[LI]) - 32);
	  IF (LTITLE = 'CONSOLE:') OR (LTITLE = '#1:') THEN
	    BEGIN NOISY := FALSE; CONLIST := TRUE END;
	  REWRITE(LP,LTITLE);
	  LOK := IORESULT = 0;
	  IF LOK THEN BEGIN LSTOPEN := TRUE; LIST := TRUE END
	  ELSE
	    BEGIN LERR := IORESULT;
	      WRITE(OUTPUT,CHR(7));
	      WRITE(OUTPUT,'I/O Error #');
	      WRITE(OUTPUT,LERR);
	      WRITE(OUTPUT,' occured while opening ');
	      WRITE(OUTPUT,LTITLE);
	      WRITELN(OUTPUT)
	    END
	END
      ELSE LSTOPEN := FALSE
    UNTIL LOK;
  IF NOISY THEN
    BEGIN
      FOR IC := 1 TO 4 DO WRITELN(OUTPUT);
      WRITE(OUTPUT,'Apple Pascal Compiler [1.3]');
      WRITELN(OUTPUT);
      WRITE(OUTPUT,'<   0>')
    END;
  WITH DISPLAY[0] DO
    BEGIN FNAME := NIL; FFILE := NIL; FLABEL := NIL; OCCUR := BLCK END;
  SMALLESTSPACE := MEMAVAIL;
  GETNEXTPAGE;
  INSYMBOL;
  ENTSTDTYPES;   ENTSTDNAMES;   ENTUNDECL;
  ENTSPCPROCS;   ENTSTDPROCS;
  IF SYSCOMP THEN
    BEGIN OUTERBLOCK := NIL; SEG := 0; NEXTSEG := 1;
      GLEV := 1; BLOCKBEGSYS := BLOCKBEGSYS + [UNITSY]
    END
  ELSE
    BEGIN TOP := 1; LEVEL := 1;
      WITH DISPLAY[1] DO
	BEGIN FNAME := NIL; FFILE := NIL;
	  FLABEL := NIL; OCCUR := BLCK
	END;
      LC := LC+2; GLEV := 3; (*KEEP STACK STRAIGHT FOR NOW*)
      SEGTABLE[SEGSLOT].SEGNUM := SEG;
      NEW(OUTERBLOCK,PROC,DECLARED,ACTUAL,FALSE);
      WITH OUTERBLOCK^ DO
	BEGIN NEXT := NIL; LOCALLC := LC;
	  NAME := 'PROGRAM '; IDTYPE := NIL; KLASS := PROC;
	  PFDECKIND := DECLARED; PFLEV := 0; PFNAME := 1; PFSEG := SEG;
	  PFKIND := ACTUAL; FORWDECL := FALSE; EXTURNAL := FALSE;
	  INSCOPE := TRUE
	END
    END;
  SEGMAP[SEG] := SEGSLOT;
  IF SY = PROGSY THEN
    BEGIN INSYMBOL;
      IF SY = IDENT THEN
	BEGIN SEGTABLE[SEGSLOT].SEGNAME := ID;
	  IF OUTERBLOCK <> NIL THEN
	    BEGIN
	      OUTERBLOCK^.NAME := ID;
	      ENTERID(OUTERBLOCK) (*ALLOWS EXIT ON PROGRAM NAME*)
	    END
	END
      ELSE ERROR(2); INSYMBOL;
      IF SY = LPARENT THEN
	BEGIN
	  REPEAT INSYMBOL
	  UNTIL SY IN [RPARENT,SEMICOLON]+BLOCKBEGSYS;
	  IF SY = RPARENT THEN INSYMBOL ELSE ERROR(4)
	END;
      IF SY = SEMICOLON THEN INSYMBOL ELSE ERROR(14)
    END;
  MARK(MARKP);
  NEW(TOS);
  WITH TOS^ DO  (*MAKE LEXSTKREC FOR OUTERBLOCK*)
    BEGIN
      PREVLEXSTACKP := NIL;
      BFSY := PERIOD;
      DFPROCP := OUTERBLOCK;
      DLLC := LC;
      DOLDLEV := LEVEL;
      DOLDTOP := TOP;
      POLDPROC := CURPROC;
      ISSEGMENT := FALSE;
      DMARKP := MARKP
    END
END (*COMPINIT*);
