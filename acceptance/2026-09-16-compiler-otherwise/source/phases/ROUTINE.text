{ Segment 10. Seventeen procedures, and the largest of the phases that
  BODYPART carries: every standard procedure and function that is not
  READ or WRITE is compiled here.

  II.0 split this work in two. ROUTINE took the twenty-one keys that need
  real argument parsing, and the rest stayed inline in CALL as a CASE.
  Apple moved the whole CASE down here as SPECIALS -- procedure 17 -- and
  turned ROUTINE's body inside out to reach it:

      IF NOT (LKEY IN [12,...,43]) THEN SPECIALS ELSE CASE LKEY OF ...

  The LNOT before the FJP is what proves the NOT; the twenty-one keys are
  a three-word set constant, $F000 $88EC $0F7D, which is exactly II.0's
  membership test with the sense reversed.

  Two small helpers are Apple's and not II.0's. GETCOMMA and CHECKINT are
  procedures 2 and 3, and they replace the twenty-odd copies of
  IF SY = COMMA THEN INSYMBOL ELSE ERROR(20) and
  IF GATTR.TYPTR <> INTPTR THEN ERROR(125) that II.0 wrote out longhand.
  They have to be declared first: every CIP 2 and CIP 3 in the segment
  depends on it, and so does the number of every procedure after them.

  ROUTINE's own frame is 24 bytes -- twelve words -- of which six are the
  parameters, leaving three four-word sets. They are assigned in the first
  three statements of the body, and reverse allocation inside a list
  (finding 33) fixes which offset is which: the FSYS + [COMMA] set is at
  15, FSYS + [RPARENT] at 11, and their union at 7. The names are
  SPECULATION; nothing in the binary carries them. What is not speculation
  is that they exist and are computed once -- II.0 rebuilt
  FSYS + [COMMA,RPARENT] at every call site, and Apple did not.

  Where Apple changed II.0's behaviour, and not just its shape:

    - IDSEARCH and TREESEARCH compile to ERROR(124) instead of CSP 7 and
      CSP 8. They are native 6502 in 1.3 and are reached as ordinary
      external procedures, so the compiler refuses the intrinsic form.
      TREESEARCH also leaves GATTR.TYPTR NIL where II.0 set INTPTR.
    - UNITSTATUS, key 44, is Apple's and has no II.0 original.
    - NEWSTMT's two ERROR(116) become ERROR(125), and so does STR's.
    - SEEK with no second argument is ERROR(20), not ERROR(125).
    - ORD rejects a type whose SIZE is not 1 as well as one whose FORM is
      at or past POWER.
    - EXIT calls NEWPROC when PFNAME is still 0, and its LINKERREF is
      guarded by NOT INTRINSIC and points at IC-1, not IC-2; the second
      LINKERREF for a separate procedure is gone.
    - CONCAT and COPYDELETE both drop II.0's final LC := LLC, and CONCAT
      passes LEVEL to STR and LDA where II.0 passed 0. }

    VAR FSYSC,FSYSR,FSYSCR: SETOFSYS;

    PROCEDURE GETCOMMA;
      BEGIN
	IF SY = COMMA THEN INSYMBOL ELSE ERROR(20)
      END (*GETCOMMA*) ;

    PROCEDURE CHECKINT;
      BEGIN
	IF GATTR.TYPTR <> INTPTR THEN ERROR(125)
      END (*CHECKINT*) ;

    PROCEDURE STRGVAR(FSYS: SETOFSYS; MUSTBEVAR: BOOLEAN);
      BEGIN EXPRESSION(FSYS);
	WITH GATTR DO
	  IF ((KIND = CST) AND (TYPTR = CHARPTR))
	      OR STRGTYPE(TYPTR) THEN
	    IF KIND = VARBL THEN LOADADDRESS
	    ELSE
	      BEGIN
		IF MUSTBEVAR THEN ERROR(154);
		IF KIND = CST THEN
		  BEGIN
		    IF TYPTR = CHARPTR THEN
		      BEGIN
			WITH SCONST^ DO
			  BEGIN CCLASS := STRG; SLGTH := 1;
			    SVAL[1] := CHR(CVAL.IVAL)
			  END;
			CVAL.VALP := SCONST;
			NEW(TYPTR,ARRAYS,TRUE,TRUE);
			TYPTR^ := STRGPTR^;
			TYPTR^.MAXLENG := 1
		      END;
		    LOADADDRESS
		  END
	      END
	  ELSE
	    BEGIN
	      IF GATTR.TYPTR <> NIL THEN ERROR(125);
	      GATTR.TYPTR := STRGPTR
	    END
      END (*STRGVAR*) ;

    PROCEDURE NEWSTMT;
      LABEL 1;
      VAR LSP,LSP1: STP;
	  VARTS,LMIN,LMAX: INTEGER;
	  LSIZE,LSZ: ADDRRANGE;
	  LVAL: VALU;
      BEGIN VARIABLE(FSYSCR); LOADADDRESS;
	LSP := NIL; VARTS := 0; LSIZE := 0;
	IF GATTR.TYPTR <> NIL THEN
	  WITH GATTR.TYPTR^ DO
	    IF FORM = POINTER THEN
	      BEGIN
		IF ELTYPE <> NIL THEN
		  WITH ELTYPE^ DO
		    BEGIN LSIZE := SIZE;
		      IF FORM = RECORDS THEN LSP := RECVAR
		    END
	      END
	    ELSE ERROR(125);
	WHILE SY = COMMA DO
	  BEGIN INSYMBOL;
	    CONSTANT(FSYSCR,LSP1,LVAL);
	    VARTS := VARTS + 1;
	    IF LSP = NIL THEN ERROR(158)
	    ELSE
	      IF LSP^.FORM <> TAGFLD THEN ERROR(162)
	      ELSE
		IF LSP^.TAGFIELDP <> NIL THEN
		  IF STRGTYPE(LSP1) OR (LSP1 = REALPTR) THEN ERROR(159)
		  ELSE
		    IF COMPTYPES(LSP^.TAGFIELDP^.IDTYPE,LSP1) THEN
		      BEGIN
			LSP1 := LSP^.FSTVAR;
			WHILE LSP1 <> NIL DO
			  WITH LSP1^ DO
			    IF VARVAL.IVAL = LVAL.IVAL THEN
			      BEGIN LSIZE := SIZE; LSP := SUBVAR;
				GOTO 1
			      END
			    ELSE LSP1 := NXTVAR;
			LSIZE := LSP^.SIZE; LSP := NIL
		      END
		    ELSE ERROR(125);
      1:  END (*WHILE*) ;
	GENLDC(LSIZE);
	GEN1(30(*CSP*),1(*NEW*))
      END (*NEWSTMT*) ;

    PROCEDURE MOVE;
      BEGIN VARIABLE(FSYSC); BYTEADDRESS;
	GETCOMMA;
	IF LKEY = 27 THEN
	  BEGIN EXPRESSION(FSYSC); LOAD END
	ELSE
	  BEGIN VARIABLE(FSYSC); BYTEADDRESS END;
	GETCOMMA;
	EXPRESSION(FSYSR); LOAD;
	IF LKEY = 27 THEN GEN1(30(*CSP*),10(*FLC*))
	ELSE
	  IF LKEY = 21 THEN GEN1(30(*CSP*),2(*MVL*))
	  ELSE GEN1(30(*CSP*),3(*MVR*))
      END (*MOVE*) ;

    PROCEDURE EXIT;
      VAR LCP: CTP;
      BEGIN
	IF SY = IDENT THEN
	  BEGIN SEARCHID([PROC,FUNC],LCP); INSYMBOL END
	ELSE
	  IF SY = PROGSY THEN
	    BEGIN LCP := OUTERBLOCK; INSYMBOL END
	  ELSE LCP := NIL;
	IF LCP <> NIL THEN
	  WITH LCP^ DO
	    IF PFDECKIND = DECLARED THEN
	      BEGIN
		IF PFNAME = 0 THEN NEWPROC(LCP);
		GENLDC(PFSEG);
		IF INMODULE THEN
		  IF NOT INTRINSIC THEN
		    LINKERREF(PROC,PFSEG,IC-1);
		GENLDC(PFNAME)
	      END
	    ELSE ERROR(125)
	ELSE ERROR(125);
	GEN1(30(*CSP*),4(*XIT*))
      END (*EXIT*) ;

    PROCEDURE UNITIO;
      BEGIN
	CHECKINT;
	GETCOMMA;
	VARIABLE(FSYSC); BYTEADDRESS;
	GETCOMMA;
	EXPRESSION(FSYSCR); LOAD;
	CHECKINT;
	IF SY = COMMA THEN
	  BEGIN INSYMBOL;
	    IF SY = COMMA THEN GENLDC(0)
	    ELSE
	      BEGIN
		EXPRESSION(FSYSCR); LOAD;
		CHECKINT
	      END
	  END
	ELSE GENLDC(0);
	IF SY = COMMA THEN
	  BEGIN INSYMBOL;
	    EXPRESSION(FSYSR); LOAD;
	    CHECKINT
	  END
	ELSE GENLDC(0);
	IF LKEY = 13 THEN GEN1(30(*CSP*),5(*URD*))
	ELSE GEN1(30(*CSP*),6(*UWT*))
      END (*UNITIO*) ;

    PROCEDURE CONCAT;
      VAR LLC: ADDRRANGE; TEMPLGTH: INTEGER;
      BEGIN TEMPLGTH := 0;
	LLC := LC; LC := LC + (STRGLGTH DIV CHRSPERWD) + 1;
	GENLDC(0); GEN2(56(*STR*),LEVEL,LLC);
	GEN2(50(*LDA*),LEVEL,LLC);
	REPEAT
	  STRGVAR(FSYSCR,FALSE);
	  TEMPLGTH := TEMPLGTH + GATTR.TYPTR^.MAXLENG;
	  IF TEMPLGTH < STRGLGTH THEN GENLDC(TEMPLGTH)
	  ELSE GENLDC(STRGLGTH);
	  GEN2(77(*CXP*),0(*SYS*),23(*SCONCAT*));
	  GEN2(50(*LDA*),LEVEL,LLC);
	  TEST := SY <> COMMA;
	  IF NOT TEST THEN INSYMBOL
	UNTIL TEST;
	IF TEMPLGTH < STRGLGTH THEN
	  LC := LLC + (TEMPLGTH DIV CHRSPERWD) + 1
	ELSE TEMPLGTH := STRGLGTH;
	IF LC > LCMAX THEN LCMAX := LC;
	WITH GATTR DO
	  BEGIN NEW(TYPTR,ARRAYS,TRUE,TRUE);
	    TYPTR^ := STRGPTR^;
	    TYPTR^.MAXLENG := TEMPLGTH
	  END
      END (*CONCAT*) ;

    PROCEDURE COPYDELETE;
      VAR LLC: ADDRRANGE; LSP: STP;
      BEGIN
	IF LKEY = 19 THEN
	  BEGIN LLC := LC;
	    LC := LC + (STRGLGTH DIV CHRSPERWD) + 1
	  END;
	IF LKEY <> 43 THEN
	  BEGIN
	    STRGVAR(FSYSC,LKEY = 18);
	    IF LKEY = 19 THEN
	      BEGIN LSP := GATTR.TYPTR;
		GEN2(50(*LDA*),LEVEL,LLC)
	      END;
	    GETCOMMA
	  END;
	EXPRESSION(FSYSC); LOAD;
	IF GATTR.TYPTR <> NIL THEN CHECKINT;
	GETCOMMA;
	EXPRESSION(FSYSR); LOAD;
	IF GATTR.TYPTR <> NIL THEN CHECKINT;
	IF LKEY = 19 THEN
	  BEGIN
	    GEN2(77(*CXP*),0(*SYS*),25(*SCOPY*));
	    GEN2(50(*LDA*),LEVEL,LLC);
	    IF LSP^.MAXLENG < STRGLGTH THEN
	      LC := LLC + (LSP^.MAXLENG DIV CHRSPERWD) + 1;
	    IF LC > LCMAX THEN LCMAX := LC;
	    GATTR.TYPTR := LSP
	  END
	ELSE
	  IF LKEY = 43 THEN
	    GEN2(77(*CXP*),0(*SYS*),29(*GOTOXY*))
	  ELSE GEN2(77(*CXP*),0(*SYS*),26(*SDELETE*))
      END (*COPYDELETE*) ;

    PROCEDURE STR;
      BEGIN
	WITH GATTR DO
	  BEGIN
	    IF COMPTYPES(LONGINTPTR,TYPTR) THEN
	    ELSE
	      IF TYPTR = INTPTR THEN
		BEGIN
		  GENLDC(18(*DCVT*)); GENNR(30,4(*DECOPS*));
		  TYPTR := LONGINTPTR
		END
	      ELSE ERROR(125);
	    GETCOMMA;
	    STRGVAR(FSYSR,TRUE);
	    IF STRGTYPE(TYPTR) THEN
	      BEGIN GENLDC(TYPTR^.MAXLENG); GENLDC(12(*DSTR*));
		GENNR(30,4(*DECOPS*))
	      END
	    ELSE ERROR(125)
	  END
      END (*STR*) ;

    PROCEDURE CLOSE;
      BEGIN
	VARIABLE(FSYSCR); LOADADDRESS;
	IF GATTR.TYPTR <> NIL THEN
	  IF GATTR.TYPTR^.FORM <> FILES THEN ERROR(125);
	IF SY = COMMA THEN
	  BEGIN INSYMBOL;
	    IF SY = IDENT THEN
	      BEGIN
		IF ID = 'NORMAL  ' THEN GENLDC(0)
		ELSE
		  IF ID = 'LOCK    ' THEN GENLDC(1)
		  ELSE
		    IF ID = 'PURGE   ' THEN GENLDC(2)
		    ELSE
		      IF ID = 'CRUNCH  ' THEN GENLDC(3)
		      ELSE ERROR(2);
		INSYMBOL
	      END
	    ELSE ERROR(2)
	  END
	ELSE GENLDC(0);
	GEN2(77(*CXP*),0(*SYS*),6(*FCLOSE*));
	IF IOCHECK THEN GEN1(30(*CSP*),0(*IOC*))
      END (*CLOSE*) ;

    PROCEDURE GETPUTETC;
      BEGIN
	VARIABLE(FSYSCR); LOADADDRESS;
	IF GATTR.TYPTR <> NIL THEN
	  IF GATTR.TYPTR^.FORM <> FILES THEN ERROR(125)
	  ELSE
	    IF GATTR.TYPTR^.FILTYPE = NIL THEN ERROR(399);
	CASE LKEY OF
	  32: BEGIN
		IF SY = COMMA THEN
		  BEGIN
		    INSYMBOL; EXPRESSION(FSYSR); LOAD;
		    CHECKINT
		  END
		ELSE ERROR(20);
		GENNR(31,2(*SEEK*))
	      END;
	  34: GEN2(77(*CXP*),0(*SYS*),7(*FGET*));
	  35: GEN2(77(*CXP*),0(*SYS*),8(*FPUT*));
	  40: BEGIN
		IF GATTR.TYPTR <> NIL THEN
		  IF GATTR.TYPTR^.FILTYPE <> CHARPTR THEN ERROR(399);
		GENLDC(12); GENLDC(0);
		GEN2(77(*CXP*),0(*SYS*),17(*WRC*))
	      END
	END (*CASES*) ;
	IF IOCHECK THEN GEN1(30(*CSP*),0(*IOC*))
      END (*GETPUTETC*) ;

    PROCEDURE SCAN;
      BEGIN
	IF GATTR.TYPTR <> NIL THEN CHECKINT;
	GETCOMMA;
	IF SY = RELOP THEN
	  BEGIN
	    IF OP = EQOP THEN GENLDC(0)
	    ELSE
	      IF OP = NEOP THEN GENLDC(1)
	      ELSE ERROR(125);
	    INSYMBOL
	  END
	ELSE ERROR(125);
	EXPRESSION(FSYSC); LOAD;
	IF GATTR.TYPTR <> NIL THEN
	  IF GATTR.TYPTR <> CHARPTR THEN ERROR(125);
	GETCOMMA;
	VARIABLE(FSYSCR); BYTEADDRESS;
	IF SY = COMMA THEN
	  BEGIN INSYMBOL;
	    EXPRESSION(FSYSR); LOAD
	  END
	ELSE GENLDC(0);
	GEN1(30(*CSP*),11(*SCN*));
	GATTR.TYPTR := INTPTR
      END (*SCAN*) ;

    PROCEDURE BLOCKIO;
      BEGIN
	VARIABLE(FSYSC); LOADADDRESS;
	IF GATTR.TYPTR <> NIL THEN
	  IF GATTR.TYPTR^.FORM <> FILES THEN ERROR(125)
	  ELSE
	    IF GATTR.TYPTR^.FILTYPE <> NIL THEN ERROR(399);
	GETCOMMA;
	VARIABLE(FSYSC); BYTEADDRESS;
	GETCOMMA;
	EXPRESSION(FSYSCR); LOAD;
	CHECKINT;
	IF SY = COMMA THEN
	  BEGIN INSYMBOL;
	    EXPRESSION(FSYSR); LOAD;
	    CHECKINT
	  END
	ELSE GENLDC(-1);
	IF LKEY = 37 THEN GENLDC(1) ELSE GENLDC(0);
	GENLDC(0); GENLDC(0);
	GEN2(77(*CXP*),0(*SYS*),28(*BLOCKIO*));
	IF IOCHECK THEN GEN1(30(*CSP*),0(*IOC*));
	GATTR.TYPTR := INTPTR
      END (*BLOCKIO*) ;

    PROCEDURE SIZEOF;
      VAR LCP: CTP;
      BEGIN
	IF SY = IDENT THEN
	  BEGIN SEARCHID(VARS + [TYPES,FIELD],LCP); INSYMBOL;
	    IF LCP^.IDTYPE <> NIL THEN
	      GENLDC(LCP^.IDTYPE^.SIZE*CHRSPERWD)
	  END;
	GATTR.TYPTR := INTPTR
      END (*SIZEOF*) ;

    PROCEDURE SPECIALS;
      BEGIN
	CASE LKEY OF
	  5,6: BEGIN (*EOF & EOLN*)
		 IF FWASLPAR THEN
		   BEGIN VARIABLE(FSYSR); LOADADDRESS;
		     IF GATTR.TYPTR <> NIL THEN
		       IF GATTR.TYPTR^.FORM <> FILES THEN ERROR(125)
		       ELSE
			 IF (GATTR.TYPTR^.FILTYPE <> CHARPTR) AND
			     (LKEY = 6) THEN ERROR(399)
		   END
		 ELSE LOADIDADDR(INPUTPTR);
		 GENLDC(0); GENLDC(0);
		 IF LKEY = 5 THEN GEN2(77(*CXP*),0(*SYS*),10(*FEOF*))
		 ELSE GEN2(77(*CXP*),0(*SYS*),11(*FEOLN*));
		 GATTR.TYPTR := BOOLPTR
	       END (*EOF*) ;
	  7,8: BEGIN GENLDC(1); (*PREDSUCC*)
		 IF GATTR.TYPTR <> NIL THEN
		   IF GATTR.TYPTR^.FORM = SCALAR THEN
		     IF LKEY = 8 THEN GEN0(2(*ADI*))
		     ELSE GEN0(21(*SBI*))
		   ELSE ERROR(115)
	       END (*PREDSUCC*) ;
	    9: BEGIN (*ORD*)
		 IF GATTR.TYPTR <> NIL THEN
		   IF (GATTR.TYPTR^.FORM >= POWER)
		       OR (GATTR.TYPTR^.SIZE <> 1) THEN ERROR(125);
		 GATTR.TYPTR := INTPTR
	       END (*ORD*) ;
	   10: BEGIN (*SQR*)
		 IF GATTR.TYPTR <> NIL THEN
		   IF GATTR.TYPTR = INTPTR THEN GEN0(24(*SQI*))
		   ELSE
		     IF GATTR.TYPTR = REALPTR THEN GEN0(25(*SQR*))
		     ELSE BEGIN ERROR(125); GATTR.TYPTR := INTPTR END
	       END (*SQR*) ;
	   11: BEGIN (*ABS*)
		 IF GATTR.TYPTR <> NIL THEN
		   IF GATTR.TYPTR = INTPTR THEN GEN0(0(*ABI*))
		   ELSE
		     IF GATTR.TYPTR = REALPTR THEN GEN0(1(*ABR*))
		     ELSE BEGIN ERROR(125); GATTR.TYPTR := INTPTR END
	       END (*ABS*) ;
	   16: BEGIN (*LENGTH*)
		 STRGVAR(FSYSR,FALSE);
		 GENLDC(0(*INDEX*)); GEN0(62(*LDB*));
		 GATTR.TYPTR := INTPTR
	       END (*LENGTH*) ;
	   17: BEGIN (*INSERT*)
		 STRGVAR(FSYSC,FALSE);
		 GETCOMMA;
		 STRGVAR(FSYSC,TRUE);
		 GENLDC(GATTR.TYPTR^.MAXLENG);
		 GETCOMMA;
		 EXPRESSION(FSYSR); LOAD;
		 IF GATTR.TYPTR <> NIL THEN CHECKINT;
		 GEN2(77(*CXP*),0(*SYS*),24(*SINSERT*))
	       END (*INSERT*) ;
	   20: BEGIN (*POS*)
		 STRGVAR(FSYSC,FALSE);
		 GETCOMMA;
		 STRGVAR(FSYSR,FALSE);
		 GENLDC(0); GENLDC(0);
		 GEN2(77(*CXP*),0(*SYS*),27(*SPOS*));
		 GATTR.TYPTR := INTPTR
	       END (*POS*) ;
	   24: BEGIN (*IDSEARCH*)
		 VARIABLE(FSYSC); LOADADDRESS;
		 GETCOMMA;
		 VARIABLE(FSYSR); LOADADDRESS;
		 ERROR(124)
	       END (*IDSEARCH*) ;
	   25: BEGIN (*TREESEARCH*)
		 GETCOMMA;
		 VARIABLE(FSYSC); LOADADDRESS;
		 GETCOMMA;
		 VARIABLE(FSYSR); LOADADDRESS;
		 GATTR.TYPTR := NIL;
		 ERROR(124)
	       END (*TREESEARCH*) ;
	   26: BEGIN (*TIME*)
		 VARIABLE(FSYSC); LOADADDRESS;
		 IF GATTR.TYPTR <> NIL THEN CHECKINT;
		 GETCOMMA;
		 VARIABLE(FSYSR); LOADADDRESS;
		 IF GATTR.TYPTR <> NIL THEN CHECKINT;
		 GEN1(30(*CSP*),9(*TIM*))
	       END (*TIME*) ;
   28,29,30,33: BEGIN (*OPEN,RESET,REWRITE*)
		 VARIABLE(FSYSCR); LOADADDRESS;
		 IF GATTR.TYPTR <> NIL THEN
		   IF GATTR.TYPTR^.FORM <> FILES THEN ERROR(125);
		 IF SY <> COMMA THEN
		   IF LKEY = 33 THEN
		     GEN2(77(*CXP*),0(*SYS*),4(*FRESET*))
		   ELSE ERROR(20)
		 ELSE
		   BEGIN INSYMBOL;
		     STRGVAR(FSYSR,FALSE);
		     IF (LKEY = 28) OR (LKEY = 30) THEN GENLDC(0)
		     ELSE GENLDC(1);
		     GENLDC(0); GEN2(77(*CXP*),0(*SYS*),5(*FOPEN*))
		   END;
		 IF IOCHECK THEN GEN1(30(*CSP*),0(*IOC*))
	       END (*OPEN*) ;
	   39: BEGIN (*TRUNC*)
		 IF GATTR.TYPTR = INTPTR THEN
		   BEGIN GEN0(10(*FLT*)); GATTR.TYPTR := REALPTR END;
		 IF GATTR.TYPTR <> NIL THEN
		   IF GATTR.TYPTR = REALPTR THEN
		     GEN1(30(*CSP*),23(*TNC*))
		   ELSE
		     IF GATTR.TYPTR^.FORM = LONGINT THEN
		       BEGIN
			 GENLDC(20(*DTNC*)); GENNR(30,4(*DECOPS*))
		       END
		     ELSE ERROR(125);
		 GATTR.TYPTR := INTPTR
	       END (*TRUNC*) ;
	   44: BEGIN (*UNITSTATUS*)
		 EXPRESSION(FSYSC); LOAD;
		 IF GATTR.TYPTR <> NIL THEN CHECKINT;
		 GETCOMMA;
		 VARIABLE(FSYSC); BYTEADDRESS;
		 GETCOMMA;
		 EXPRESSION(FSYSCR); LOAD;
		 GEN1(30(*CSP*),12(*UST*))
	       END (*UNITSTATUS*)
	END (*CASES*)
      END (*SPECIALS*) ;

  BEGIN (*ROUTINE*)
    FSYSR := FSYS + [RPARENT];
    FSYSC := FSYS + [COMMA];
    FSYSCR := FSYSR + FSYSC;
    IF NOT (LKEY IN [12,13,14,15,18,19,21,22,23,27,31,32,34,35,36,37,38,
		     40,41,42,43]) THEN SPECIALS
    ELSE
      CASE LKEY OF
	12:       NEWSTMT;
	13,14:    UNITIO;
	15:       CONCAT;
	18,19,43: COPYDELETE;
	21,22,27: MOVE;
	23:       EXIT;
	31:       CLOSE;
	32,34,
	35,40:    GETPUTETC;
	36:       SCAN;
	37,38:    BLOCKIO;
	41:       SIZEOF;
	42:       STR
      END (*CASES*)
  END;
