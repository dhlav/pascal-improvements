{ Segment 20. In UCSD II.0 this is not a procedure at all: it is the tail
  of PASCALCOMPILER's own body, from `IF SY <> PERIOD` to the last
  WRITECODE (block.text:120-163). Apple lifted it into a phase so the code
  is not resident while the compile runs. }
BEGIN
  IF SY <> PERIOD THEN ERROR(21);
  IF LIST THEN
    BEGIN SCREENDOTS := SCREENDOTS+1;
      SYMBUFP^[SYMCURSOR] := CHR(EOL);
      SYMCURSOR := SYMCURSOR+1;
      PRINTLINE
    END;
  USERINFO.ERRBLK := 0;
  TIME(LGTH,STARTDOTS); LOWTIME := STARTDOTS-LOWTIME;
  IF LINKINFO THEN
    BEGIN SEGTABLE[SEGMAP[SEG]].SEGKIND := 1;
      WRITELINKERINFO
    END;
  RELEASE(MARKP);
  CLOSE(LIBRARY);
  CLOSE(LP,LOCK);
  IF NOISY THEN WRITELN(OUTPUT);
  WRITE(OUTPUT,SCREENDOTS,' lines');
  IF LOWTIME > 0 THEN
    WRITE(OUTPUT,', ',(LOWTIME+30) DIV 60,' secs, ',
        ROUND((3600/LOWTIME)*SCREENDOTS),' lines/min');
  IF NOISY THEN
    BEGIN
      WRITELN(OUTPUT);
      WRITE(OUTPUT,'Smallest available space = ',SMALLESTSPACE,' words');
    END;
  IC := 0;
  NEW(CODEP);
  FOR SEG := 0 TO MAXSEG DO
    WITH SEGTABLE[SEG] DO
      BEGIN GENWORD(DISKADDR); GENWORD(CODELENG) END;
  FOR SEG := 0 TO MAXSEG DO
    BEGIN MOVELEFT(SEGTABLE[SEG].SEGNAME,CODEP^[IC],8);
      IC := IC+8
    END;
  FOR SEG := 0 TO MAXSEG DO GENWORD(SEGTABLE[SEG].SEGKIND);
  FOR SEG := 0 TO MAXSEG DO GENWORD(SEGTABLE[SEG].TEXTADDR);
  FOR SEG := 0 TO MAXSEG DO GENWORD(SEGINFO(SEG));
  IF ISPROG THEN
    BEGIN MOVELEFT(SEGSUSED,PROCTABLE,8);
      GENWORD(PROCTABLE[0]); GENWORD(PROCTABLE[1]);
      GENWORD(PROCTABLE[2]); GENWORD(PROCTABLE[3])
    END
  ELSE
    BEGIN GENWORD(0); GENWORD(0); GENWORD(0); GENWORD(0) END;
  TEST := TRUE;
  IF IC >= 432 THEN ERROR(407);
  WHILE IC < 432 DO GENBYTE(0);
  IF COMMENT = NIL THEN
    BEGIN NEW(COMMENT); COMMENT^[0] := CHR(0) END;
  MOVELEFT(COMMENT^[1],CODEP^[IC],LENGTH(COMMENT^));
  FILLCHAR(CODEP^[IC+LENGTH(COMMENT^)],80-LENGTH(COMMENT^),CHR(0));
  IC := IC+80;
  CURBLK := 0; CURBYTE := 0; WRITECODE(TRUE)
END;
