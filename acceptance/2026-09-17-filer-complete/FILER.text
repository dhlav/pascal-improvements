{ FILER -- Apple Pascal 1.3 SYSTEM.FILER, reconstructed.

  One segment, FILEHAND, and 56 procedures. The shape is the
  linker's: a (*$U-*) compilation whose first SEGMENT PROCEDURE
  is segment 1, finished by the Librarian, which leaves slot 0
  of the shipped file blank (finding 267).

  The ancestor is UCSD's II.0 Filer (Roger T. Sumner and Steven
  S. Thomson, 1979), evidence/reference/ucsd-ii0-filer. It was
  written as a segment of the operating system, so the program
  shell here carries the 1.3 operating system's own declarations
  from src/pascal/os/1.3/PASCALSYSTEM.text, comments dropped. Its
  procedure tree and most of its bodies survive into Apple's 1.3
  binary, and its names are used wherever the binary confirms the
  role. What Apple added carries a placeholder: FL<n> for
  procedure n, G<n> for global word n, L<n> for local word n.

  Apple merged CHECKRSLT into MESSAGES, renumbered the Filer's
  error constants from 1013..1034 to 1..22 in the same order,
  rewrote LISTVOLS, BADBLOCKS and KRUNCH, and reworded most
  messages. The bodies here are read from the binary.

  Range and I/O checking are off: (*$U-*), and the binary has no
  CHK and no CSP 0. }

  { II.0's MESSAGES and CHECKRSLT in one: RSLT is an I/O result,
    ignored when it is 0, 13 or 14 and there is no Filer error,
    and NUMBER is a Filer error. }
  PROCEDURE MESSAGES(RSLT, NUMBER : INTEGER; EXXIT : BOOLEAN);
  VAR
    STR : STRING[40];
  BEGIN
    IF (RSLT IN [0,13,14]) AND (NUMBER = 0) THEN
      EXIT(MESSAGES);
    STR := '';

    CASE RSLT OF
      1 : STR := 'Parity (CRC) error';
      2 : STR := 'Illegal volume number';
      3 : STR := 'Bad I/O operation';
      4 : STR := 'Timeout error';
      5 : STR := 'Volume went off-line';
      6 : STR := 'File lost in directory';
      7 : STR := 'Bad file name';
      8 : STR := 'No room on volume';
      9 : STR := 'No such volume on-line';
      10: STR := 'File not found';
    END;

    CASE NUMBER OF
      VOLONLINE: STR := 'Volume already on-line';
      TEXTLOST: STR := 'Text file lost';
      CODELOST: STR := 'Code file lost';
      FOUNDFILE: STR := 'File found';
      BLKDEXP,BLKD: STR := 'Blocked volume';
      UNBLKDEXP,UNBLKD: STR := 'Unblocked volume';
      FILEEXP: STR := 'File name';
      FILEUNBLKDEXP: STR := 'File/(unblocked volume)';
      FILEBLKDEXP: STR := 'File/(blocked volume)';
      FILEVOLEXP: STR := 'File/volume';
      VOLEXP: STR := 'Volume name';
      FILEFULL: STR := 'Output file full';
      WRKSAVED: STR := 'Workfile is saved';
      NODIR: STR := 'No directory on volume';
      NOWRK: STR := 'No workfile to save';
      NOWILD: STR := 'Wildcard not allowed';
      BADFORM: STR := 'Bad form (Wild <to> Non-Wild) card';
      ILLFILEVOL: STR := 'Illegal file/volume name';
      ILLCHANGE: STR := 'Illegal change (Volume <to> file) name';
      BADDEST: STR := 'Bad destination for files found'
    END;
    IF NUMBER >= BLKDEXP THEN
      STR := CONCAT(STR,' expected');
    CLEARLINE;
    IF (RSLT > 10) OR (IORESULT <> 0) AND SYSCOM^.MISCINFO.SLOWTERM THEN
      WRITE('I/O error #',RSLT)
    ELSE
      IF SYSCOM^.MISCINFO.SLOWTERM THEN
        WRITE('Filer error #',NUMBER)
      ELSE
        WRITE(STR);
    IF EXXIT THEN
      EXIT(CALLPROC)
  END;

  PROCEDURE WRITEANDCLEAR;

  END;

  FUNCTION NGETCHAR(FLUSH : BOOLEAN) : BOOLEAN;
  BEGIN
    CH := GETCHAR(FLUSH);
    IF CH = CHR(27) THEN
      EXIT(CALLPROC);
    NGETCHAR := CH = 'Y';
    IF NOT EOLN THEN
      WRITELN
  END;

  PROCEDURE NSPACEWAIT(FLUSH : BOOLEAN);
  BEGIN
    IF SPACEWAIT(FLUSH) THEN
      EXIT(CALLPROC)
  END;

  PROCEDURE UPDATEDIR;
  BEGIN
    WRITEDIR(SOURCEUNIT,GDIR);
    MESSAGES(IORESULT,0,TRUE)
  END;

  PROCEDURE EATSPACES(VAR STRG : LONGSTRING);
  VAR
    I : INTEGER;

  BEGIN
    I := 1;
    WHILE I <= LENGTH(STRG) DO
      IF (ORD(STRG[I]) >= 33) AND (ORD(STRG[1]) <= 125) THEN
        BEGIN
          IF (STRG[I] >= 'a') AND (STRG[I] <= 'z') THEN
            STRG[I] := CHR( ORD( STRG[I] ) - ORD( 'a' ) + ORD ('A' ));
          I := I + 1
        END
      ELSE
        DELETE(STRG,I,1);
    IF STRG = '' THEN
      EXIT(CALLPROC);
  END;

  PROCEDURE GETBLOCKS(MESS1,MESS2,MESS3 : SHORTSTRING; LASTBLK: INTEGER;
                                        VAR NBLOCKS: INTEGER);
  VAR
    OK : BOOLEAN;
  BEGIN
    OK := FALSE;
    IF GDIR <> NIL THEN
      IF GDIR^[0].DEOVBLK >= LASTBLK THEN
        BEGIN
          CLEARLINE;
          WRITE(MESS1,' ',GDIR^[0].DEOVBLK,' ',MESS2,' ? (Y/N) ');
          OK := NGETCHAR(TRUE);
          NBLOCKS := GDIR^[0].DEOVBLK
        END;
    IF NOT OK THEN
      BEGIN
        CLEARLINE;
        WRITE(MESS3,' ? ');
        READLN(NBLOCKS);
        IF NBLOCKS < LASTBLK THEN
          BEGIN
            CLEARLINE;
            WRITE('Invalid #');
            EXIT(CALLPROC)
          END;
      END
  END;

  PROCEDURE RISKVOLUME;
  BEGIN
    IF (LASTSTATE = OKDIR) AND (GDIR <> NIL) THEN
      BEGIN
        CLEARLINE;
        WRITE('Remove all files on ',GVID,': ? (Y/N)');
        IF NOT NGETCHAR(TRUE) THEN
          EXIT(CALLPROC)
      END
  END;

  PROCEDURE INSERTVOLUME(INTUNIT : INTEGER; VID1 : VID; CHECK : BOOLEAN);
  VAR
    OK : BOOLEAN;
    OLDUNIT, NEWUNIT : VID;

  BEGIN { INSERTVOLUME }
    OLDUNIT := '#  ';
    IF (INTUNIT DIV 10) = 1 THEN
      OLDUNIT[2] := '1';
    OLDUNIT [3] := CHR(ORD('0') + INTUNIT MOD 10);
    EATSPACES(OLDUNIT);
    OK := CHECK;
    IF CHECK THEN
      IF VOLSEARCH(VID1,TRUE,GDIR) <> INTUNIT THEN
        BEGIN
          NEWUNIT := OLDUNIT;
          OK := VOLSEARCH(NEWUNIT,TRUE,GDIR) <> 0;
          OK := OK AND (NEWUNIT = VID1)
        END;
    IF NOT OK THEN
      BEGIN
        CLEARLINE;
        WRITELN('Put ',VID1,': in unit ',OLDUNIT);
        NSPACEWAIT(TRUE);
        IF CHECK THEN
          BEGIN
            OK := (VOLSEARCH(OLDUNIT,TRUE,GDIR) <> 0);
            IF (NOT OK) OR (OLDUNIT <> VID1) THEN
              EXIT(CALLPROC)
          END
      END
  END { INSERTVOLUME };

  PROCEDURE ZAPENTRIES(DIRMAP : BITMAP; UPDATE : BOOLEAN);
  VAR
    LOC : INTEGER;
  BEGIN
    IF DIRMAP.ENTRIES > 0 THEN
      BEGIN
        INSERTVOLUME(SOURCEUNIT,SOURCEVID,TRUE);
        IF GDIR <> NIL THEN
          BEGIN
            FOR LOC := GDIR^ [0].DNUMFILES DOWNTO 1 DO
              IF DIRMAP.DIRENTRY [LOC] THEN
                DELENTRY(LOC,GDIR);
            IF UPDATE THEN
              UPDATEDIR
          END
      END;
  END;

  FUNCTION PURGEIT(NAME,MESS : SHORTSTRING):BOOLEAN;
  VAR
    GFIB : FILE;
  BEGIN
    RESET(GFIB,NAME);
    PURGEIT := IORESULT = 0;
    IF IORESULT = 0 THEN
      BEGIN
        IF MESS <> '' THEN
          BEGIN
            CLEARLINE;
            WRITE(MESS,' ',NAME,' ? ');
            IF NOT NGETCHAR(TRUE) THEN
              BEGIN
                PURGEIT := FALSE;
                EXIT(PURGEIT)
              END
          END;
        CLOSE(GFIB,PURGE);
        MESSAGES(IORESULT,0,TRUE)
      END;
  END;

  PROCEDURE PRINTMESS(VID1 : VID; TID1 : TID; DEST : SHORTSTRING);
  VAR
    SCRWIDTH : INTEGER;   { screen width, taken as 40 below 80 }
  BEGIN
    CLEARLINE;
    WRITE(VID1,':',TID1);
    SCRWIDTH := SYSCOM^.CRTINFO.WIDTH;
    IF SCRWIDTH < 80 THEN
      SCRWIDTH := 40;
    IF (LENGTH(DEST) + 31) > SCRWIDTH THEN
      WRITEANDCLEAR
    ELSE
      IF NOT SYSCOM^.MISCINFO.SLOWTERM THEN
        WRITE('':24-(LENGTH(VID1)+LENGTH(TID1)));
    WRITELN(' --> ',DEST)
  END;

    PROCEDURE MAKECALL(RSLT, ERR : INTEGER; STATE : CHECKS);
    VAR
      PRINTERROR : BOOLEAN;

    BEGIN
      LASTSTATE := STATE;
      IF (STATE IN CHECK) OR (ERROR = 0) THEN
        EXIT(SCANINPUT);
      IF GTITLE = '' THEN
        EXIT(CALLPROC);
      HOMECURSOR;
      WRITEANDCLEAR;

      IF GTITLE [1] = '*' THEN
        BEGIN
          DELETE(GTITLE,1,1);
          IF GTITLE [1] = ':' THEN
            DELETE(GTITLE,1,1);
          WRITE(SYVID,':')
        END
      ELSE
        IF GTITLE [1] = ':' THEN
          WRITE(DKVID);

      WRITE(GTITLE);
      IF SYSCOM^.CRTINFO.WIDTH >= 80 THEN
        WRITE(' - ')
      ELSE
        WRITEANDCLEAR;
      MESSAGES(RSLT,ERR,FALSE);

      CASE STATE OF
        BADTITLE : PRINTERROR := TRUE;
        NOVOL,BADUNIT,BADDIR : PRINTERROR := (ERROR=FILEEXP) AND (GTID='');
        UNBLKDVOL : PRINTERROR := ERROR IN [BLKDEXP,FILEEXP,FILEBLKDEXP];
        OKFILE,BADFILE : PRINTERROR := ERROR IN [BLKDEXP,UNBLKDEXP,VOLEXP];
        OKDIR : PRINTERROR := ERROR IN [UNBLKDEXP,FILEEXP,FILEUNBLKDEXP]
      END;
      IF NOT SYSCOM^.MISCINFO.SLOWTERM THEN
        BEGIN
          IF PRINTERROR THEN
            BEGIN
              IF SYSCOM^.CRTINFO.WIDTH < 80 THEN
                WRITEANDCLEAR
              ELSE
                WRITE(', ');
              MESSAGES(0,ERROR,FALSE)
            END;

          IF WHERE = SOURCE THEN
            WRITE(' <source>');
          IF WHERE = DESTINATION THEN
            WRITE(' <destination>');
        END;
      EXIT(CALLPROC)
    END;

  BEGIN { SCANINPUT }
    GUNIT := 0;
    IF SCANTITLE(GTITLE,GVID,GTID,GSEGS,GKIND) THEN
      BEGIN
        IF GETDIR THEN
          MARK(NEWDIR);
        GVID2 := GVID;
        GUNIT := VOLSEARCH(GVID,TRUE,GDIR);
        IF GDIR = NIL THEN
          BEGIN
            IF GUNIT = 0 THEN
              MAKECALL(9,0,NOVOL);
            UNITCLEAR(GUNIT);
            IF IORESULT <> 0 THEN
              MAKECALL(2,0,BADUNIT);
            IF UNITABLE [GUNIT].UISBLKD THEN
              MAKECALL(0,NODIR,BADDIR);
            MAKECALL(0,UNBLKD,UNBLKDVOL)
          END;
        IF GTID = '' THEN
          MAKECALL(0,BLKD,OKDIR);
        IF DIRSEARCH(GTID,TRUE,GDIR) <> 0 THEN
          MAKECALL(0,FOUNDFILE,OKFILE);
        MAKECALL(10,0,BADFILE)
      END;
    MAKECALL(0,ILLFILEVOL,BADTITLE)
  END { SCANINPUT };

  FUNCTION SEARCHDIR(MESSAGE : STRNG; VAR GINX : INTEGER;
                        DEST, SCREENCLEAR, SWAPPING : BOOLEAN) : BOOLEAN;
  VAR
    X : INTEGER;
    NEWSTRING : TID;

      PROCEDURE FOUNDFILE;
      BEGIN
        WITH GDIR^ [GINX] DO
          BEGIN
            SOURCETITLE := DTID;
            FROMWHERE := CONCAT(VOLNAME1,':',DTID);
            CH := 'Y';
            FOUND := FILEFOUND;
            IF (MESSAGE <> '') AND QUESTION THEN
              BEGIN
                CLEARLINE;
                IF NOT SYSCOM^.MISCINFO.SLOWTERM THEN
                  WRITE(MESSAGE,' ');
                WRITE(DTID,' ? ');
                CH := GETCHAR(FALSE);
                IF NOT EOLN THEN
                  WRITELN;
                IF CH = CHR(27) THEN
                  BEGIN
                    FOUND := ABORTIT;
                    EXIT(SEARCHDIR)
                  END
              END
          END;
        SEARCHDIR := CH = 'Y'
      END;

  FUNCTION TESTSTR(STR : TID; START : INTEGER) : BOOLEAN;
  VAR
    TEMP : TID;
  BEGIN
    TEMP [0] := STR [0];
    MOVELEFT(GDIR^ [X].DTID[START],TEMP[1],LENGTH(STR));
    TESTSTR := TEMP = STR
  END;

  BEGIN { SEARCHDIR }
    SEARCHDIR := FALSE;
    IF GINX = 0 THEN
      BEGIN
        DEST := DEST AND UNITABLE [DESTUNIT].UISBLKD;
        IF SCREENCLEAR AND WILDCARD THEN
          BEGIN
            CLEARSCREEN;
            WRITELN
          END
        ELSE
          CLEARLINE;
        FOUND := NOFILES;
        MARKING := DEST AND (SOURCEVID = DESTVID);
        FOR X := 1 TO GDIR^ [0].DNUMFILES DO
        WITH GDIR^ [X] DO
          BEGIN
            STATUS := FALSE;
            IF (LENGTH(STRING1) + LENGTH(STRING2)) <= LENGTH(DTID) THEN
              IF TESTSTR(STRING1,1) AND
                TESTSTR(STRING2,LENGTH(DTID) - LENGTH(STRING2) + 1) THEN
                  WITH DIRMAP DO
                  BEGIN
                    IF (STRING1=DTID) OR WILDCARD THEN
                      BEGIN
                        STATUS := MARKING;
                        DIRENTRY [X] := TRUE;
                        ENTRIES := ENTRIES + 1;
                      END;
                    FOUND := FILESNOGOOD
                  END;
          END;
        IF MARKING THEN
          UPDATEDIR
      END
    ELSE
      IF DEST THEN
        GINX := GINX - 1;
    IF DIRMAP.ENTRIES > 0 THEN
    BEGIN
      IF SWAPPING THEN
        INSERTVOLUME(SOURCEUNIT,SOURCEVID,TRUE);
      CH := 'N';
      WHILE (GINX < GDIR^[0].DNUMFILES) AND (CH <> 'Y') DO
        WITH GDIR^ [GINX+1] DO
          BEGIN
            GINX := GINX + 1;
            IF MARKING THEN
              BEGIN
                DIRMAP.DIRENTRY [GINX] := STATUS;
                IF STATUS THEN
                  BEGIN
                    STATUS := FALSE;
                    UPDATEDIR
                  END
              END;
            IF DIRMAP.DIRENTRY [GINX] THEN
              BEGIN
                IF DEST THEN
                  BEGIN
                    NEWSTRING := COPY(DTID,LENGTH(STRING1) + 1,
                            LENGTH(DTID) - LENGTH(STRING1) - LENGTH(STRING2));
                    X := LENGTH(NEWSTRING) +
                      SCAN(LENGTH(STRING3),= '[',STRING3[1]) + LENGTH(STRING4);
                    IF (X <= TIDLENG) AND ((X > 0) OR NOT WILDCARD) THEN
                      BEGIN
                        TOWHERE := CONCAT(VOLNAME2,':',STRING3,
                                                   NEWSTRING,STRING4);
                        IF (STRING1 = DTID) OR WILDCARD THEN
                          FOUNDFILE
                      END
                    ELSE
                      PRINTMESS(SOURCEVID,DTID,'not processed');
                  END
                ELSE
                  FOUNDFILE;
                DIRMAP.DIRENTRY [GINX] := FALSE;
                DIRMAP.ENTRIES := DIRMAP.ENTRIES - 1;
              END;
          END;
    END;
    IF FOUND = NOFILES THEN
      MESSAGES(ORD(INOFILE),0,FALSE);
    IF FOUND = FILESNOGOOD THEN
      MESSAGES(0,BADDEST,FALSE);
  END { SEARCHDIR };

  PROCEDURE CHECKFILE(MSG1,MSG2 : SHORTSTRING; DEFAULT,ERROR1 : INTEGER;
                     WILD,FILLE : BOOLEAN; CHECK1 : CHCKS);
  VAR
    SRCSTR : STRING;

    PROCEDURE PROCESSDATA(MSG:SHORTSTRING; FIRSTCALL:BOOLEAN; VAR VOLNAME:VID;
                       VAR FIRSTSTR,SECONDSTR:SHORTSTRING; VAR WHERETO:STRNG);
    VAR
      STR,OLD : STRING;
      WHERE : LOCATION;
      X,I,LOC : INTEGER;

      PROCEDURE FINDDELIM(SIZE,MESSAGE : INTEGER; VAR STRIING : STRNG);
      VAR
        ERROR : BOOLEAN;

        FUNCTION SCAN2(CH : CHAR) : BOOLEAN;
        BEGIN
          SCAN2 := SCAN(LOC,= CH,STR[1]) >= LOC;
        END;

      BEGIN { FINDDELIM }
        ERROR := LOC > SIZE;
        IF (NOT ERROR) AND SCAN2('$') AND SCAN2('=') AND SCAN2('?') THEN
          BEGIN
            STRIING := COPY(STR,1,LOC);
            DELETE(STR,1,LOC);
            DELETE(STR,1,1);
            EXIT(FINDDELIM);
          END;
        CLEARLINE;
        IF NOT SYSCOM^.MISCINFO.SLOWTERM THEN
          WRITE(COPY(STR,1,LOC));
        IF SYSCOM^.CRTINFO.WIDTH < 80 THEN
          WRITEANDCLEAR;
        IF ERROR THEN
          WRITE('...too long <');
        CASE MESSAGE OF
          1 : WRITE(' file name ');
          2 : WRITE(' scan string ');
          3 : WRITE(' volume name ');
        END;
        IF ERROR THEN
          BEGIN
            IF MESSAGE = 3 THEN
              WRITE(VIDLENG)
            ELSE
              WRITE(TIDLENG);
            WRITE(' - character maximum >');
          END
        ELSE
          WRITE('- Illegal format');
        EXIT(CALLPROC)
      END { FINDDELIM };

      FUNCTION SCAN1(CH : CHAR) : INTEGER;
      BEGIN
        SCAN1 := SCAN(LENGTH(STR), = CH,STR[1]);
      END;

    BEGIN
      IF INSTRING = '' THEN
        BEGIN
          CLEARLINE;
          WRITE(MSG);
          IF FIRSTCALL AND FAST THEN
            IF FILLE THEN
              WRITE(' what file')
            ELSE
              WRITE(' what volume');
          WRITE(' ? ');
          READLN(INSTRING);
          EATSPACES(INSTRING)
        END;

      LOC := SCAN(LENGTH(INSTRING), = ',',INSTRING[1]);
      IF LOC > 35 THEN
        EXIT(CALLPROC);
      STR := COPY(INSTRING,1,LOC);
      DELETE(INSTRING,1,LOC);
      DELETE(INSTRING,1,1);

      QUESTION := QUESTION OR (SCAN1('?') < LENGTH(STR));
      LOC := SCAN1(':');
      IF (STR [1] = '#') OR (LOC < LENGTH(STR)) THEN
        FINDDELIM(VIDLENG,3,VOLNAME)
      ELSE
        IF STR[1] = '*' THEN
          BEGIN
            DELETE(STR,1,1);
            VOLNAME := '*'
          END
        ELSE
          IF STR[1] = '%' THEN
            BEGIN
              DELETE(STR,1,1);
              VOLNAME := RUN_VOL
            END;
      WHERETO := CONCAT(VOLNAME,':');

      I := SCAN1('$');
      IF I < LENGTH(STR) THEN
        BEGIN
          IF LENGTH(STR)+LENGTH(SRCSTR)-1 > 35 THEN
            EXIT(CALLPROC);
          OLD := STR;
          STR[0] := CHR(LENGTH(SRCSTR)+LENGTH(STR)-1);
          MOVELEFT(SRCSTR[1],STR[I+1],LENGTH(SRCSTR));
          MOVELEFT(OLD[I+2],STR[I+1+LENGTH(SRCSTR)],LENGTH(OLD)-I-1);
        END;

      SRCSTR := STR;
      LOC := SCAN1('=');
      IF LOC = LENGTH(STR) THEN
        LOC := SCAN1('?');
      IF LOC < LENGTH(STR) THEN
        IF WILD THEN
          BEGIN
            IF NOT (FIRSTCALL OR WILDCARD) THEN
              MESSAGES(0,BADFORM,TRUE);
            WILDCARD := TRUE;
            FINDDELIM(TIDLENG,2,FIRSTSTR);
            LOC := LENGTH(STR);
            FINDDELIM(TIDLENG,2,SECONDSTR)
          END
        ELSE
          MESSAGES(0,NOWILD,TRUE)
      ELSE
        BEGIN
          IF (NOT FIRSTCALL) AND WILDCARD AND (DEFAULT = 0) THEN
            BEGIN
              SCANINPUT(CONCAT(VOLNAME2,':'),[],0,NEITHER,FALSE);
              IF LASTSTATE <> UNBLKDVOL THEN
                MESSAGES(0,BADFORM,TRUE);
            END;
          FINDDELIM(SHSTRLENG,1,FIRSTSTR);
          WHERETO := CONCAT(WHERETO,FIRSTSTR)
        END;
      IF NOT FIRSTCALL THEN
        BEGIN
          SCANINPUT(WHERETO,[],0,WHERE,TRUE);
          IF GUNIT = 0 THEN
            DESTUNIT := 4
          ELSE
            DESTUNIT := GUNIT;
          DESTVID := GVID;
          IF GDIR <> NIL THEN
            DESTBLKS := GDIR^[0].DEOVBLK
          ELSE
            DESTBLKS := 0;
          INSERTVOLUME(SOURCEUNIT,SOURCEVID,TRUE);
        END
      ELSE
        IF (INSTRING <> '') AND (DEFAULT <= 0) THEN
          WHERE := DESTINATION
        ELSE
          WHERE := NEITHER;

      IF WILDCARD THEN
        SCANINPUT(FROMWHERE,[OKDIR],BLKDEXP,SOURCE,TRUE)
      ELSE
        SCANINPUT(FROMWHERE,CHECK1,ERROR1,SOURCE,TRUE);
    END;

  BEGIN { CHECKFILE }
    INITGLOBALS;
    SRCSTR := '';
    PROCESSDATA(MSG1,TRUE,VOLNAME1,STRING1,STRING2,FROMWHERE);
    SOURCEUNIT := GUNIT;
    SOURCEVID := GVID;
    IF GDIR <> NIL THEN
      SRCBLKS := GDIR^[0].DEOVBLK
    ELSE
      SRCBLKS := 0;
    IF DEFAULT > 0 THEN
      EXIT(CHECKFILE);
    IF (INSTRING = '') AND (DEFAULT < 0) THEN
      EXIT(CHECKFILE);
    PROCESSDATA(MSG2,FALSE,VOLNAME2,STRING3,STRING4,TOWHERE);
  END;

  FUNCTION FINDSAME(DOO : BOOLEAN):BOOLEAN;
  BEGIN
    FINDSAME := TRUE;
    IF (LASTSTATE = OKFILE) AND (DOO OR (SOURCETITLE <> GTID) OR
                                            (SOURCEVID <> GVID)) THEN
      FINDSAME := PURGEIT(CONCAT(GVID,':',GTID),'Remove old')
  END;

  PROCEDURE CHANGER;
  VAR
    GFIB : UNTYPED;
    GFIBP : FIBP;
    IORSLT,LOC : INTEGER;

  BEGIN { CHANGER }
    REPEAT
      CHECKFILE('Change','Change to what',0,FILEBLKDEXP,TRUE,TRUE,
                                                      [OKFILE,OKDIR]);
      IF ((STRING1 <> '') AND (STRING3 <> '')) OR WILDCARD THEN
        BEGIN
          VOLNAME2 := VOLNAME1;
          TOWHERE := CONCAT(VOLNAME1,':');
          IF NOT WILDCARD THEN
            TOWHERE := CONCAT(TOWHERE,STRING3);
          LOC := 0;
          WHILE SEARCHDIR('Change',LOC,TRUE,TRUE,FALSE) DO
            BEGIN
              RESET(GFIB,FROMWHERE);
              MESSAGES(IORESULT,0,TRUE);
              GFIBP := GETPTR(GFIB);
              SCANINPUT(TOWHERE,[BADFILE,OKFILE],FILEEXP,DESTINATION,TRUE);
              IF FINDSAME(FALSE) THEN
                WITH GFIBP^ DO
                  BEGIN
                    FHEADER.DACCESS.YEAR := 100;
                    PRINTMESS(FVID,FHEADER.DTID,GTID);
                    FHEADER.DTID := GTID;
                  END;
              CLOSE(GFIB);
              MESSAGES(IORESULT,0,TRUE)
            END
        END
      ELSE
        IF LENGTH(STRING1) + LENGTH(STRING3) = 0 THEN
          BEGIN
            SCANINPUT(TOWHERE,[NOVOL,OKDIR],BLKDEXP,DESTINATION,TRUE);
            IF LASTSTATE = OKDIR THEN
              MESSAGES(0,VOLONLINE,TRUE);
            NEW(GDIR);
            UNITREAD(SOURCEUNIT,GDIR^,SIZEOF(DIRECTORY),DIRBLK);
            MESSAGES(IORESULT,0,TRUE);
            GDIR^[0].DVID := GVID;
            UNITWRITE(SOURCEUNIT,GDIR^,SIZEOF(DIRECTORY),DIRBLK);
            IORSLT := IORESULT;
            RELEASE(GDIR);
            MESSAGES(IORSLT,0,TRUE);
            UNITABLE[SOURCEUNIT].UVID := GVID;
            IF (SYVID = SOURCEVID) AND (SYSCOM^.SYSUNIT = SOURCEUNIT) THEN
              SYVID := GVID;
            IF DKVID = SOURCEVID THEN
              DKVID := GVID;
            PRINTMESS(SOURCEVID,'',CONCAT(GVID,':'))
          END
        ELSE
          MESSAGES(0,ILLCHANGE,TRUE)
    UNTIL INSTRING = ''
  END {CHANGER} ;

  PROCEDURE REMOVER;
  VAR
    DELETIONS : BITMAP;
    LINE,LOC : INTEGER;

  BEGIN { REMOVER }
    REPEAT
      CHECKFILE('Remove','',1,FILEEXP,TRUE,TRUE,[OKFILE]);
      LINE := 0;
      LOC := 0;
      FILLCHAR(DELETIONS,SIZEOF(DELETIONS),CHR(0));
      WHILE SEARCHDIR('Remove',LOC,FALSE,TRUE,FALSE) DO
        BEGIN
          IF NOT QUESTION THEN
            BEGIN
              PRINTMESS(GVID,GDIR^[LOC].DTID,'removed');
              IF SYSCOM^.CRTINFO.HEIGHT = LINE THEN
                BEGIN
                  NSPACEWAIT(FALSE);
                  CLEARSCREEN;
                  LINE := 0
                END;
              LINE := LINE+1;
            END;
          DELETIONS.ENTRIES := DELETIONS.ENTRIES + 1;
          DELETIONS.DIRENTRY [LOC] := TRUE
        END;
      IF (FOUND IN [FILEFOUND,ABORTIT]) AND (DELETIONS.ENTRIES > 0) THEN
        BEGIN
          CLEARLINE;
          WRITE('Update directory ? ');
          IF NGETCHAR(TRUE) THEN
            ZAPENTRIES(DELETIONS,TRUE);
        END;
    UNTIL INSTRING = ''
  END {REMOVER} ;

  PROCEDURE TRANSFER;
  VAR
    LASTBLK,LOC : INTEGER;
    L3 : INTEGER;   { never read }

    PROCEDURE MOVEFILE;
    VAR
      RELBLK,NUMBLKS,NBLOCKS : INTEGER;
      FIRSTCALL,SINGLEDRIVE : BOOLEAN;
      GFIBP : FIBP;
      GFIB : UNTYPED;

    BEGIN
      RESET(GFIB,FROMWHERE);
      MESSAGES(IORESULT,0,TRUE);
      GFIBP := GETPTR(GFIB);
      RELBLK := 0;
      FIRSTCALL := TRUE;
      REPEAT
        NUMBLKS := LASTBLK - RELBLK;
        IF NUMBLKS > GBUFBLKS THEN
          NUMBLKS := GBUFBLKS;
        NBLOCKS := BLOCKREAD(GFIB,GBUF^,NUMBLKS);
        MESSAGES(IORESULT,0,TRUE);
        RELBLK := RELBLK + NUMBLKS;
        IF FIRSTCALL THEN
          BEGIN
            FIRSTCALL := FALSE;
            SCANINPUT(TOWHERE,[NOVOL,BADDIR,BADFILE,UNBLKDVOL,OKDIR,OKFILE],
                                           FILEVOLEXP,DESTINATION,TRUE);
            IF ((GVID2 <> '') AND (GVID2 [1] = '#') AND (GUNIT = SOURCEUNIT)
                      AND UNITABLE [GUNIT].UISBLKD) OR (GUNIT = 0) THEN
              BEGIN
                CLEARSCREEN;
                IF GUNIT = 0 THEN
                  WRITELN('Put in ',GVID,':')
                ELSE
                  WRITELN('Insert destination disk');
                NSPACEWAIT(TRUE);
                SCANINPUT(TOWHERE,[BADFILE,OKFILE,BADDIR,OKDIR,UNBLKDVOL],
                                      FILEVOLEXP,DESTINATION,TRUE);
              END;
            IF GUNIT IN [1,2] THEN
              BEGIN
                CLEARSCREEN;
                WRITELN
              END;
            IF NOT FINDSAME(FALSE) THEN
              EXIT(MOVEFILE);
            IF NBLOCKS > DIRBLK THEN
              RISKVOLUME;
            REWRITE(LFIB,TOWHERE);
            MESSAGES(IORESULT,0,TRUE);
            LFIBP := GETPTR(LFIB);
            IF NOT LFIBP^.FISBLKD AND GFIBP^.FISBLKD AND
                                   (GFIBP^.FHEADER.DFKIND = TEXTFILE) THEN
              BEGIN
                NBLOCKS := NBLOCKS-2;
                MOVELEFT(GBUF^[FBLKSIZE+FBLKSIZE],GBUF^,NBLOCKS*FBLKSIZE)
              END;
            SINGLEDRIVE := (LFIBP^.FVID <> GFIBP^.FVID) AND
                                           (LFIBP^.FUNIT = GFIBP^.FUNIT)
          END
        ELSE
          IF SINGLEDRIVE THEN
            INSERTVOLUME(LFIBP^.FUNIT,LFIBP^.FVID,FALSE);
        NUMBLKS := BLOCKWRITE(LFIB,GBUF^,NBLOCKS);
        MESSAGES(IORESULT,0,TRUE);
        IF NBLOCKS <> NUMBLKS THEN
          MESSAGES(0,FILEFULL,TRUE);
        IF SINGLEDRIVE AND NOT EOF(GFIB) THEN
           INSERTVOLUME(LFIBP^.FUNIT,GFIBP^.FVID,FALSE)
      UNTIL (RELBLK = LASTBLK) OR EOF(GFIB);
      WITH LFIBP^,GFIBP^.FHEADER DO
        BEGIN
          FHEADER.DLASTBYTE := DLASTBYTE;
          FHEADER.DFKIND := DFKIND;
          FHEADER.DACCESS := DACCESS;
          IF (DACCESS.MONTH = 0) AND (THEDATE.MONTH > 0) THEN
            FHEADER.DACCESS := THEDATE
        END;
      CLOSE(LFIB,LOCK);
      MESSAGES(IORESULT,0,TRUE);
      PRINTMESS(GFIBP^.FVID,GFIBP^.FHEADER.DTID,
                      CONCAT(LFIBP^.FVID,':',LFIBP^.FHEADER.DTID));
      CLOSE(GFIB);
    END;

  BEGIN { TRANSFER }
    REPEAT
      CHECKFILE('Transfer','To where',0,FILEVOLEXP,TRUE,TRUE,
                               [BADDIR,OKFILE,OKDIR,UNBLKDVOL]);
      LASTBLK := MAXINT;
      LOC := 0;
      IF (STRING1 = '') AND NOT WILDCARD THEN
        BEGIN
          IF (STRING3 = '') AND (SOURCEUNIT IN [4,5,9..12]) AND
             (DESTUNIT IN [4,5,9..12]) AND (SRCBLKS <> DESTBLKS) AND
             (SRCBLKS <> 0) AND (DESTBLKS <> 0) THEN
            BEGIN
              CLEARLINE;
              WRITELN('Number of blocks on the destination volume is not ',
                      'the same as the source volume');
              WRITE('Do volume transfer anyway? (Y/N) ');
              IF NOT NGETCHAR(TRUE) THEN
                EXIT(CALLPROC)
            END;
          IF LASTSTATE IN [OKDIR,BADDIR] THEN
            GETBLOCKS('Transfer','blocks','# of blocks to transfer',1,LASTBLK);
          MOVEFILE
        END
      ELSE
        WHILE SEARCHDIR('Transfer',LOC,TRUE,TRUE,TRUE) DO
          MOVEFILE
    UNTIL INSTRING = '';
  END { TRANSFER };

  PROCEDURE MAKEFILE;
  VAR
  GFIB : UNTYPED;
  GFIBP : FIBP;

  BEGIN { MAKEFILE }
    REPEAT
      CHECKFILE('Make','',1,FILEEXP,FALSE,TRUE,[BADFILE,OKFILE]);
      IF FINDSAME(TRUE) THEN
        BEGIN
          REWRITE(GFIB,FROMWHERE);
          MESSAGES(IORESULT,0,TRUE);
          GFIBP := GETPTR(GFIB);
          WITH GFIBP^ DO
            FMAXBLK := FHEADER.DLASTBLK-FHEADER.DFIRSTBLK;
          CLOSE(GFIB,LOCK);
          MESSAGES(IORESULT,0,TRUE);
          CLEARLINE;
          WRITELN(GVID,':',GTID,' made')
        END
    UNTIL INSTRING = ''
  END {MAKEFILE} ;

  FUNCTION SAVEWORK : BOOLEAN;
  TYPE
    FILESTRG = STRING [4];
  VAR
    GS : SHORTSTRING;
    OK : BOOLEAN;

    FUNCTION SAVEIT(WHATFILE : FILESTRG; WHICH : FILEKIND;
        VAR TITLE : TID; VAR SAVED,GOTIT  : BOOLEAN; MSG : INTEGER) : BOOLEAN;
    VAR
      GFIB : UNTYPED;
      GFIBP : FIBP;

    BEGIN { SAVEIT }
      SAVED := TRUE;
      WITH USERINFO DO
        BEGIN
          RESET(GFIB,CONCAT('*SYSTEM.WRK.',WHATFILE));
          GFIBP := GETPTR(GFIB);
          WITH USERINFO, GFIBP^.FHEADER DO
          IF GFIBP^.FISOPEN THEN
            BEGIN
              DACCESS.YEAR := 100;
              TITLE := CONCAT(WORKTID,'.',WHATFILE);
              DTID := TITLE;
              SAVEIT := TRUE;
              CLOSE(GFIB,NORMAL)
            END
          ELSE
            BEGIN
              SAVEIT := FALSE;
              GOTIT := FALSE;
              MESSAGES(0,MSG,FALSE)
            END
        END
    END { SAVEIT };

    PROCEDURE CONCATIT(STR : FILESTRG; SAVED : BOOLEAN);
    BEGIN
      IF NOT SAVED THEN
        INSTRING := CONCAT(INSTRING,'*SYSTEM.WRK.',STR,',',
                                         GVID2,':',GTID,'.',STR,',')
    END;

  BEGIN { SAVEWORK }
    WITH USERINFO DO
      BEGIN
        SAVEWORK := FALSE;
        GTID := WORKTID;
        GVID := WORKVID;
        GVID2 := WORKVID;
        IF TEXTSAVED AND CODESAVED THEN
          BEGIN
            WRITELN;
            IF GOTSYM OR GOTCODE THEN
              MESSAGES(0,WRKSAVED,TRUE);
            MESSAGES(0,NOWRK,TRUE);
          END;
        OK := FALSE;
        IF WORKTID <> '' THEN
          BEGIN
            CLEARLINE;
            WRITE('Save as ',WORKVID,':',WORKTID,' ? ');
            OK := NGETCHAR(TRUE)
          END;
        IF NOT OK THEN
          CHECKFILE('Save as','',1,FILEEXP,FALSE,TRUE,
                                   [NOVOL,BADDIR,BADFILE,OKDIR,OKFILE]);
        IF LENGTH(GTID) > TIDLENG-5 THEN
          MESSAGES(0,ILLFILEVOL,TRUE);
        IF GVID2 <> SYVID THEN
          BEGIN
            INSTRING := '';
            CONCATIT('TEXT',TEXTSAVED);
            CONCATIT('CODE',CODESAVED);
            DELETE(INSTRING,LENGTH(INSTRING),1);
            SAVEWORK := TRUE;
            EXIT(SAVEWORK)
          END;
        WORKTID := GTID;
        WORKVID := GVID;
        CLEARLINE;
        IF NOT TEXTSAVED THEN
          BEGIN
            IF SAVEIT('TEXT',TEXTFILE,SYMTID,TEXTSAVED,GOTSYM,TEXTLOST) THEN
              BEGIN
                IF CODESAVED THEN
                  IF PURGEIT(CONCAT('*',WORKTID,'.CODE'),'') THEN
                    WRITE('Old code removed, ');
                WRITE('Text file saved ')
              END;
            IF NOT CODESAVED THEN
              WRITE('& ')
          END;
        IF NOT CODESAVED THEN
          IF SAVEIT('CODE',CODEFILE,CODETID,CODESAVED,GOTCODE,CODELOST) THEN
            WRITE('Code file saved')
      END
  END {SAVEWORK} ;

  PROCEDURE WHATWORK;
  BEGIN
    WRITEANDCLEAR;
    WITH USERINFO DO
      IF GOTSYM OR GOTCODE THEN
        BEGIN
          IF WORKTID = '' THEN
            WRITE('not named')
          ELSE
            WRITE('Workfile is ',WORKVID,':',WORKTID);
          IF NOT (TEXTSAVED AND CODESAVED) THEN
            WRITE(' (not saved)')
        END
      ELSE
        WRITE('No workfile')
  END {WHATWORK} ;

  PROCEDURE CLEARWORK;
  BEGIN
    WITH USERINFO DO
      BEGIN
        GOTSYM := FALSE;
        GOTCODE := FALSE;
        WORKTID := '';
        SYMTID := '';
        CODETID := ''
      END
  END;

  PROCEDURE NEWWORK(GIVEBLURB: BOOLEAN);
  VAR
    GFIB : FILE;
  BEGIN { NEWWORK }
    WITH USERINFO DO
      BEGIN
        IF NOT (TEXTSAVED AND CODESAVED) THEN
          BEGIN
            CLEARLINE;
            WRITE('Throw away current workfile ? ');
            IF NOT NGETCHAR(TRUE) THEN
              EXIT(CALLPROC);
          END
        ELSE
          IF GIVEBLURB THEN
            WRITELN;

        IF PURGEIT('*SYSTEM.WRK.TEXT','') THEN;
        IF PURGEIT('*SYSTEM.WRK.CODE','') THEN;
        IF PURGEIT('*SYSTEM.LST.TEXT','') THEN;

        IF PURGEIT(CONCAT(WORKTID,'.BACK'),'Remove') THEN;

        TEXTSAVED := TRUE;
        CODESAVED := TRUE;
        IF GIVEBLURB THEN
          WITH USERINFO DO
            BEGIN
              CLEARLINE;
              WRITE('Workfile cleared');
              CLEARWORK;
            END
      END
  END { NEWWORK };

  PROCEDURE GETWORK;
  TYPE
    SHORT = STRING[5];
  VAR
    DONE,OK : BOOLEAN;
    X : INTEGER;

    FUNCTION CHECKIT(SUFFIX,MESS:SHORT; VAR TITLE:TID; VAR VOLID:VID) : BOOLEAN;
    BEGIN
      WITH USERINFO DO
      BEGIN
        CHECKIT := FALSE;
        SCANINPUT(CONCAT(WORKVID,':',WORKTID,SUFFIX),[OKFILE],0,NEITHER,TRUE);
        IF LASTSTATE = OKFILE THEN
          BEGIN
            CHECKIT := TRUE;
            DONE := TRUE;
            TITLE := CONCAT(WORKTID,SUFFIX);
            VOLID := WORKVID;
            IF GOTSYM THEN
              WRITE('& ');
            WRITE(MESS)
          END
      END
    END;

  BEGIN { GETWORK }
    NEWWORK(FALSE);
    CHECKFILE('Get','',1,FILEEXP,FALSE,TRUE,[BADFILE,OKFILE]);
    WITH USERINFO DO
    BEGIN
      CLEARWORK;
      WORKVID := GVID;
      WORKTID := GTID;
      X := LENGTH(WORKTID);
      OK := X <= TIDLENG-5;
      CLEARLINE;
      REPEAT
        DONE := NOT OK;
        IF DONE AND (X > 5) THEN
          IF (COPY(WORKTID,X-4,5)='.TEXT') OR (COPY(WORKTID,X-4,5)='.CODE') THEN
            DELETE(WORKTID,X - 4,5);
        IF (LENGTH(WORKTID) <= TIDLENG-5) AND (WORKTID <> '') THEN
          BEGIN
            GOTSYM := CHECKIT('.TEXT','Text ',SYMTID,SYMVID);
            GOTCODE := CHECKIT('.CODE','Code ',CODETID,CODEVID)
          END;
        OK := FALSE
      UNTIL DONE;
      IF NOT (GOTSYM OR GOTCODE) THEN
        BEGIN
          CLEARWORK;
          WRITE('No ')
        END;
      WRITE('file loaded')
    END
  END {GETWORK} ;

  PROCEDURE DATESET;
  CONST
    DASH = '-';
  VAR
    NUM : INTEGER;

    PROCEDURE ZAPIT;
    BEGIN
      DELETE(INSTRING,1,SCAN(LENGTH(INSTRING),=DASH,INSTRING[1]));
      DELETE(INSTRING,1,1)
    END;

    FUNCTION GETNUMBER(MAX : INTEGER) : BOOLEAN;
    VAR
      X,STOP : INTEGER;
    BEGIN
      NUM := 0;
      STOP := SCAN(LENGTH(INSTRING),=DASH,INSTRING[1]);
      FOR X := 1 TO STOP DO
        IF INSTRING[X] IN DIGITS THEN
          NUM := NUM*10+ORD(INSTRING[X])-ORD('0');
      GETNUMBER := (NUM > 0) AND (NUM <= MAX)
    END;

  BEGIN {DATESET}
    WITH THEDATE DO
      BEGIN
        IF MONTH <> 0 THEN
          WRITELN('Today is ',DAY,DASH,COPY(MONTHSTR,MONTH*3+1,3),DASH,YEAR);
        WRITELN('Date format: dd-mmm-yy (month, year optional)');
        WRITE('New date : ');
        READLN(INSTRING);
        EATSPACES(INSTRING);
        IF GETNUMBER(31) THEN
          DAY := NUM;
        ZAPIT;
        IF INSTRING[0] > CHR(2) THEN
          BEGIN
            GVID [0] := CHR(3);
            MOVELEFT(INSTRING[1],GVID[1],3);
            FOR NUM := 2 TO 3 DO
              IF (GVID [NUM] >= 'A') AND (GVID [NUM] <= 'Z') THEN
                GVID [NUM] := CHR( ORD(GVID [NUM]) - ORD('A') + ORD('a'));
            FOR NUM := 1 TO 12 DO
             IF COPY(MONTHSTR,NUM*3+1,3) = GVID THEN
               MONTH := NUM
          END;
        ZAPIT;
        IF GETNUMBER(99) THEN
          YEAR := NUM;
        SCANINPUT(CONCAT(SYVID,':'),[OKDIR],0,NEITHER,TRUE);
        IF (LASTSTATE = OKDIR) AND (GUNIT = SYSCOM^.SYSUNIT) THEN
          BEGIN
            GDIR^[0].DLASTBOOT := THEDATE;
            WRITEDIR(GUNIT,GDIR);
          END;
        WRITE('The date is ',DAY,DASH,COPY(MONTHSTR,MONTH*3+1,3),DASH,YEAR)
    END
  END { DATESET } ;

  PROCEDURE LISTDIR(DETAIL: BOOLEAN);
  VAR
    NOFILES,ALTFILE : BOOLEAN;
    L4 : ^INTEGER;
    L5 : DIRECTORY;   { the source directory, across the REWRITE }
    OUT : TEXT;
    LISTED,LOC,LINE,LARGEST,FREEBLKS,USEDAREA,USEDBLKS: INTEGER;

    PROCEDURE WRITELINE;
    BEGIN
      IF (LINE = SYSCOM^.CRTINFO.HEIGHT) OR (LINE = 0) THEN
        BEGIN
          HOMECURSOR;
          CLEARLINE;
          IF NOT ((LINE = 0) OR QUESTION) THEN
            NSPACEWAIT(FALSE);
          IF (NOT UNITABLE[DESTUNIT].UISBLKD) OR QUESTION THEN
            CLEARSCREEN;
          WRITELN(OUT);
          WRITE(OUT,SOURCEVID,':');
          LINE := 2;
          IF ALTFILE THEN
            LINE := SYSCOM^.CRTINFO.HEIGHT;
        END;
      WRITELN(OUT);
      LINE := LINE+1
    END {WRITELINE} ;

    PROCEDURE FREECHECK(FIRSTOPEN,NEXTUSED: INTEGER);
    VAR
      FREEAREA: INTEGER;

    BEGIN
      FREEAREA := NEXTUSED-FIRSTOPEN;
      IF FREEAREA > LARGEST THEN
        LARGEST := FREEAREA;
      FREEBLKS := FREEBLKS+FREEAREA;
      IF (FREEAREA > 0) AND DETAIL THEN
        BEGIN
          WRITE(OUT,'< UNUSED >',FREEAREA:12,'':12);
          IF FAST THEN
            WRITE(OUT,FIRSTOPEN:6)
          ELSE
            WRITE(OUT,FIRSTOPEN:5);
          WRITELINE
        END
    END {FREECHECK} ;

  BEGIN {LISTDIR}
    CHECKFILE('Directory listing of','',-1,FILEBLKDEXP,TRUE,FALSE,
                                                        [OKDIR,OKFILE]);
    ALTFILE := TOWHERE <> '';
    IF ALTFILE THEN
      L5 := GDIR^
    ELSE
      TOWHERE := '#1:';
    REWRITE(OUT,TOWHERE);
    MESSAGES(IORESULT,0,TRUE);
    IF ALTFILE THEN
      GDIR^ := L5;
    LISTED := 0;
    LOC := 0;
    LINE := 0;
    LARGEST := 0;
    FREEBLKS := 0;
    USEDAREA := 0;
    USEDBLKS := GDIR^[0].DLASTBLK;
    NOFILES := FALSE;
    IF STRING1 = '' THEN
      BEGIN
        NOFILES := NOT WILDCARD AND (GDIR^ [0].DNUMFILES = 0);
        WILDCARD := TRUE;
      END;
    IF WILDCARD THEN
      WRITELINE
    ELSE
      LINE := SYSCOM^.CRTINFO.HEIGHT+1;
    IF NOT NOFILES THEN
      WHILE SEARCHDIR('List',LOC,FALSE,FALSE,FALSE) DO
        BEGIN
        IF UNITABLE[DESTUNIT].UISBLKD AND (NOT QUESTION) THEN
          IF LISTED = 0 THEN
            BEGIN
              WRITEANDCLEAR;
              WRITE('Writing')
            END
          ELSE
            WRITE('.');
        LISTED := LISTED + 1;
        WITH GDIR^[LOC] DO
          BEGIN
            FREECHECK(GDIR^[LOC-1].DLASTBLK,DFIRSTBLK);
            USEDAREA := DLASTBLK-DFIRSTBLK;
            USEDBLKS := USEDBLKS+USEDAREA;
            WRITE(OUT,DTID,'':TIDLENG-LENGTH(DTID)+1,USEDAREA:6,DACCESS.DAY:5
                 ,'-',COPY(MONTHSTR,DACCESS.MONTH*3+1,3),'-',DACCESS.YEAR:2);
            IF DETAIL THEN
              IF FAST THEN
                WRITE(OUT,DFIRSTBLK:6,DLASTBYTE:6,'':3,
                              COPY(TYPESTR,ORD(DFKIND)*4+1,4),'file')
              ELSE
                WRITE(OUT,DFIRSTBLK:5,' ',COPY(TYPESTR,ORD(DFKIND)*4+1,4));
            WRITELINE
          END
      END;
    IF (FOUND IN [FILEFOUND,ABORTIT]) OR NOFILES THEN
      BEGIN
        IF WILDCARD THEN
        BEGIN
          FREECHECK(GDIR^[LOC].DLASTBLK,GDIR^[0].DEOVBLK);
          WRITE(OUT,LISTED,'/',GDIR^[0].DNUMFILES,' files');
          IF FAST THEN
            WRITE(OUT,' <listed/in dir>, ',USEDBLKS,' blocks used');
          WRITE(OUT,', ',FREEBLKS,' unused',', ',LARGEST,' in largest');
        END;
        MESSAGES(IORESULT,0,TRUE);
        IF ALTFILE THEN
          BEGIN
            MARK(L4);
            WRITELN(OUT);
            CLOSE(OUT,LOCK)
          END;
        MESSAGES(IORESULT,0,TRUE);
      END
  END {LISTDIR} ;

  PROCEDURE LISTVOLS;
  VAR
    L1 : INTEGER;
    L2 : ARRAY [0..3] OF INTEGER;   { UNITSTATUS result }
    L6 : INTEGER;
    L7 : RECORD
           CASE BOOLEAN OF
             TRUE: (I: INTEGER);
             FALSE: (P: ^INTEGER)
         END;
  BEGIN
    WRITELN;
    WRITELN;
    WRITELN('Volume # - Volume Name - # Blocks');
    WRITELN;
    FOR GUNIT := 1 TO MAXUNIT DO
      UNITCLEAR(GUNIT);
    GUNIT := VOLSEARCH(GVID,TRUE,GDIR);
    FOR GUNIT := 1 TO MAXUNIT DO
     WITH UNITABLE[GUNIT] DO
      IF UVID <> '' THEN
        BEGIN
          WRITE(GUNIT:5,'':8,UVID,':');
          IF UISBLKD THEN
            BEGIN
              L1 := 12 - LENGTH(UVID);
              WRITE('':L1,UEOVBLK:5)
            END;
          WRITELN
        END
      ELSE
        IF GUNIT IN [4,5,9..20] THEN
          BEGIN
            UNITSTATUS(GUNIT,L2,1);
            IF IORESULT = 0 THEN
              WRITELN(GUNIT:5,'':8,'<no dir>','':5,L2[0]:5)
          END;
    FOR L6 := 0 TO 15 DO
      BEGIN
        L7.I := -383 + 3*L6;
        IF L7.P^ <> 0 THEN
          WRITELN(L6+128:5,'':8,'<driver>')
      END;
    WRITELN;
    WRITELN('System volume is - ',SYVID,':');
    WRITELN('Prefix volume is - ',DKVID,':')
  END;

  PROCEDURE ZEROVOLUME;
  VAR
    LDE: DIRENTRY;
    VOLCONF: BOOLEAN;

  BEGIN { ZEROVOLUME }
    VOLCONF := FALSE;
    FILLCHAR(LDE,SIZEOF(LDE),CHR(0));
    LDE.DLASTBLK := DIRLASTBLK;
    CHECKFILE('Zero directory of','',1,BLKDEXP,FALSE,FALSE,[OKDIR,BADDIR]);
    RISKVOLUME;
    GETBLOCKS('Are there','blocks on the disk','# of blocks on the disk',
                                       LDE.DLASTBLK,LDE.DEOVBLK);
    REPEAT
      WRITE('Enter new volume name (<ret> for no change) : ');
      READLN(INSTRING);
      IF INSTRING <> '' THEN
        BEGIN
          EATSPACES(INSTRING);
          IF INSTRING [LENGTH(INSTRING)] <> ':' THEN
            INSTRING := CONCAT(INSTRING,':');
          SCANINPUT(INSTRING,[NOVOL,OKDIR],BLKDEXP,NEITHER,TRUE);
          WRITE(GVID,': correct ? ');
          VOLCONF := NGETCHAR(TRUE)
        END
      ELSE
        VOLCONF := TRUE
    UNTIL VOLCONF;
    WITH LDE DO
    BEGIN
      DFKIND := UNTYPEDFILE;
      DVID := GVID;
      DLASTBOOT := THEDATE;
      INSERTVOLUME(SOURCEUNIT,SOURCEVID,TRUE);
      UNITWRITE(SOURCEUNIT,LDE,SIZEOF(LDE),DIRBLK);
      MESSAGES(IORESULT,0,TRUE);
      WRITE(DVID,': zeroed')
    END
  END {ZEROVOLUME} ;

  PROCEDURE PRINTFILES;
  VAR
    I : INTEGER;
  BEGIN
    IF DIRMAP.ENTRIES > 0 THEN
      BEGIN
        WRITELN('File(s) endangered:');
        FOR I := 0 TO MAXDIR DO
          IF DIRMAP.DIRENTRY [I] THEN
            WITH GDIR^ [I] DO
              BEGIN
                IF I = 0 THEN
                  WRITE('Directory','':7)
                ELSE
                  WRITE(DTID,'':16-LENGTH(DTID));
                WRITELN(DFIRSTBLK:6,DLASTBLK:6)
              END
      END
  END;

  PROCEDURE WHICHFILE(VAR BADBLK:INTEGER; MARK : BOOLEAN);
  VAR
    X : INTEGER;
  BEGIN
    IF GDIR <> NIL THEN
      BEGIN
        FOR X := 0 TO GDIR^ [0].DNUMFILES DO
          IF GDIR^ [X].DLASTBLK > BADBLK THEN
            BEGIN
              IF NOT MARK THEN
                BADBLK := X
              ELSE
                IF GDIR^ [X].DFIRSTBLK <= BADBLK THEN
                  BEGIN
                    DIRMAP.ENTRIES := DIRMAP.ENTRIES + 1;
                    DIRMAP.DIRENTRY [X] := TRUE
                  END;
              EXIT(WHICHFILE)
            END;
        IF NOT MARK THEN
          BADBLK := GDIR^ [0].DNUMFILES + 1
      END
  END;

  { Reads a buffer's worth at a time, and block by block only
    where a whole read fails. }
  PROCEDURE BADBLOCKS;
  VAR
    BLK,TOTAL, NBLOCKS : INTEGER;
    L4 : INTEGER;   { blocks in this read }
    L5 : INTEGER;   { first block of this read }

  BEGIN { BADBLOCKS }
    CHECKFILE('Bad block scan of','',1,BLKDEXP,FALSE,FALSE,[OKDIR,BADDIR]);
    GETBLOCKS('Scan','blocks','Scan how many blocks',1,NBLOCKS);
    TOTAL := 0;
    L5 := 0;
    REPEAT
      L4 := NBLOCKS - L5;
      IF L4 > GBUFBLKS THEN
        L4 := GBUFBLKS;
      UNITREAD(GUNIT,GBUF^,FBLKSIZE*L4,L5);
      IF IORESULT <> 0 THEN
        FOR BLK := L5 TO L5+L4-1 DO
          BEGIN
            UNITREAD(GUNIT,GBUF^,FBLKSIZE,BLK);
            IF IORESULT <> 0 THEN
              BEGIN
                TOTAL := TOTAL + 1;
                WHICHFILE(BLK,TRUE);
                WRITELN('Block ',BLK,' is bad')
              END
          END;
      L5 := L5 + L4
    UNTIL L5 >= NBLOCKS;
    WRITELN(TOTAL,' bad blocks');
    PRINTFILES
  END {BADBLOCKS} ;

  PROCEDURE XBLOCKS;
  VAR
    NEWDIR : ^INTEGER;
    BAD : ARRAY [0..HALFMAXDIR] OF RECORD
            FIRST,LAST : INTEGER
          END;
    BLK,LASTBLOCK,LOC,FIRSTBLK,LASTBLK : INTEGER;
    LDE : DIRENTRY;
    A,B : ABLOCK;

  BEGIN { XBLOCKS }
    CHECKFILE('Examine blocks on','',1,BLKDEXP,FALSE,FALSE,[OKDIR,BADDIR]);
    CLEARLINE;
    WRITE('Block-range ? ');
    READ(FIRSTBLK);
    IF EOLN THEN
      LASTBLK := FIRSTBLK
    ELSE
      BEGIN
        READ(LASTBLK);
        IF NOT EOLN THEN
          WRITELN;
        LASTBLK := ABS(LASTBLK)
      END;
    IF GDIR <> NIL THEN
      IF LASTBLK >= GDIR^ [0].DEOVBLK THEN
        LASTBLK := GDIR^ [0].DEOVBLK-1;
    IF (FIRSTBLK < 0) OR (FIRSTBLK > LASTBLK) THEN
      EXIT(XBLOCKS);
    CLEARSCREEN;
    WRITELN;
    FOR BLK := FIRSTBLK TO LASTBLK DO
      WHICHFILE(BLK,TRUE);
    IF DIRMAP.ENTRIES > 0 THEN
      BEGIN
        PRINTFILES;
        WRITE('Fix bad blocks? ');
        IF NOT NGETCHAR(TRUE) THEN
          EXIT(CALLPROC)
      END;
    FILLCHAR(BAD,SIZEOF(BAD),0);
    FILLCHAR(DIRMAP,SIZEOF(DIRMAP),0);
    LOC := 0;
    LASTBLOCK := -10;
    FOR BLK := FIRSTBLK TO LASTBLK DO
      BEGIN
        WRITE('Block ',BLK);
        UNITREAD(GUNIT,A,FBLKSIZE,BLK);
        UNITWRITE(GUNIT,A,FBLKSIZE,BLK);
        IF IORESULT = 16 THEN
          BEGIN
            WRITELN;
            WRITELN('Disk is write-protected. Cannot examine blocks.');
            EXIT(XBLOCKS)
          END;
        IF IORESULT = 0 THEN
          UNITREAD(GUNIT,B,FBLKSIZE,BLK);
        IF (IORESULT = 0) AND (A = B) THEN
          WRITELN(' may be ok')
        ELSE
          BEGIN
            WRITELN(' is bad');
            WHICHFILE(BLK,TRUE);
            IF GDIR <> NIL THEN
              IF BLK > GDIR^ [0].DLASTBLK THEN
                BEGIN
                  IF LASTBLOCK+1 <> BLK THEN
                    BEGIN
                      BAD [0].FIRST := BAD [0].FIRST + 1;
                      LOC := LOC + 1;
                      IF LOC > (MAXDIR+DIRMAP.ENTRIES-GDIR^ [0].DNUMFILES) THEN
                        MESSAGES(ORD(INOROOM),0,TRUE);
                      BAD [LOC].FIRST := BLK
                    END;
                  BAD [LOC].LAST := BLK;
                  LASTBLOCK := BLK
                END
          END
      END;
    IF BAD [0].FIRST = 0 THEN
      EXIT(XBLOCKS);
    PRINTFILES;
    WRITE('Mark bad blocks  ?');
    IF DIRMAP.ENTRIES > 0 THEN
      WRITE(' (files will be removed !)');
    WRITE(' (Y/N) ');
    IF NOT NGETCHAR(TRUE) THEN
      EXIT(CALLPROC);
    ZAPENTRIES(DIRMAP,FALSE);
    WITH LDE DO
      BEGIN
        DFKIND := XDSKFILE;
        DLASTBYTE := FBLKSIZE;
        DACCESS := THEDATE;
        DTID := 'BAD.xxxxx.BAD'
      END;
    FOR BLK := 1 TO BAD [0].FIRST DO
      WITH LDE,BAD[BLK] DO
        BEGIN
          DFIRSTBLK := FIRST;
          DLASTBLK := LAST+1;
          FOR LOC := 4 DOWNTO 0 DO
            BEGIN
              DTID[9-LOC] := CHR(FIRST DIV IPOT[LOC] + ORD('0'));
              FIRST := FIRST MOD IPOT[LOC]
            END;
          LOC := GDIR^[0].DNUMFILES;
          WHILE DFIRSTBLK < GDIR^[LOC].DLASTBLK DO
            LOC := LOC - 1;
          INSENTRY(LDE,LOC+1,GDIR);
        END;
    UPDATEDIR;
    WRITE('Bad blocks marked')
  END {XBLOCKS} ;

  { One KRUNCHIT for both directions: the counts are signed, and
    DIRSIGN is 0 moving forward and 1 moving back. }
  PROCEDURE KRUNCH;
  VAR
    L1 : STRING[9];   { 'forward ' or 'back ' }
    NBLOCKS,GINX,SPLIT,FIRSTBLK,RELBLOCK,CHUNKSIZE : INTEGER;
    DIRSIGN : INTEGER;
    REBOOT: BOOLEAN;

    PROCEDURE KRUNCHIT(VAR STARTING,OTHER,STOPPING : INTEGER);
    BEGIN
    WITH GDIR^[GINX] DO
      IF DFKIND <> XDSKFILE THEN
        BEGIN
          WRITELN('Moving ',L1,DTID);
          IF DTID = 'SYSTEM.PASCAL' THEN
            REBOOT := GVID = SYVID;
          NBLOCKS := OTHER-STARTING;
          RELBLOCK := 0;
          REPEAT
            CHUNKSIZE := NBLOCKS-RELBLOCK;
            IF ABS(CHUNKSIZE) > GBUFBLKS THEN
              CHUNKSIZE := GBUFBLKS*(CHUNKSIZE DIV ABS(CHUNKSIZE));
            UNITREAD(GUNIT,GBUF^,FBLKSIZE*ABS(CHUNKSIZE),
                     STARTING+RELBLOCK+DIRSIGN*CHUNKSIZE);
            MESSAGES(IORESULT,0,TRUE);
            UNITWRITE(GUNIT,GBUF^,FBLKSIZE*ABS(CHUNKSIZE),
                      STOPPING+RELBLOCK+DIRSIGN*CHUNKSIZE);
            MESSAGES(IORESULT,0,TRUE);
            RELBLOCK := RELBLOCK+CHUNKSIZE
          UNTIL ABS(RELBLOCK) >= ABS(NBLOCKS);
          OTHER := STOPPING+NBLOCKS;
          STARTING := STOPPING;
          UPDATEDIR
        END
    END;

  BEGIN { KRUNCH }
    CHECKFILE('Crunch','',1,BLKDEXP,FALSE,FALSE,[OKDIR]);
    GETBLOCKS('From end of Volume, block','','Starting at block #',
                                               GDIR^ [0].DLASTBLK,SPLIT);
    WHICHFILE(SPLIT,FALSE);
    REBOOT := FALSE;
    SYSCOM^.MISCINFO.NOBREAK := TRUE;
    CLEARSCREEN;
    DIRSIGN := 0;
    WRITELN;
    L1 := 'forward ';
    FOR GINX := 1 TO SPLIT-1 DO
      WITH GDIR^ [GINX] DO
        IF DFIRSTBLK > GDIR^ [GINX-1].DLASTBLK THEN
          KRUNCHIT(DFIRSTBLK,DLASTBLK,GDIR^ [GINX-1].DLASTBLK);
    FIRSTBLK := GDIR^[0].DEOVBLK;
    DIRSIGN := 1;
    L1 := 'back ';
    FOR GINX := GDIR^[0].DNUMFILES DOWNTO SPLIT DO
      WITH GDIR^ [GINX] DO
        BEGIN
          IF DLASTBLK < FIRSTBLK THEN
            KRUNCHIT(DLASTBLK,DFIRSTBLK,FIRSTBLK);
          FIRSTBLK := DFIRSTBLK
        END;
    WRITELN(GVID,': crunched');
    IF REBOOT THEN
      BEGIN
        WRITELN('Please re-boot');
        WHILE TRUE DO
      END;
    SYSCOM^.MISCINFO.NOBREAK := FALSE
  END { KRUNCH };

  PROCEDURE CALLPROC;
  VAR
    X,Y : INTEGER;
    OK : BOOLEAN;

    PROCEDURE PROMPTEM(STR : STRING);
    BEGIN
      PL := STR;
      PROMPT;
      CH := GETCHAR(NOT OK);
      OK := CH IN [' ','?','B'..'E','G','K'..'N','P'..'T','V'..'X','Z']
    END;

  BEGIN
    INITGLOBALS;
    INSTRING := '';
    OK := TRUE;
    REPEAT
      IF FAST THEN
        PROMPTEM(
'Filer: L(dir E(dir R(em T(rans C(hng D(ate P(refix K(rnch Z(ero V(ols Q(uit ?')
      ELSE
        PROMPTEM('Filer: L, E, R, T, C, D, P, K, Z, V, Q,?');
      IF CH = '?' THEN
        IF FAST THEN
          PROMPTEM(
            'Filer: M(ake W(hat N(ew S(ave G(et B(ad-blks X(amine [1.3]')
        ELSE
          PROMPTEM('Filer: M, W, N, S, G, B, X [1.3]')
    UNTIL OK;
    HOMECURSOR;
    CLEARLINE;
    IF CH IN [' ','B','D','Q','V','Z'] THEN
      CLEARSCREEN;
    FOR X := 1 TO 19 DO
    WITH UNITABLE[X] DO
      IF UVID <> '' THEN
        FOR Y := X+1 TO MAXUNIT DO
          IF UVID = UNITABLE[Y].UVID THEN
            BEGIN
              WRITEANDCLEAR;
              WRITE('Warning - volumes #',X,' and #',Y,' have the same name.');
              WRITEANDCLEAR
            END;
    CASE CH OF
      'L' : LISTDIR(FALSE);
      'E' : LISTDIR(TRUE);
      'G' : GETWORK;
      'N' : NEWWORK(TRUE);
      'C' : CHANGER;
      'R' : REMOVER;
      'T' : TRANSFER;
      'S' : IF SAVEWORK THEN TRANSFER;
      'P' : BEGIN
              WRITELN('Prefix is ',DKVID,':');
              CHECKFILE('Prefix filenames by','',1,VOLEXP,FALSE,
                                  FALSE,[NOVOL,BADDIR,OKDIR,UNBLKDVOL]);
              DKVID := GVID;
              CLEARLINE;
              WRITE('Prefix is ',DKVID,':')
            END;
      'W' : WHATWORK;
      'M' : MAKEFILE;
      'V' : LISTVOLS;
      'B' : BADBLOCKS;
      'Z' : ZEROVOLUME;
      'X' : XBLOCKS;
      'K' : KRUNCH;
      'D' : DATESET;
      'Q' : BEGIN
              IF CHAIN_MSG <> '' THEN
                BEGIN
                  IF LENGTH(CHAIN_MSG) > 24 THEN
                    CHAIN_NAME := COPY(CHAIN_MSG,1,24)
                  ELSE
                    CHAIN_NAME := CHAIN_MSG;
                  CHAIN_MSG := ''
                END;
              EXIT(FILEHANDLER)
            END;
      ' ': ; (* accepted by the prompt set; no command *)
      OTHERWISE
        BEGIN
          WRITE(CHR(7));
          WRITEANDCLEAR;
          WRITE('Unknown command')
        END
    END
  END;

  { The version check the linker has too: the byte at $BF21 is 4
    in a 1.3 SYSTEM.PASCAL. }
  PROCEDURE CHECKVER;
    TYPE BYTE1 = PACKED ARRAY [0..0] OF 0..255;
    VAR R: RECORD
             CASE BOOLEAN OF
               TRUE: (I: INTEGER);
               FALSE: (P: ^BYTE1)
           END;
  BEGIN
    R.I := -16607;
    IF R.P^[0] <> 4 THEN
      BEGIN
        WRITELN;
        WRITELN;
        WRITELN('Version 1.3 of SYSTEM.FILER cannot run');
        WRITELN('with a non-1.3 version of SYSTEM.PASCAL');
        EXIT(FILEHANDLER)
      END
  END;

BEGIN {FILEHANDLER}
  CHECKVER;
  WITH USERINFO DO
    BEGIN
      TEXTSAVED := NOT GOTSYM OR (GOTSYM AND (SYMTID <> 'SYSTEM.WRK.TEXT'));
      CODESAVED := NOT GOTCODE OR (GOTCODE AND (CODETID <> 'SYSTEM.WRK.CODE'))
    END;
  FAST := (SYSCOM^.CRTINFO.WIDTH >= 80) AND (NOT SYSCOM^.MISCINFO.SLOWTERM);
  MONTHSTR := '???JanFebMarAprMayJunJulAugSepOctNovDec?????????';
  TYPESTR := '    Bad CodeTextInfoDataGrafFoto';

  REPEAT
    MARK(GBUF);
    GBUFBLKS := 0;
    REPEAT
      NEW(BLOCKPTR);
      GBUFBLKS := GBUFBLKS+1;
    UNTIL ((MEMAVAIL < SIZEOF(DIRECTORY)+SIZEOF(DIRECTORY)+SIZEOF(FIB)+1024)
          AND (MEMAVAIL > 0)) OR (GBUFBLKS = 63);
    CALLPROC;
    IF UNITABLE[DESTUNIT].UISBLKD THEN
      CLOSE(LFIB,PURGE)
    ELSE
      CLOSE(LFIB);
    RELEASE(GBUF)
  UNTIL FALSE
END;
