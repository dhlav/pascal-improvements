;---------------------------------------------------
; UTILITY's six native procedures, BACKUP.CODE
; segment 7: UPCHAR..RSTSTAT.
;
; Declared EXTERNAL in ppm/src/UTILITY.TEXT. Assemble
; with Apple's SYSTEM.ASSMBLER and link the result
; into BACKUP with the compiled unit; the assembler
; emits the relocation tables.
;
; Shipped order in the segment is 43, 44, 46, 47,
; 48, 45 -- assembly order, not interface order.
;
; UPCHAR uppercases one CHAR (a-z and $E1-$FA).
; UPSTR uppercases a VAR string in place, high bit
; off. FORMDISK indexes a table in its own image and
; JSR $1300. SAVSTAT saves the processor status in
; SAVEP; DISINT SEI; RSTSTAT restores SAVEP. SAVEP is
; .DEF/.REF so RSTSTAT's load is segment-relative.
;
; Zero page $00-$03. A native procedure runs inside
; the interpreter; everything below $50 is free.
;---------------------------------------------------

RET     .EQU 00
STRP    .EQU 02

        .FUNC UPCHAR,1
        .TITLE "UPCHAR Function"
; FUNCTION UPCHAR(CH: CHAR): CHAR
        PLA
        STA RET
        PLA
        STA RET+1
;
; Four bytes of function result, then CH. Low byte
; is the character; the high byte is dropped.
;
        PLA
        PLA
        PLA
        PLA
        PLA
        TAY
        PLA
        LDA #00
        PHA
        TYA
        CMP #061
        BCC UNDER
        CMP #07B
        BCC CONV
UNDER   CMP #0E1
        BCC STORE
        CMP #0FB
        BCS STORE
CONV    CLD
        SEC
        SBC #020
STORE   PHA
        LDA RET+1
        PHA
        LDA RET
        PHA
        RTS
        .BYTE 000

        .PROC UPSTR,1
        .TITLE "UPSTR Procedure"
; PROCEDURE UPSTR(VAR S: STR80)
        PLA
        STA RET
        PLA
        STA RET+1
        PLA
        STA STRP
        PLA
        STA STRP+1
        CLD
        LDY #00
        LDA @STRP,Y
        BEQ DONE
        TAX
LOOP    INY
        LDA @STRP,Y
        AND #07F
        CMP #061
        BCC HI
        CMP #07B
        BCC CONV44
HI      CMP #0E1
        BCC NEXT
        CMP #0FB
        BCS NEXT
CONV44  SEC
        SBC #020
NEXT    STA @STRP,Y
        DEX
        BNE LOOP
DONE    LDA RET+1
        PHA
        LDA RET
        PHA
        RTS

        .PROC SAVSTAT,0
        .TITLE "SAVSTAT Procedure"
; PROCEDURE SAVSTAT -- PHP into SAVEP for RSTSTAT to restore.
        .DEF SAVEP
        PHP
        PLA
        STA SAVEP
        RTS
SAVEP   .BYTE 000,000

        .PROC DISINT,0
        .TITLE "DISINT Procedure"
; PROCEDURE DISINT -- disable interrupts.
        SEI
        RTS

        .PROC RSTSTAT,0
        .TITLE "RSTSTAT Procedure"
; PROCEDURE RSTSTAT -- restore the status SAVSTAT saved.
        .REF SAVEP
        LDA SAVEP
        PHA
        PLP
        RTS

        .FUNC FORMDISK,1
        .TITLE "FORMDISK Function"
; FUNCTION FORMDISK(N: INTEGER): INTEGER
;
; Low byte of N indexes TABLE; A goes to $1300.
; RTN, TABLE and TEMP live in this procedure's
; image, which is why the seven relocations are
; procedure-relative. JSR $1300 is a literal.
;
        PLA
        STA RTN
        PLA
        STA RTN+1
        PLA
        PLA
        PLA
        PLA
        PLA
        TAX
        PLA
        LDA TABLE,X
        JSR 1300
        STA TEMP
        LDA #00
        PHA
        LDA TEMP
        PHA
        LDA RTN+1
        PHA
        LDA RTN
        PHA
        RTS
RTN     .BYTE 000,000
TABLE   .BYTE 000,000,000,000,060,0E0,000,000,000
        .BYTE 040,0C0,050,0D0
TEMP    .BYTE 000

        .END
