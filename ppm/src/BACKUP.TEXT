{ BACKUP -- Pascal ProFile Manager backup, host segment of
  BACKUP.CODE. USES CHAINSTUFF (intrinsic 28) and UTILITY
  (segment 7). SEGMENT PROCEDURES INITIALI, DOBACKUP,
  DORESTOR take 8, 9, 10. The host declares no variables:
  UTILITY's interface is the whole global frame.

  Host procedures 2..19 are FORWARD so the SEGMENT
  PROCEDURES can sit before any body (error 399 /
  CODEINSEG). INITIALI calls 14, 15, 16, 18, 19. }

USES CHAINSTUFF,
     (*$U WORKHD:UTILITY.CODE*) UTILITY;

  NAME7 = PACKED ARRAY [1..7] OF CHAR;
  BYTEARR = PACKED ARRAY [0..47] OF 0..255;
  BAP = ^BYTEARR;

(*$I-*)
(*$R+*)
SEGMENT PROCEDURE INITIALI;

  PROCEDURE SCANPROF;
  VAR
    PUN: RECORD
           CASE BOOLEAN OF
             FALSE: (A: INTEGER);
             TRUE: (Q: ^NAME7)
         END;
    B: RECORD
         CASE BOOLEAN OF
           FALSE: (I: INTEGER);
           TRUE: (P: ^BAP)
       END;
    LIM: INTEGER;
  BEGIN
    W5325 := 128;
    WHILE W5324 = 0 DO
      BEGIN
        B.I := 232;
        (*$R-*)
        MOVELEFT(B.P^^[3 * (W5325 - 128) + 1], PUN.A, 2);
        (*$R+*)
        PUN.A := PUN.A + 3;
        IF PUN.Q^ = 'PROFILE' THEN
          W5324 := PUN.A - 3
        ELSE
          BEGIN
            W5325 := W5325 + 1;
            IF W5325 > 143 THEN
              P18(12, 0)
          END
      END;
    UNITSTATUS(W5325, VOLS, 16384);
    W791 := IORESULT;
    IF W791 <> 0 THEN
      P18(12, 0);
    W5327 := 1;
    VOLS2 := VOLS;
    W5509 := W5418;
    W5519 := 1;
    LIM := 30;
    WHILE W5519 <= LIM DO
      BEGIN
        VOLS[W5519].BUSY := FALSE;
        W5519 := W5519 + 1
      END;
    W5326 := [];
    W275 := -1;
    W5653 := -1;
    W111 := 0;
    LIM := 7;
    WHILE W111 <= LIM DO
      BEGIN
        P16;
        IF W791 = 0 THEN
          BEGIN
            W5326 := W5326 + [W111];
            IF W5653 < 0 THEN
              W5653 := W111;
            W275 := W111;
            DIRBLK[4] := DIRBLK[4] MOD 16;
            MOVELEFT(DIRBLK[4], VOLNAM[W111], 16)
          END;
        W111 := W111 + 1
      END;
    W275 := W275 + 1;
    IF W275 = 0 THEN
      P18(17, 0)
  END;

BEGIN
  W5324 := 0;
  W5056 := 1;
  W5629 := 0;
  LISTING := 'CONSOLE:';
  DASHES := '-';
  U17(DASHES, 80, '-');
  W5054 := 280;
  W4996 := 5;
  P14;
  FILLCHAR(BLK661, 256, CHR(0));
  W5327 := 0;
  SCANPROF;
  IF W5653 < 0 THEN
    BEGIN
      WRITE('No ProFiles are Available');
      WRITELN;
      U41;
      P19
    END
  ELSE
    W111 := W5653;
  P15
END;
