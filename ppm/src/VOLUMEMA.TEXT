{ VOLUMEMA -- Pascal ProFile Manager, segment 8 of PPM.CODE.

  A plain unit. VOLUI is the one interface heading, so it
  is procedure 2. The helpers below are private and are
  numbered from 3 in this order. IOFAIL and SAFEDIV..PDVALID are
  the same bodies as PASCALPR, compiled in this unit so a
  global load is the long LDO form and EXIT carries
  segment 8. TITLELN..SETWPROT and VOLUI (with nested VOLCELL, CLRCELL)
  are the volume-manager commands. The unit uses VMGR;
  the linker rewrites those operands onto the host frame
  and turns the VMGR call into CXP 7. }
UNIT VOLUMEMA;
INTERFACE
USES (*$U WORKHD:VMGR.CODE*) VMGR;
  PROCEDURE VOLUI;
IMPLEMENTATION
(*$V-*)
  FUNCTION IOFAIL: BOOLEAN;
  BEGIN
    IF (IORESULT <> 0) OR (ERRNUM <> 0) THEN
      BEGIN
        IOERR := IORESULT;
        SHOWERR;
        IOFAIL := TRUE
      END
    ELSE IOFAIL := FALSE
  END;
  { Dividend is the first parameter. Names in one group are
    allocated downward, and a function reserves REALSIZE
    words for its result before them. }
  FUNCTION SAFEDIV(DVND, DVSR: INTEGER): INTEGER;
  VAR T: INTEGER;
  BEGIN
    IF DVND >= 0 THEN
      SAFEDIV := DVND DIV DVSR
    ELSE
      BEGIN
        T := (DVND + 32767) + 1;
        SAFEDIV := ((T DIV DVSR) + (32767 DIV DVSR))
               + ((((T MOD DVSR) + (32767 MOD DVSR)) + 1) DIV DVSR)
      END
  END;

  FUNCTION SAFEMOD(DVND, DVSR: INTEGER): INTEGER;
  VAR T: INTEGER;
  BEGIN
    IF DVND >= 0 THEN
      SAFEMOD := DVND MOD DVSR
    ELSE
      BEGIN
        T := (DVND + 32767) + 1;
        SAFEMOD := (((T MOD DVSR) + (32767 MOD DVSR)) + 1) MOD DVSR
      END
  END;

  PROCEDURE CLRERR;
  BEGIN
    IOERR := 0;
    ERRNUM := 0
  END;

  { ProDOS name: length nibble, first character, then the
    other fourteen. The array arrives by value, so the
    compiler copies it before this body. }
  PROCEDURE MKPDSTR(LEN: INTEGER; CH: CHAR; NAME: NAMEREST;
                VAR DEST: STRING);
  VAR I: INTEGER;
  BEGIN
    { Index 0 is the length byte. IXS rejects 0, so the
      1.3 compiler's check has to stay off for this store. }
    (*$R-*)
    DEST[0] := CHR(LEN);
    (*$R+*)
    IF LEN > 0 THEN
      BEGIN
        DEST[1] := CH;
        FOR I := 2 TO LEN DO
          DEST[I] := NAME[I]
      END
  END;

  FUNCTION DRVCTL(N: DRVNUM): INTEGER;
  BEGIN
    DRVCTL := 2 + N * 8192
  END;

  { Line editor. Backspace and delete erase a character;
    any other control rings the bell. DIGITS keeps only
    '0'..'9'. Returns true when the key was escape.
    No string-index checks: the length byte is stored
    directly, and a checked index of 0 would fault. }
  (*$R-*)
  FUNCTION EDITLINE(VAR S: ANYSTR; LIM: INTEGER;
               DIGITS: BOOLEAN): BOOLEAN;
  VAR N: INTEGER;
      CH: CHAR;
  BEGIN
    N := 0;
    S := '';
    REPEAT
      READ(KEYBOARD, CH);
      IF (NOT EOLN(KEYBOARD)) AND (CH <> CHR(27)) THEN
        IF CH IN [CHR(8), CHR(127)] THEN
          IF N > 0 THEN
            BEGIN
              WRITE(CHR(8));
              WRITE(CHR(32));
              WRITE(CHR(8));
              N := N - 1
            END
          ELSE
            WRITE(CHR(7))
        ELSE
          IF CH IN [CHR(0)..CHR(31)] THEN
            WRITE(CHR(7))
          ELSE
            IF N >= LIM THEN
              WRITE(CHR(7))
            ELSE
              IF (NOT DIGITS) OR (CH IN ['0'..'9']) THEN
                BEGIN
                  N := N + 1;
                  S[N] := CH;
                  WRITE(CH)
                END
              ELSE
                WRITE(CHR(7))
    UNTIL EOLN(KEYBOARD) OR (CH = CHR(27));
    S[0] := CHR(N);
    EDITLINE := CH = CHR(27)
  END;
  (*$R+*)

  { Read up to five digits into N. True means escape. }
  FUNCTION READNUM(VAR N: INTEGER): BOOLEAN;
  VAR BUF: STRING[5];
      ESC: BOOLEAN;
      I: INTEGER;
  BEGIN
    ESC := EDITLINE(BUF, 5, TRUE);
    READNUM := ESC;
    N := 0;
    IF NOT ESC THEN
      FOR I := 1 TO LENGTH(BUF) DO
        N := N * 10 + ORD(BUF[I]) - ORD('0')
  END;
  FUNCTION VOLIDX(N: INTEGER): INTEGER;
  BEGIN
    IF N IN [4, 5] THEN
      VOLIDX := N - 3
    ELSE
      IF N IN [9..20] THEN
        VOLIDX := N - 6
      ELSE
        VOLIDX := (N - 128) + 15
  END;

  FUNCTION UNITNUM(N: INTEGER): INTEGER;
  BEGIN
    IF (N = 1) OR (N = 2) THEN
      UNITNUM := N + 3
    ELSE
      IF N IN [3..14] THEN
        UNITNUM := N + 6
      ELSE
        UNITNUM := (N + 128) - 15
  END;

  PROCEDURE BLANKLN(ROW, NLINES: INTEGER);
  VAR I: INTEGER;
  BEGIN
    GOTOXY(0, ROW);
    FOR I := 1 TO NLINES DO
      BEGIN
        CLRLINE;
        WRITELN
      END;
    GOTOXY(0, ROW)
  END;

  FUNCTION YESNO: BOOLEAN;
  VAR CH: CHAR;
      OK: BOOLEAN;
  BEGIN
    UNITCLEAR(2);
    RESET(KEYBOARD);
    REPEAT
      READ(KEYBOARD, CH);
      OK := CH IN [CHR(27), 'N', 'Y', 'n', 'y'];
      IF NOT OK THEN
        WRITE(CHR(7))
    UNTIL OK;
    IF CH <> CHR(27) THEN
      WRITE(CH);
    YESNO := CH IN ['Y', 'y']
  END;
  { Fold lowercase to upper, then remove every space. }
  PROCEDURE CLEANNM(VAR S: STRING);
  VAR AT: INTEGER;
      I: INTEGER;
  BEGIN
    FOR I := 1 TO LENGTH(S) DO
      IF S[I] IN ['a'..'z'] THEN
        S[I] := CHR(ORD(S[I]) - 32);
    IF POS(' ', S) <> 0 THEN
      REPEAT
        AT := POS(' ', S);
        DELETE(S, AT, 1)
      UNTIL POS(' ', S) = 0
  END;
  { Pascal volume name: one to seven characters, from
    space through 'z', except space, '$', ',', '=' and '?'. }
  FUNCTION PVVALID(NAME: ANYSTR): BOOLEAN;
  VAR I: INTEGER;
  BEGIN
    PVVALID := FALSE;
    IF LENGTH(NAME) IN [1..7] THEN
      BEGIN
        FOR I := 1 TO LENGTH(NAME) DO
          IF (NOT (NAME[I] IN [' '..'z']))
             OR (NAME[I] IN [' ', '$', ',', '=', '?']) THEN
            EXIT(PVVALID);
        PVVALID := TRUE
      END
  END;

  { ProDOS name: one to fifteen characters. The first is
    'A'..'Z'; the rest may also be a digit or a period. }
  FUNCTION PDVALID(NAME: STRING): BOOLEAN;
  VAR I: INTEGER;
  BEGIN
    PDVALID := TRUE;
    IF LENGTH(NAME) IN [1..15] THEN
      BEGIN
        IF NOT (NAME[1] IN ['A'..'Z']) THEN
          PDVALID := FALSE;
        FOR I := 2 TO LENGTH(NAME) DO
          IF NOT (NAME[I] IN ['.', '0'..'9', 'A'..'Z']) THEN
            PDVALID := FALSE
      END
    ELSE
      PDVALID := FALSE
  END;

  { Title on row 0, then '...'. Value STRING[80] is
    copied on entry. }
  PROCEDURE TITLELN(S: STRING);
  BEGIN
    GOTOXY(0, 0);
    CLRLINE;
    WRITE(S);
    WRITE('...')
  END;

  { Cycle DRIVE to the next member of MARKED. }
  PROCEDURE NEXTDRV;
  VAR OLD: INTEGER;
  BEGIN
    CLRERR;
    IF MARKED = [] THEN
      EXIT(NEXTDRV);
    OLD := DRIVE;
    { DRVNUM stores would emit CHK 0..7. }
    (*$R-*)
    REPEAT
      DRIVE := (DRIVE + 1) MOD (7 + 1)
    UNTIL DRIVE IN MARKED;
    IF OLD = DRIVE THEN
      BEGIN
        DRIVE := OLD;
        EXIT(NEXTDRV)
      END;
    (*$R+*)
    BLANKLN(0, 4);
    TITLELN('Moving to next drive');
    VOLS[1].STAT := DRIVE;
    UNITSTATUS(PFUNIT, VOLS, 16386);
    MOUNTDRV(DRIVE);
    IF IOFAIL THEN
      DRVS[0].EXTRA := 0;
    REDRAW
  END;

  PROCEDURE RELVOL;
  VAR UNITN: INTEGER;
      VOL: INTEGER;
  BEGIN
    CLRERR;
    BLANKLN(0, 4);
    TITLELN('Release a volume');
    WRITELN;
    VOL := CHOOSE;
    IF VOL = 0 THEN
      EXIT(RELVOL);
    UNITN := DRVS[VOL].ATTRIB.UNITNO;
    IF UNITN IN [128..143, 4, 5, 9..20] THEN
      IF VOLS[VOLIDX(UNITN)].BUSY THEN
        FREEUNIT(UNITN);
    IF IOFAIL THEN
      EXIT(RELVOL);
    DRAWUNIT(VOL, 0)
  END;

  PROCEDURE ASNVOL;
  VAR VOL: INTEGER;
      UNITN: INTEGER;
  BEGIN
    CLRERR;
    BLANKLN(0, 4);
    TITLELN('Assign a volume');
    WRITELN;
    VOL := CHOOSE;
    IF VOL = 0 THEN
      EXIT(ASNVOL);
    REPEAT
      GOTOXY(0, 1);
      CLRLINE;
      WRITE('Assign to what Pascal unit number? ');
      IF READNUM(UNITN) THEN
        EXIT(ASNVOL);
      IF NOT (UNITN IN [128..143, 4, 5, 9..20]) THEN
        BEGIN
          ERRNUM := 5;
          SHOWERR
        END
    UNTIL UNITN IN [128..143, 4, 5, 9..20];
    BINDUNIT(VOL, UNITN);
    IF IOFAIL THEN
      EXIT(ASNVOL);
    GOTOXY(0, CURROW);
    CLRLINE;
    REDRAW
  END;

  PROCEDURE DOKRUNCH;
  BEGIN
    CLRERR;
    BLANKLN(0, 4);
    WRITELN;
    WRITE('Krunch the Pascal area? (Y/N) ');
    IF YESNO THEN
      BEGIN
        GOTOXY(0, 3);
        CLRLINE;
        WRITE('Krunching...');
        KRUNCHPA;
        IF NOT IOFAIL THEN
          BEGIN
            GOTOXY(0, 3);
            CLRLINE
          END
      END
  END;

  PROCEDURE DELVOL;
  VAR VOL: INTEGER;
      UNUSED: STRING;
      SPARE: INTEGER;
      KRUNCH: BOOLEAN;
  BEGIN
    KRUNCH := FALSE;
    CLRERR;
    BLANKLN(0, 4);
    TITLELN('Delete a volume');
    WRITELN;
    VOL := CHOOSE;
    IF VOL = 0 THEN
      EXIT(DELVOL);
    IF DRVS[VOL].ATTRIB.WPROT THEN
      BEGIN
        ERRNUM := 10;
        SHOWERR;
        EXIT(DELVOL)
      END;
    GOTOXY(0, 2);
    CLRLINE;
    WRITE('Delete Highlighted Volume ');
    WRITE(VNAMES[VOL]);
    WRITE(': (Y/N)? ');
    IF YESNO THEN
      BEGIN
        GOTOXY(0, 2);
        CLRLINE;
        WRITE('Krunch the Pascal area (Y/N)?');
        IF YESNO THEN
          BEGIN
            KRUNCH := TRUE;
            GOTOXY(0, 3);
            CLRLINE;
            WRITE('Krunching...');
            REMVOL(VOL, TRUE)
          END
        ELSE
          REMVOL(VOL, FALSE);
        IF IOFAIL THEN
          EXIT(DELVOL);
        IF KRUNCH THEN
          BEGIN
            GOTOXY(0, 3);
            CLRLINE
          END;
        REDRAW
      END
  END;

  PROCEDURE MAKEVOL;
  VAR SIZE: INTEGER;
      SPARE: INTEGER;
      NAME: STRING[7];
      DESC: STRING[15];
      OK: BOOLEAN;
      RES: INTEGER;
  BEGIN
    CLRERR;
    IF DRVS[0].EXTRA = 31 THEN
      BEGIN
        ERRNUM := 3;
        SHOWERR;
        EXIT(MAKEVOL)
      END;
    BLANKLN(0, 4);
    TITLELN('Create a volume');
    REPEAT
      OK := FALSE;
      BLANKLN(1, 1);
      WRITE('What is the name for this volume? ');
      IF EDITLINE(NAME, 7, FALSE) THEN
        BEGIN
          GOTOXY(0, CURROW);
          CLRLINE;
          WRITE(CHR(14));
          DRAWROW(HILITE);
          EXIT(MAKEVOL)
        END;
      CLEANNM(NAME);
      OK := PVVALID(NAME);
      IF NOT OK THEN
        BEGIN
          ERRNUM := 9;
          SHOWERR
        END
    UNTIL OK;
    GOTOXY(0, CURROW);
    CLRLINE;
    BLANKLN(1, 1);
    WRITE('What is the description field? ');
    IF EDITLINE(DESC, 15, FALSE) THEN
      EXIT(MAKEVOL);
    REPEAT
      OK := FALSE;
      BLANKLN(1, 1);
      WRITE('What is the size of this volume in blocks? ');
      IF READNUM(SIZE) THEN
        EXIT(MAKEVOL);
      OK := NOT (SIZE <= 6);
      IF NOT OK THEN
        BEGIN
          ERRNUM := 13;
          SHOWERR
        END
    UNTIL OK;
    GOTOXY(0, CURROW);
    CLRLINE;
    RES := ADDVOL(NAME, DESC, SIZE);
    IF RES <> 0 THEN
      REDRAW
    ELSE
      SHOWERR
  END;

  PROCEDURE MODVOL;
  VAR VOL: INTEGER;
      DESC: STRING;
      NAME: STRING;
      OK: BOOLEAN;
  BEGIN
    CLRERR;
    BLANKLN(0, 4);
    TITLELN('Modify a volume');
    VOL := CHOOSE;
    IF VOL = 0 THEN
      EXIT(MODVOL);
    IF DRVS[VOL].ATTRIB.WPROT THEN
      BEGIN
        ERRNUM := 10;
        SHOWERR;
        EXIT(MODVOL)
      END;
    REPEAT
      OK := FALSE;
      BLANKLN(1, 1);
      WRITE('What is the new name for this volume? RETURN for no change. ');
      IF EDITLINE(NAME, 7, FALSE) THEN
        BEGIN
          GOTOXY(0, CURROW);
          CLRLINE;
          WRITE(CHR(14));
          DRAWROW(HILITE);
          EXIT(MODVOL)
        END;
      CLEANNM(NAME);
      OK := PVVALID(NAME) OR (NAME = '');
      IF NOT OK THEN
        BEGIN
          ERRNUM := 9;
          SHOWERR
        END
    UNTIL OK;
    GOTOXY(0, CURROW);
    CLRLINE;
    BLANKLN(1, 1);
    WRITE('What is the new description field? RETURN for no change. ');
    IF EDITLINE(DESC, 15, FALSE) THEN
      EXIT(MODVOL);
    RENAVOL(VOL, NAME, DESC);
    IF IOFAIL THEN
      EXIT(MODVOL);
    DRAWROW(VOL)
  END;

  PROCEDURE SETWPROT;
  VAR VOL: INTEGER;
      FLAG: BOOLEAN;
  BEGIN
    CLRERR;
    BLANKLN(0, 4);
    TITLELN('Change the write-protection of a volume');
    WRITELN;
    VOL := CHOOSE;
    IF VOL = 0 THEN
      EXIT(SETWPROT);
    IF DRVS[VOL].ATTRIB.WPROT THEN
      FLAG := FALSE
    ELSE
      FLAG := TRUE;
    SETWP(VOL, FLAG);
    IF IOFAIL THEN
      EXIT(SETWPROT);
    DRAWWP(VOL, FLAG)
  END;

  PROCEDURE VOLUI;
  VAR CH1: CHAR;
      CH: CHAR;
      I: INTEGER;
      ROW: INTEGER;
      COL: INTEGER;

    { Screen cell for volume index N. Two columns of
      sixteen, rows starting at 7. }
    PROCEDURE VOLCELL(N: INTEGER);
    BEGIN
      IF (N <= 16) OR ((N >= 33) AND (N <= 48)) THEN
        COL := 2
      ELSE
        COL := 42;
      IF N <= 16 THEN
        ROW := N - 1
      ELSE
        IF N <= 32 THEN
          ROW := N - 17
        ELSE
          IF N <= 48 THEN
            ROW := N - 33
          ELSE
            IF N <= 64 THEN
              ROW := N - 49;
      ROW := ROW + 7
    END;

    PROCEDURE CLRCELL;
    BEGIN
      VOLCELL(HILITE);
      GOTOXY(COL - 2, ROW);
      WRITE('  ')
    END;

  BEGIN (*$R VMGR*)
    CLRERR;
    BLANKLN(0, 4);
    INITDRV;
    ERRHERE := 1;
    IF MARKED = [] THEN
      BEGIN
        GOTOXY(0, CURROW);
        WRITE(CHR(7));
        WRITELN('ERROR: There are no Profiles with Pascal Areas.');
        REPEAT
          GOTOXY(0, 4);
          CLRLINE;
          WRITE(CHR(15));
          WRITE('Press SPACE to continue');
          WRITE(CHR(14));
          UNITCLEAR(1);
          READ(CH1)
        UNTIL (CH1 = ' ') AND NOT EOLN;
        EXIT(VOLUI)
      END;
    IF DRIVE IN MARKED THEN
      MOUNTDRV(DRIVE)
    ELSE
      BEGIN
        I := 0;
        WHILE I <= 7 DO
          IF I IN MARKED THEN
            BEGIN
              MOUNTDRV(I);
              I := 7 + 1
            END
          ELSE
            I := I + 1
      END;
    IF IOFAIL THEN
      DRVS[0].EXTRA := 0;
    REDRAW;
    REPEAT
      IF ERRNUM + IOERR = 0 THEN
        BEGIN
          CLRCELL;
          WRITE(CHR(14));
          DRAWROW(HILITE);
          HILITE := 0
        END;
      CLRERR;
      BLANKLN(0, 3);
      WRITE('Volume Manager         Copyright 1983 Apple Computer, Inc.     Version 1.0.1');
      WRITELN;
      GOTOXY(0, 1);
      WRITE('----------------------------------------------------------------------------');
      GOTOXY(0, 2);
      WRITE('A(ssign, R(elease, C(reate, D(elete, W(Prot, M(odify, K(runch, N(ext, Q(uit ');
      READ(KEYBOARD, CH);
      IF CH IN ['a'..'z'] THEN
        CH := CHR(ORD(CH) - ORD('a') + ORD('A'));
      GOTOXY(0, CURROW);
      CLRLINE;
      WRITELN;
      CASE CH OF
        'C': MAKEVOL;
        'D': DELVOL;
        'A': ASNVOL;
        'R': RELVOL;
        'M': MODVOL;
        'W': SETWPROT;
        'K': DOKRUNCH;
        'N': NEXTDRV
      END
    UNTIL CH = 'Q';
    BLANKLN(0, 4)
  END;
BEGIN
END.
