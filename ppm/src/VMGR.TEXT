{ VMGR -- Pascal ProFile Manager, segment 7 of PPM.CODE.

  A plain unit, compiled on its own and USE'd by PASCALPR.
  Not a SEGMENT PROCEDURE: those nest at lex 2 and the
  program cannot call them (error 104). Every procedure
  here is lex 1, and procedure 1 returns with RNP, which
  is the unit initialisation UNITBODY builds (finding 232).

  Screen helpers (PUTCRT, HOMECURSOR, V23, V24, P27) load
  SYSCOM as LOD 2,1. 1.3 does not predefine SYSCOM, so this
  file is compiled (*$U-*) with SYSCOM declared at level 0
  before the unit. (*$NS 7*) keeps the unit on segment 7.
  (*$R+*) and (*$I+*) put back the checks (*$U-*) cleared.

  V22's "Pascal System" error string ships with a high-ASCII
  space (byte 160) between the two words. Parked: the source
  keeps a normal space and we will change it later. Until
  then ppm/work/stage_vmgr.py can write the shipped byte;
  stagefile.py is ASCII-only.

  Three 1.3-vs-1.1 leftovers, same class as FILEHAND's six
  version bytes: P58 WHILE AND is LAND then one FJP (1.1
  chained two FJPs); P66's match-arm UJP is +7 (1.1 omitted
  skip-else after GOTO, so +5); P77's last jump-table word
  is IC-JTAB (1.1 self-rel is off by data_size 24). }
(*$U-*)
(*$NS 7*)
(*$R+*)
(*$I+*)

  PROCEDURE PUTCRT(BIT: INTEGER; CH: CHAR);
  BEGIN
    WITH SYSCOM^ DO
      IF CH <> CHR(0) THEN
        BEGIN
          IF CRTCTRL.PREFIXED[BIT] THEN
            WRITE(CRTCTRL.ESCAPE);
          WRITE(CH)
        END
  END;
  (*$R+*)

  PROCEDURE HOMECURSOR;
  BEGIN
    PUTCRT(4, SYSCOM^.CRTCTRL.HOME)
  END;

  PROCEDURE V23;
  BEGIN
    HOMECURSOR;
    WITH SYSCOM^, CRTCTRL DO
      IF ERASEEOS <> CHR(0) THEN
        PUTCRT(3, ERASEEOS)
      ELSE
        PUTCRT(6, CLEARSCREEN)
  END;

  PROCEDURE V24;
  VAR I: INTEGER;
  BEGIN
    (*$I-*)
    WITH SYSCOM^, CRTCTRL DO
      IF ERASEEOL <> CHR(0) THEN
        PUTCRT(2, ERASEEOL)
      ELSE
        IF CLEARLINE <> CHR(0) THEN
          PUTCRT(7, CLEARLINE)
        ELSE
          IF (NOT MISCINFO.SLOWTERM) AND (RLF <> CHR(0)) THEN
            BEGIN
              FOR I := 2 TO CRTINFO.WIDTH DO
                WRITE(' ');
              WRITELN;
              PUTCRT(0, RLF)
            END
    (*$I+*)
  END;

  PROCEDURE P27;
  BEGIN
    WITH SYSCOM^ DO
      BEGIN
        W937 := CRTINFO.WIDTH;
        W1969 := CRTINFO.HEIGHT;
        W2228 := ORD(CRTINFO.CHARDEL);
        W935 := ORD(CRTINFO.LINEDEL);
        W930 := ORD(CRTINFO.ETX);
        W1971 := ORD(CRTINFO.ALTMODE);
        SETB := [CHR(11), CHR(15)];
        SETA := [CHR(10), CHR(12)];
        W936 := ORD(CRTINFO.RIGHT);
        W934 := ORD(CRTINFO.LEFT)
      END
  END;

  FUNCTION IOFAIL: BOOLEAN;
  BEGIN
    IOERR := IORESULT;
    IOFAIL := (IORESULT <> 0) OR (ERRNUM <> 0)
  END;

  FUNCTION P04(DVND, DVSR: INTEGER): INTEGER;
  VAR T: INTEGER;
  BEGIN
    IF DVND >= 0 THEN
      P04 := DVND DIV DVSR
    ELSE
      BEGIN
        T := (DVND + 32767) + 1;
        P04 := ((T DIV DVSR) + (32767 DIV DVSR))
               + ((((T MOD DVSR) + (32767 MOD DVSR)) + 1) DIV DVSR)
      END
  END;

  FUNCTION P05(DVND, DVSR: INTEGER): INTEGER;
  VAR T: INTEGER;
  BEGIN
    IF DVND >= 0 THEN
      P05 := DVND MOD DVSR
    ELSE
      BEGIN
        T := (DVND + 32767) + 1;
        P05 := (((T MOD DVSR) + (32767 MOD DVSR)) + 1) MOD DVSR
      END
  END;

  PROCEDURE CLRERR;
  BEGIN
    IOERR := 0;
    ERRNUM := 0
  END;

  PROCEDURE P07(LEN: INTEGER; CH: CHAR; NAME: NAMEREST;
                VAR DEST: STRING);
  VAR I: INTEGER;
  BEGIN
    (*$R-*)
    DEST[0] := CHR(LEN);
    (*$R+*)
    IF LEN > 0 THEN
      BEGIN
        DEST[1] := CH;
        FOR I := 2 TO LEN DO
          DEST[I] := NAME[I]
      END
  END;

  FUNCTION P08(N: DRVNUM): INTEGER;
  BEGIN
    P08 := 2 + N * 8192
  END;

  (*$I-*)
  (*$R-*)
  FUNCTION P09(VAR S: ANYSTR; LIM: INTEGER;
               DIGITS: BOOLEAN): BOOLEAN;
  VAR N: INTEGER;
      CH: CHAR;
  BEGIN
    N := 0;
    S := '';
    REPEAT
      READ(KEYBOARD, CH);
      IF (NOT EOLN(KEYBOARD)) AND (CH <> CHR(27)) THEN
        IF CH IN [CHR(8), CHR(127)] THEN
          IF N > 0 THEN
            BEGIN
              WRITE(CHR(8));
              WRITE(CHR(32));
              WRITE(CHR(8));
              N := N - 1
            END
          ELSE
            WRITE(CHR(7))
        ELSE
          IF CH IN [CHR(0)..CHR(31)] THEN
            WRITE(CHR(7))
          ELSE
            IF N >= LIM THEN
              WRITE(CHR(7))
            ELSE
              IF (NOT DIGITS) OR (CH IN ['0'..'9']) THEN
                BEGIN
                  N := N + 1;
                  S[N] := CH;
                  WRITE(CH)
                END
              ELSE
                WRITE(CHR(7))
    UNTIL EOLN(KEYBOARD) OR (CH = CHR(27));
    S[0] := CHR(N);
    P09 := CH = CHR(27)
  END;
  (*$R+*)
  (*$I+*)

  FUNCTION P10(VAR N: INTEGER): BOOLEAN;
  VAR BUF: STRING[5];
      ESC: BOOLEAN;
      I: INTEGER;
  BEGIN
    ESC := P09(BUF, 5, TRUE);
    P10 := ESC;
    N := 0;
    IF NOT ESC THEN
      FOR I := 1 TO LENGTH(BUF) DO
        N := N * 10 + ORD(BUF[I]) - ORD('0')
  END;

  FUNCTION P11(N: INTEGER): INTEGER;
  BEGIN
    IF N IN [4, 5] THEN
      P11 := N - 3
    ELSE
      IF N IN [9..20] THEN
        P11 := N - 6
      ELSE
        P11 := (N - 128) + 15
  END;

  FUNCTION P12(N: INTEGER): INTEGER;
  BEGIN
    IF (N = 1) OR (N = 2) THEN
      P12 := N + 3
    ELSE
      IF N IN [3..14] THEN
        P12 := N + 6
      ELSE
        P12 := (N + 128) - 15
  END;

  (*$I-*)
  PROCEDURE P13(ROW, NLINES: INTEGER);
  VAR I: INTEGER;
  BEGIN
    GOTOXY(0, ROW);
    FOR I := 1 TO NLINES DO
      BEGIN
        V24;
        WRITELN
      END;
    GOTOXY(0, ROW)
  END;

  FUNCTION P14: BOOLEAN;
  VAR CH: CHAR;
      OK: BOOLEAN;
  BEGIN
    UNITCLEAR(2);
    RESET(KEYBOARD);
    REPEAT
      READ(KEYBOARD, CH);
      OK := CH IN [CHR(27), 'N', 'Y', 'n', 'y'];
      IF NOT OK THEN
        WRITE(CHR(7))
    UNTIL OK;
    IF CH <> CHR(27) THEN
      WRITE(CH);
    P14 := CH IN ['Y', 'y']
  END;
  (*$I+*)

  PROCEDURE P15(VAR S: STRING);
  VAR AT: INTEGER;
      I: INTEGER;
  BEGIN
    FOR I := 1 TO LENGTH(S) DO
      IF S[I] IN ['a'..'z'] THEN
        S[I] := CHR(ORD(S[I]) - 32);
    IF POS(' ', S) <> 0 THEN
      REPEAT
        AT := POS(' ', S);
        DELETE(S, AT, 1)
      UNTIL POS(' ', S) = 0
  END;

  FUNCTION P16(NAME: ANYSTR): BOOLEAN;
  VAR I: INTEGER;
  BEGIN
    P16 := FALSE;
    IF LENGTH(NAME) IN [1..7] THEN
      BEGIN
        FOR I := 1 TO LENGTH(NAME) DO
          IF (NOT (NAME[I] IN [' '..'z']))
             OR (NAME[I] IN [' ', '$', ',', '=', '?']) THEN
            EXIT(P16);
        P16 := TRUE
      END
  END;

  FUNCTION P17(NAME: STRING): BOOLEAN;
  VAR I: INTEGER;
  BEGIN
    P17 := TRUE;
    IF LENGTH(NAME) IN [1..15] THEN
      BEGIN
        IF NOT (NAME[1] IN ['A'..'Z']) THEN
          P17 := FALSE;
        FOR I := 2 TO LENGTH(NAME) DO
          IF NOT (NAME[I] IN ['.', '0'..'9', 'A'..'Z']) THEN
            P17 := FALSE
      END
    ELSE
      P17 := FALSE
  END;

  PROCEDURE P18(VAR BUF: DIRBLOCK; VAR ENTLEN, ENTS: INTEGER);
  BEGIN
    (*$I-*)
    UNITREAD(W5, BUF, 512, 2, P08(DRIVE));
    IOERR := IORESULT;
    (*$I+*)
    ENTLEN := ORD(BUF[35]);
    ENTS := ORD(BUF[36])
  END;

  FUNCTION P19(VAR BUF: DIRBLOCK; VAR N, BLK, OFF, ENTLEN,
               ENTS: INTEGER): BOOLEAN;
  VAR IX: INTEGER;
  BEGIN
    P19 := TRUE;
    IF N = 0 THEN
      BEGIN
        P18(BUF, ENTLEN, ENTS);
        IF IORESULT <> 0 THEN
          EXIT(P19);
        BLK := 2
      END;
    N := N + 1;
    IX := N MOD ENTS;
    IF IX = 0 THEN
      BEGIN
        BLK := ORD(BUF[3]) * 256 + ORD(BUF[2]);
        IF BLK <> 0 THEN
          BEGIN
            (*$I-*)
            UNITREAD(W5, BUF, 512, BLK, P08(DRIVE));
            IOERR := IORESULT;
            IF IORESULT <> 0 THEN
              EXIT(P19)
            (*$I+*)
          END
        ELSE
          P19 := FALSE
      END;
    OFF := 4 + IX * ENTLEN
  END;

  FUNCTION P20(VAR BUF: DIRBLOCK; VAR N, BLK, OFF: INTEGER): INTEGER;
  VAR ENTLEN, ENTS: INTEGER;
      ENTRY: DIRENT;
      S: STRING[15];
  BEGIN
    P20 := 0;
    N := 0;
    WHILE P19(BUF, N, BLK, OFF, ENTLEN, ENTS) DO
      BEGIN
        IF IORESULT <> 0 THEN
          EXIT(P20);
        MOVELEFT(BUF[OFF], ENTRY, 40);
        P07(ENTRY.NAMELEN, ENTRY.FIRST, ENTRY.REST, S);
        IF S = 'PASCAL.AREA' THEN
          IF ENTRY.STORTYPE <> 4 THEN
            BEGIN
              ERRNUM := 17;
              EXIT(P20)
            END
          ELSE
            BEGIN
              P20 := ENTRY.KEYHI * 256 + ENTRY.KEYLO;
              EXIT(P20)
            END
      END;
    ERRNUM := 11
  END;

  FUNCTION P21(VAR BUF: DIRBLOCK; VAR N, BLK, OFF: INTEGER): BOOLEAN;
  VAR ENTLEN, ENTS: INTEGER;
      ENTRY: DIRENT;
      S: STRING[15];
  BEGIN
    P21 := FALSE;
    N := 0;
    WHILE P19(BUF, N, BLK, OFF, ENTLEN, ENTS) DO
      BEGIN
        IF IORESULT <> 0 THEN
          EXIT(P21);
        IF BUF[OFF] = 0 THEN
          BEGIN
            P21 := TRUE;
            EXIT(P21)
          END
      END
  END;

  PROCEDURE SPLIT(N: INTEGER; VAR QUOT, HI, LO: INTEGER);
  BEGIN
    QUOT := W913 + P04(N, 8 * 512);
    HI := P05(N, 8 * 512) DIV 8;
    LO := P05(N, 8 * 512) MOD 8
  END;

  FUNCTION ULESS(A, B: INTEGER): BOOLEAN;
  BEGIN
    IF A >= 0 THEN
      IF B >= 0 THEN
        ULESS := A < B
      ELSE
        ULESS := TRUE
    ELSE
      IF B >= 0 THEN
        ULESS := FALSE
      ELSE
        ULESS := A < B
  END;

  PROCEDURE SETBITS(HIBLK, LOBLK: INTEGER; FLAG: BOOLEAN);
  VAR LOQB, LOHB, LOLB: INTEGER;
      HIQB, HIHB, HILB: INTEGER;
      BUF: BITBUF;
      BIT: INTEGER;
      BOT: INTEGER;
      TOP: INTEGER;
  BEGIN
    IF ULESS(HIBLK, LOBLK) THEN
      EXIT(SETBITS);
    SPLIT(HIBLK, HIQB, HIHB, HILB);
    SPLIT(LOBLK, LOQB, LOHB, LOLB);
    TOP := HIHB * 8 + HILB;
    BOT := 0;
    REPEAT
      (*$I-*)
      UNITREAD(W5, BUF, 512, HIQB, P08(DRIVE));
      IF IORESULT <> 0 THEN
        EXIT(SETBITS);
      (*$I+*)
      IF HIQB = LOQB THEN
        BOT := LOHB * 8 + LOLB;
      FOR BIT := TOP DOWNTO BOT DO
        BUF[ORD(ODD(BIT) AND ODD(4088))
            + (7 - ORD(ODD(BIT) AND ODD(7)))] := FLAG;
      (*$I-*)
      UNITWRITE(W5, BUF, 512, HIQB, P08(DRIVE));
      IF IORESULT <> 0 THEN
        EXIT(SETBITS);
      (*$I+*)
      TOP := 4095;
      HIQB := HIQB - 1
    UNTIL ULESS(HIQB, LOQB)
  END;

  FUNCTION ALLBITS(HIBLK, LOBLK: INTEGER; VAR COUNT: INTEGER;
                   FLAG: BOOLEAN): BOOLEAN;
  VAR QUOTL, HIL, LOL: INTEGER;
      QUOTH, HIH, LOH: INTEGER;
      SPAN: INTEGER;
      BUF: BITBUF;
      BIT: INTEGER;
      BOT: INTEGER;
      TOP: INTEGER;
  BEGIN
    ALLBITS := TRUE;
    COUNT := 0;
    IF LOBLK = 0 THEN
      EXIT(ALLBITS);
    SPAN := HIBLK - LOBLK + 1;
    IF ULESS(HIBLK, SPAN) THEN
      BEGIN
        SPAN := 0;
        ALLBITS := FALSE
      END;
    SPLIT(HIBLK, QUOTH, HIH, LOH);
    SPLIT(SPAN, QUOTL, HIL, LOL);
    TOP := HIH * 8 + LOH;
    BOT := 0;
    REPEAT
      (*$I-*)
      UNITREAD(W5, BUF, 512, QUOTH, P08(DRIVE));
      IF IORESULT <> 0 THEN
        EXIT(ALLBITS);
      (*$I+*)
      IF QUOTH = QUOTL THEN
        BOT := HIL * 8 + LOL;
      FOR BIT := TOP DOWNTO BOT DO
        BEGIN
          IF BUF[ORD(ODD(BIT) AND ODD(4088))
                 + (7 - ORD(ODD(BIT) AND ODD(7)))] <> FLAG THEN
            BEGIN
              ALLBITS := FALSE;
              EXIT(ALLBITS)
            END;
          COUNT := COUNT + 1
        END;
      QUOTH := QUOTH - 1;
      TOP := 4095
    UNTIL ULESS(QUOTH, QUOTL)
  END;

  FUNCTION P51(N: INTEGER): BOOLEAN;
  BEGIN
    IF (N >= 1) AND (N <= DRVS[0].EXTRA) THEN
      P51 := FALSE
    ELSE
      BEGIN
        P51 := TRUE;
        ERRNUM := 1
      END
  END;

  FUNCTION P52(N: INTEGER): BOOLEAN;
  VAR SLOT, BLK, OFF: INTEGER;
  BEGIN
    P52 := FALSE;
    IF N IN UNITS THEN
      BEGIN
        (*$R-*)
        DRIVE := N;
        (*$R+*)
        PAKEY := P20(PABUF, SLOT, BLK, OFF);
        IF (ERRNUM <> 0) OR (IOERR <> 0) THEN
          EXIT(P52);
        (*$I-*)
        UNITREAD(W5, DRVS, W13, PAKEY, P08(N));
        IF IORESULT <> 0 THEN
          EXIT(P52);
        (*$I+*)
        IF DRVS[0].DNAME <> 'PPM' THEN
          EXIT(P52);
        P52 := TRUE;
        EXIT(P52)
      END
    ELSE
      EXIT(P52)
  END;

  PROCEDURE P53(N: INTEGER);
  BEGIN
    IF (N <= 16) OR ((N >= 33) AND (N <= 48)) THEN
      XPOS := 2
    ELSE
      XPOS := 42;
    IF N <= 16 THEN
      YPOS := N - 1
    ELSE
      IF N <= 32 THEN
        YPOS := N - 17
      ELSE
        IF N <= 48 THEN
          YPOS := N - 33
        ELSE
          IF N <= 64 THEN
            YPOS := N - 49;
    YPOS := YPOS + 7
  END;

  PROCEDURE V14;

    UNITWRITE(W5, BLK, 512, BLKNO, P08(DRIVE));
    (*$I+*)
    P72;
    P71 := TRUE
  END;

  FUNCTION P73(NEED: INTEGER; VAR GOT: INTEGER): BOOLEAN;
    PROCEDURE P74;
    BEGIN
      IF IOFAIL THEN
        EXIT(P73)
    END;
  BEGIN
    CLRERR;
    P73 := FALSE;
    IF ALLBITS(PAKEY - 1, NEED, GOT, TRUE) THEN
      BEGIN
        SETBITS(PAKEY - 1, PAKEY - NEED, FALSE);
        P74;
        IF P71(NEED) THEN
          P73 := TRUE
      END;
    P74
  END;

  FUNCTION P75(N: INTEGER): INTEGER;
    PROCEDURE P76;
    BEGIN
      IF IOFAIL THEN
        EXIT(P75)
    END;
  BEGIN
    CLRERR;
    P75 := 0;
    IF P71(-N) THEN
      SETBITS(PAKEY + N - 1, PAKEY, TRUE);
    P76;
    P75 := N
  END;

  PROCEDURE V07;
    { Last jump-table word is IC-JTAB; 1.1's self-rel is
      off by data_size 24. Instruction stream matches. }
    PROCEDURE P77;
    VAR
      ENDPT: INTEGER;
      BASE: INTEGER;
      LEN: INTEGER;
      MAP: INTEGER;
      GAP: INTEGER;
      I: INTEGER;
      U: INTEGER;
      BYTES: INTEGER;
      HEAP: ^INTEGER;
      NBLK: INTEGER;
      MARKP: ^INTEGER;
      LIM: INTEGER;
      PROCEDURE P78;
      VAR P: ^HEAPBLK;
      BEGIN
        NEW(HEAP);
        NBLK := 0;
        WHILE (MEMAVAIL > 556) AND (NBLK < 64) DO
          BEGIN
            NEW(P);
            NBLK := NBLK + 1
          END
      END;
    BEGIN
      MARK(MARKP);
      P78;
      IF NBLK > 0 THEN
        BEGIN
          BASE := PAKEY + DRVS[0].MAPLO;
          I := DRVS[0].EXTRA;
          LIM := 1;
          WHILE I >= LIM DO
            BEGIN
              MAP := DRVS[I].MAPLO;
              LEN := DRVS[I].EXTRA;
              ENDPT := MAP + LEN;
              GAP := BASE - ENDPT;
              IF GAP > 0 THEN
                BEGIN
                  BYTES := NBLK * 512;
                  ENDPT := ENDPT - NBLK;
                  IF ENDPT < MAP THEN
                    BEGIN
                      BYTES := BYTES - 512 * (MAP - ENDPT);
                      ENDPT := MAP
                    END;
                  (*$I-*)
                  REPEAT
                    UNITREAD(W5, HEAP^, BYTES, ENDPT, P08(DRIVE));
                    IF IOFAIL THEN
                      EXIT(V07);
                    UNITWRITE(W5, HEAP^, BYTES, ENDPT + GAP, P08(DRIVE));
                    IF IOFAIL THEN
                      EXIT(V07)
                  UNTIL ENDPT <= MAP;
                  BASE := MAP + GAP;
                  U := P66(MAP);
                  IF U <> 0 THEN
                    BEGIN
                      VOLS[P11(U)].B := BASE;
                      UNITSTATUS(W5, VOLS, 16386)
                    END;
                  DRVS[I].MAPLO := BASE;
                  UNITWRITE(W5, DRVS, W13, PAKEY, P08(DRIVE));
                  IF IOFAIL THEN
                    EXIT(V07)
                END
              ELSE
                BASE := MAP;
              I := I - 1
            END
        END;
      RELEASE(MARKP)
    END;
    PROCEDURE P79;
    VAR
      FRE: INTEGER;
      HI: INTEGER;
      LO: INTEGER;
    BEGIN
      WITH DRVS[0] DO
        IF EXTRA > 0 THEN
          LO := DRVS[1].MAPLO - 2
        ELSE
          LO := PAKEY + MAPLO - 2;
      HI := LO - PAKEY;
      FRE := P75(HI);
      IF IOFAIL THEN
        EXIT(V07);
      WITH DRVS[0] DO
        MAPLO := MAPLO - FRE;
      PAKEY := LO;
      (*$I-*)
      UNITWRITE(W5, DRVS, W13, PAKEY, P08(DRIVE));
      (*$I+*)
      IF IOFAIL THEN
        EXIT(V07)
    END;
  BEGIN
    CLRERR;
    P77;
    P79
  END;

  (*$I-*)
  PROCEDURE V03;
  VAR LAST, K, I: INTEGER;
  BEGIN
    CLRERR;
    IF N > DRVS[0].EXTRA THEN
      BEGIN
        ERRNUM := 1;
        EXIT(V03)
      END;
    IF DRVS[N].ATTRIB.WPROT THEN
      BEGIN
        ERRNUM := 10;
        EXIT(V03)
      END;
    UNITREAD(W5, PADIR, 25, DRVS[N].MAPLO + 2, P08(DRIVE));
    IF IOFAIL THEN
      EXIT(V03);
    IF PADIR[0].R2 <> 0 THEN
      BEGIN
        ERRNUM := 15;
        EXIT(V03)
      END;
    K := P66(DRVS[N].MAPLO);
    IF K <> 0 THEN
      BEGIN
        V05(K);
        IF IOFAIL THEN
          EXIT(V03)
      END;
    I := N;
    LAST := DRVS[0].EXTRA - 1;
    WHILE I <= LAST DO
      BEGIN
        DRVS[I] := DRVS[I + 1];
        VNAMES[I] := VNAMES[I + 1];
        DESCS[I] := DESCS[I + 1];
        I := I + 1
      END;
    DRVS[0].EXTRA := DRVS[0].EXTRA - 1;
    UNITWRITE(W5, DRVS, W13, PAKEY, P08(DRIVE));
    IF IOFAIL THEN
      EXIT(V03);
    IF FLAG THEN
      V07
  END;
  (*$I+*)

  FUNCTION V02;
  VAR
    SD: VOLDESC;
    SN: VOLNAME;
    GOT: INTEGER;
    LIM: INTEGER;
    PROCEDURE P80(NEED: INTEGER; VAR SLOT: INTEGER; VAR BEST: INTEGER);
    VAR ENDPT, BESTG, GAP: INTEGER;
    BEGIN
      SLOT := 0;
      GAP := 0;
      BESTG := 0;
      WHILE (SLOT <= DRVS[0].EXTRA) AND (GAP < NEED) DO
        BEGIN
          SLOT := SLOT + 1;
          IF SLOT > DRVS[0].EXTRA THEN
            BEGIN
              IF DRVS[0].EXTRA > 0 THEN
                BEGIN
                  ENDPT := DRVS[DRVS[0].EXTRA].MAPLO +
                           DRVS[DRVS[0].EXTRA].EXTRA;
                  GAP := (PAKEY + DRVS[0].MAPLO - 1) - ENDPT + 1
                END
              ELSE
                BEGIN
                  ENDPT := PAKEY + 2;
                  GAP := (PAKEY + DRVS[0].MAPLO - 1) - ENDPT + 1
                END
            END
          ELSE
            IF SLOT = 1 THEN
              BEGIN
                ENDPT := PAKEY + 2;
                GAP := DRVS[1].MAPLO - 1 - ENDPT + 1
              END
            ELSE
              BEGIN
                ENDPT := DRVS[SLOT - 1].MAPLO + DRVS[SLOT - 1].EXTRA;
                GAP := DRVS[SLOT].MAPLO - 1 - ENDPT + 1
              END;
          IF GAP > BESTG THEN
            BESTG := GAP
        END;
      IF GAP < NEED THEN
        BEGIN
          BEST := BESTG;
          SLOT := 0
        END
      ELSE
        BEST := ENDPT
    END;
    PROCEDURE P81;
    BEGIN
      FILLCHAR(PADIR, 2028, CHR(0));
      PADIR[0].W0 := 0;
      PADIR[0].W1 := 6;
      MOVELEFT(SN, PADIR[0].NM, 8);
      PADIR[0].R1 := N;
      PADIR[0].R2 := 0;
      PADIR[0].DAT.MONTH := THEDATE.MONTH;
      PADIR[0].DAT.DAY := THEDATE.DAY;
      PADIR[0].DAT.YEAR := THEDATE.YEAR;
      (*$I-*)
      UNITWRITE(W5, PADIR, 26, DRVS[W933].MAPLO + 2, P08(DRIVE));
      (*$I+*)
      IF IOFAIL THEN
        EXIT(V02)
    END;
  BEGIN
    SN := NM;
    SD := DS;
    CLRERR;
    V02 := 0;
    W933 := 0;
    IF N < 6 THEN
      BEGIN
        ERRNUM := 13;
        EXIT(V02)
      END;
    IF DRVS[0].EXTRA >= 31 THEN
      BEGIN
        ERRNUM := 3;
        EXIT(V02)
      END;
    IF NOT P16(SN) THEN
      BEGIN
        ERRNUM := 9;
        EXIT(V02)
      END;
    W933 := V10(SN, SD);
    IF ERRNUM = 1 THEN
      ERRNUM := 0;
    IF W933 <> 0 THEN
      IF DESCS[W933] = SD THEN
        BEGIN
          ERRNUM := 4;
          W933 := 0;
          EXIT(V02)
        END;
    P80(N, W933, W931);
    IF W933 = 0 THEN
      IF P73(N, GOT) THEN
        BEGIN
          DRVS[0].MAPLO := DRVS[0].MAPLO + N;
          PAKEY := PAKEY - N;
          (*$I-*)
          UNITWRITE(W5, DRVS, W13, PAKEY, P08(DRIVE));
          (*$I+*)
          IF IOFAIL THEN
            EXIT(V02);
          P80(N, W933, W931)
        END;
    IF W933 = 0 THEN
      BEGIN
        ERRNUM := 2;
        EXIT(V02)
      END;
    W932 := DRVS[0].EXTRA;
    LIM := W933;
    WHILE W932 >= LIM DO
      BEGIN
        DRVS[W932 + 1] := DRVS[W932];
        VNAMES[W932 + 1] := VNAMES[W932];
        DESCS[W932 + 1] := DESCS[W932];
        W932 := W932 - 1
      END;
    DRVS[W933].MAPLO := W931;
    DRVS[W933].EXTRA := N;
    DRVS[W933].ATTRIB.UNITNO := 0;
    DRVS[W933].ATTRIB.FILL7 := 0;
    DRVS[W933].ATTRIB.WPROT := FALSE;
    VNAMES[W933] := SN;
    DESCS[W933] := SD;
    DRVS[0].EXTRA := DRVS[0].EXTRA + 1;
    P81;
    (*$I-*)
    UNITWRITE(W5, DRVS, W13, PAKEY, P08(DRIVE));
    (*$I+*)
    IF IOFAIL THEN
      W933 := 0;
    V02 := W933
  END;

  (*$I-*)
