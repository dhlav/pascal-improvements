{ VMGR -- Pascal ProFile Manager, segment 7 of PPM.CODE.

  A plain unit, compiled on its own and USE'd by PASCALPR.
  Not a SEGMENT PROCEDURE: those nest at lex 2 and the
  program cannot call them (error 104). Every procedure
  here is lex 1, and procedure 1 returns with RNP, which
  is the unit initialisation UNITBODY builds (finding 232).

  Screen helpers (PUTCRT, HOMECURSOR, CLRSCRN, CLRLINE, LOADCRT) load
  SYSCOM as LOD 2,1. 1.3 does not predefine SYSCOM, so this
  file is compiled (*$U-*) with SYSCOM declared at level 0
  before the unit. (*$NS 7*) keeps the unit on segment 7.
  (*$R+*) and (*$I+*) put back the checks (*$U-*) cleared.

  SHOWERR's "Pascal System" error string ships with a high-ASCII
  space (byte 160) between the two words. Parked: the source
  keeps a normal space and we will change it later. Until
  then ppm/work/stage_vmgr.py can write the shipped byte;
  stagefile.py is ASCII-only.

  Three 1.3-vs-1.1 leftovers, same class as FILEHAND's six
  version bytes: TRIMEND WHILE AND is LAND then one FJP (1.1
  chained two FJPs); KEYVOL's match-arm UJP is +7 (1.1 omitted
  skip-else after GOTO, so +5); SLIDE's last jump-table word
  is IC-JTAB (1.1 self-rel is off by data_size 24). }
(*$U-*)
(*$NS 7*)
(*$R+*)
(*$I+*)

  PROCEDURE PUTCRT(BIT: INTEGER; CH: CHAR);
  BEGIN
    WITH SYSCOM^ DO
      IF CH <> CHR(0) THEN
        BEGIN
          IF CRTCTRL.PREFIXED[BIT] THEN
            WRITE(CRTCTRL.ESCAPE);
          WRITE(CH)
        END
  END;
  (*$R+*)

  PROCEDURE HOMECURSOR;
  BEGIN
    PUTCRT(4, SYSCOM^.CRTCTRL.HOME)
  END;

  PROCEDURE CLRSCRN;
  BEGIN
    HOMECURSOR;
    WITH SYSCOM^, CRTCTRL DO
      IF ERASEEOS <> CHR(0) THEN
        PUTCRT(3, ERASEEOS)
      ELSE
        PUTCRT(6, CLEARSCREEN)
  END;

  PROCEDURE CLRLINE;
  VAR I: INTEGER;
  BEGIN
    (*$I-*)
    WITH SYSCOM^, CRTCTRL DO
      IF ERASEEOL <> CHR(0) THEN
        PUTCRT(2, ERASEEOL)
      ELSE
        IF CLEARLINE <> CHR(0) THEN
          PUTCRT(7, CLEARLINE)
        ELSE
          IF (NOT MISCINFO.SLOWTERM) AND (RLF <> CHR(0)) THEN
            BEGIN
              FOR I := 2 TO CRTINFO.WIDTH DO
                WRITE(' ');
              WRITELN;
              PUTCRT(0, RLF)
            END
    (*$I+*)
  END;

  PROCEDURE LOADCRT;
  BEGIN
    WITH SYSCOM^ DO
      BEGIN
        SCRW := CRTINFO.WIDTH;
        SCRH := CRTINFO.HEIGHT;
        CHDEL := ORD(CRTINFO.CHARDEL);
        LNDEL := ORD(CRTINFO.LINEDEL);
        ETXCH := ORD(CRTINFO.ETX);
        ALTCH := ORD(CRTINFO.ALTMODE);
        SETB := [CHR(11), CHR(15)];
        SETA := [CHR(10), CHR(12)];
        RIGHTCH := ORD(CRTINFO.RIGHT);
        LEFTCH := ORD(CRTINFO.LEFT)
      END
  END;

  FUNCTION IOFAIL: BOOLEAN;
  BEGIN
    IOERR := IORESULT;
    IOFAIL := (IORESULT <> 0) OR (ERRNUM <> 0)
  END;

  FUNCTION SAFEDIV(DVND, DVSR: INTEGER): INTEGER;
  VAR T: INTEGER;
  BEGIN
    IF DVND >= 0 THEN
      SAFEDIV := DVND DIV DVSR
    ELSE
      BEGIN
        T := (DVND + 32767) + 1;
        SAFEDIV := ((T DIV DVSR) + (32767 DIV DVSR))
               + ((((T MOD DVSR) + (32767 MOD DVSR)) + 1) DIV DVSR)
      END
  END;

  FUNCTION SAFEMOD(DVND, DVSR: INTEGER): INTEGER;
  VAR T: INTEGER;
  BEGIN
    IF DVND >= 0 THEN
      SAFEMOD := DVND MOD DVSR
    ELSE
      BEGIN
        T := (DVND + 32767) + 1;
        SAFEMOD := (((T MOD DVSR) + (32767 MOD DVSR)) + 1) MOD DVSR
      END
  END;

  PROCEDURE CLRERR;
  BEGIN
    IOERR := 0;
    ERRNUM := 0
  END;

  PROCEDURE MKPDSTR(LEN: INTEGER; CH: CHAR; NAME: NAMEREST;
                VAR DEST: STRING);
  VAR I: INTEGER;
  BEGIN
    (*$R-*)
    DEST[0] := CHR(LEN);
    (*$R+*)
    IF LEN > 0 THEN
      BEGIN
        DEST[1] := CH;
        FOR I := 2 TO LEN DO
          DEST[I] := NAME[I]
      END
  END;

  FUNCTION DRVCTL(N: DRVNUM): INTEGER;
  BEGIN
    DRVCTL := 2 + N * 8192
  END;

  (*$I-*)
  (*$R-*)
  FUNCTION EDITLINE(VAR S: ANYSTR; LIM: INTEGER;
               DIGITS: BOOLEAN): BOOLEAN;
  VAR N: INTEGER;
      CH: CHAR;
  BEGIN
    N := 0;
    S := '';
    REPEAT
      READ(KEYBOARD, CH);
      IF (NOT EOLN(KEYBOARD)) AND (CH <> CHR(27)) THEN
        IF CH IN [CHR(8), CHR(127)] THEN
          IF N > 0 THEN
            BEGIN
              WRITE(CHR(8));
              WRITE(CHR(32));
              WRITE(CHR(8));
              N := N - 1
            END
          ELSE
            WRITE(CHR(7))
        ELSE
          IF CH IN [CHR(0)..CHR(31)] THEN
            WRITE(CHR(7))
          ELSE
            IF N >= LIM THEN
              WRITE(CHR(7))
            ELSE
              IF (NOT DIGITS) OR (CH IN ['0'..'9']) THEN
                BEGIN
                  N := N + 1;
                  S[N] := CH;
                  WRITE(CH)
                END
              ELSE
                WRITE(CHR(7))
    UNTIL EOLN(KEYBOARD) OR (CH = CHR(27));
    S[0] := CHR(N);
    EDITLINE := CH = CHR(27)
  END;
  (*$R+*)
  (*$I+*)

  FUNCTION READNUM(VAR N: INTEGER): BOOLEAN;
  VAR BUF: STRING[5];
      ESC: BOOLEAN;
      I: INTEGER;
  BEGIN
    ESC := EDITLINE(BUF, 5, TRUE);
    READNUM := ESC;
    N := 0;
    IF NOT ESC THEN
      FOR I := 1 TO LENGTH(BUF) DO
        N := N * 10 + ORD(BUF[I]) - ORD('0')
  END;

  FUNCTION VOLIDX(N: INTEGER): INTEGER;
  BEGIN
    IF N IN [4, 5] THEN
      VOLIDX := N - 3
    ELSE
      IF N IN [9..20] THEN
        VOLIDX := N - 6
      ELSE
        VOLIDX := (N - 128) + 15
  END;

  FUNCTION UNITNUM(N: INTEGER): INTEGER;
  BEGIN
    IF (N = 1) OR (N = 2) THEN
      UNITNUM := N + 3
    ELSE
      IF N IN [3..14] THEN
        UNITNUM := N + 6
      ELSE
        UNITNUM := (N + 128) - 15
  END;

  (*$I-*)
  PROCEDURE BLANKLN(ROW, NLINES: INTEGER);
  VAR I: INTEGER;
  BEGIN
    GOTOXY(0, ROW);
    FOR I := 1 TO NLINES DO
      BEGIN
        CLRLINE;
        WRITELN
      END;
    GOTOXY(0, ROW)
  END;

  FUNCTION YESNO: BOOLEAN;
  VAR CH: CHAR;
      OK: BOOLEAN;
  BEGIN
    UNITCLEAR(2);
    RESET(KEYBOARD);
    REPEAT
      READ(KEYBOARD, CH);
      OK := CH IN [CHR(27), 'N', 'Y', 'n', 'y'];
      IF NOT OK THEN
        WRITE(CHR(7))
    UNTIL OK;
    IF CH <> CHR(27) THEN
      WRITE(CH);
    YESNO := CH IN ['Y', 'y']
  END;
  (*$I+*)

  PROCEDURE CLEANNM(VAR S: STRING);
  VAR AT: INTEGER;
      I: INTEGER;
  BEGIN
    FOR I := 1 TO LENGTH(S) DO
      IF S[I] IN ['a'..'z'] THEN
        S[I] := CHR(ORD(S[I]) - 32);
    IF POS(' ', S) <> 0 THEN
      REPEAT
        AT := POS(' ', S);
        DELETE(S, AT, 1)
      UNTIL POS(' ', S) = 0
  END;

  FUNCTION PVVALID(NAME: ANYSTR): BOOLEAN;
  VAR I: INTEGER;
  BEGIN
    PVVALID := FALSE;
    IF LENGTH(NAME) IN [1..7] THEN
      BEGIN
        FOR I := 1 TO LENGTH(NAME) DO
          IF (NOT (NAME[I] IN [' '..'z']))
             OR (NAME[I] IN [' ', '$', ',', '=', '?']) THEN
            EXIT(PVVALID);
        PVVALID := TRUE
      END
  END;

  FUNCTION PDVALID(NAME: STRING): BOOLEAN;
  VAR I: INTEGER;
  BEGIN
    PDVALID := TRUE;
    IF LENGTH(NAME) IN [1..15] THEN
      BEGIN
        IF NOT (NAME[1] IN ['A'..'Z']) THEN
          PDVALID := FALSE;
        FOR I := 2 TO LENGTH(NAME) DO
          IF NOT (NAME[I] IN ['.', '0'..'9', 'A'..'Z']) THEN
            PDVALID := FALSE
      END
    ELSE
      PDVALID := FALSE
  END;

  PROCEDURE READDIR(VAR BUF: DIRBLOCK; VAR ENTLEN, ENTS: INTEGER);
  BEGIN
    (*$I-*)
    UNITREAD(PFUNIT, BUF, 512, 2, DRVCTL(DRIVE));
    IOERR := IORESULT;
    (*$I+*)
    ENTLEN := ORD(BUF[35]);
    ENTS := ORD(BUF[36])
  END;

  FUNCTION NEXTENT(VAR BUF: DIRBLOCK; VAR N, BLK, OFF, ENTLEN,
               ENTS: INTEGER): BOOLEAN;
  VAR IX: INTEGER;
  BEGIN
    NEXTENT := TRUE;
    IF N = 0 THEN
      BEGIN
        READDIR(BUF, ENTLEN, ENTS);
        IF IORESULT <> 0 THEN
          EXIT(NEXTENT);
        BLK := 2
      END;
    N := N + 1;
    IX := N MOD ENTS;
    IF IX = 0 THEN
      BEGIN
        BLK := ORD(BUF[3]) * 256 + ORD(BUF[2]);
        IF BLK <> 0 THEN
          BEGIN
            (*$I-*)
            UNITREAD(PFUNIT, BUF, 512, BLK, DRVCTL(DRIVE));
            IOERR := IORESULT;
            IF IORESULT <> 0 THEN
              EXIT(NEXTENT)
            (*$I+*)
          END
        ELSE
          NEXTENT := FALSE
      END;
    OFF := 4 + IX * ENTLEN
  END;

  FUNCTION FINDPAS(VAR BUF: DIRBLOCK; VAR N, BLK, OFF: INTEGER): INTEGER;
  VAR ENTLEN, ENTS: INTEGER;
      ENTRY: DIRENT;
      S: STRING[15];
  BEGIN
    FINDPAS := 0;
    N := 0;
    WHILE NEXTENT(BUF, N, BLK, OFF, ENTLEN, ENTS) DO
      BEGIN
        IF IORESULT <> 0 THEN
          EXIT(FINDPAS);
        MOVELEFT(BUF[OFF], ENTRY, 40);
        MKPDSTR(ENTRY.NAMELEN, ENTRY.FIRST, ENTRY.REST, S);
        IF S = 'PASCAL.AREA' THEN
          IF ENTRY.STORTYPE <> 4 THEN
            BEGIN
              ERRNUM := 17;
              EXIT(FINDPAS)
            END
          ELSE
            BEGIN
              FINDPAS := ENTRY.KEYHI * 256 + ENTRY.KEYLO;
              EXIT(FINDPAS)
            END
      END;
    ERRNUM := 11
  END;

  FUNCTION FREESLOT(VAR BUF: DIRBLOCK; VAR N, BLK, OFF: INTEGER): BOOLEAN;
  VAR ENTLEN, ENTS: INTEGER;
      ENTRY: DIRENT;
      S: STRING[15];
  BEGIN
    FREESLOT := FALSE;
    N := 0;
    WHILE NEXTENT(BUF, N, BLK, OFF, ENTLEN, ENTS) DO
      BEGIN
        IF IORESULT <> 0 THEN
          EXIT(FREESLOT);
        IF BUF[OFF] = 0 THEN
          BEGIN
            FREESLOT := TRUE;
            EXIT(FREESLOT)
          END
      END
  END;

  PROCEDURE SPLIT(N: INTEGER; VAR QUOT, HI, LO: INTEGER);
  BEGIN
    QUOT := DRVMAP + SAFEDIV(N, 8 * 512);
    HI := SAFEMOD(N, 8 * 512) DIV 8;
    LO := SAFEMOD(N, 8 * 512) MOD 8
  END;

  FUNCTION ULESS(A, B: INTEGER): BOOLEAN;
  BEGIN
    IF A >= 0 THEN
      IF B >= 0 THEN
        ULESS := A < B
      ELSE
        ULESS := TRUE
    ELSE
      IF B >= 0 THEN
        ULESS := FALSE
      ELSE
        ULESS := A < B
  END;

  PROCEDURE SETBITS(HIBLK, LOBLK: INTEGER; FLAG: BOOLEAN);
  VAR LOQB, LOHB, LOLB: INTEGER;
      HIQB, HIHB, HILB: INTEGER;
      BUF: BITBUF;
      BIT: INTEGER;
      BOT: INTEGER;
      TOP: INTEGER;
  BEGIN
    IF ULESS(HIBLK, LOBLK) THEN
      EXIT(SETBITS);
    SPLIT(HIBLK, HIQB, HIHB, HILB);
    SPLIT(LOBLK, LOQB, LOHB, LOLB);
    TOP := HIHB * 8 + HILB;
    BOT := 0;
    REPEAT
      (*$I-*)
      UNITREAD(PFUNIT, BUF, 512, HIQB, DRVCTL(DRIVE));
      IF IORESULT <> 0 THEN
        EXIT(SETBITS);
      (*$I+*)
      IF HIQB = LOQB THEN
        BOT := LOHB * 8 + LOLB;
      FOR BIT := TOP DOWNTO BOT DO
        BUF[ORD(ODD(BIT) AND ODD(4088))
            + (7 - ORD(ODD(BIT) AND ODD(7)))] := FLAG;
      (*$I-*)
      UNITWRITE(PFUNIT, BUF, 512, HIQB, DRVCTL(DRIVE));
      IF IORESULT <> 0 THEN
        EXIT(SETBITS);
      (*$I+*)
      TOP := 4095;
      HIQB := HIQB - 1
    UNTIL ULESS(HIQB, LOQB)
  END;

  FUNCTION ALLBITS(HIBLK, LOBLK: INTEGER; VAR COUNT: INTEGER;
                   FLAG: BOOLEAN): BOOLEAN;
  VAR QUOTL, HIL, LOL: INTEGER;
      QUOTH, HIH, LOH: INTEGER;
      SPAN: INTEGER;
      BUF: BITBUF;
      BIT: INTEGER;
      BOT: INTEGER;
      TOP: INTEGER;
  BEGIN
    ALLBITS := TRUE;
    COUNT := 0;
    IF LOBLK = 0 THEN
      EXIT(ALLBITS);
    SPAN := HIBLK - LOBLK + 1;
    IF ULESS(HIBLK, SPAN) THEN
      BEGIN
        SPAN := 0;
        ALLBITS := FALSE
      END;
    SPLIT(HIBLK, QUOTH, HIH, LOH);
    SPLIT(SPAN, QUOTL, HIL, LOL);
    TOP := HIH * 8 + LOH;
    BOT := 0;
    REPEAT
      (*$I-*)
      UNITREAD(PFUNIT, BUF, 512, QUOTH, DRVCTL(DRIVE));
      IF IORESULT <> 0 THEN
        EXIT(ALLBITS);
      (*$I+*)
      IF QUOTH = QUOTL THEN
        BOT := HIL * 8 + LOL;
      FOR BIT := TOP DOWNTO BOT DO
        BEGIN
          IF BUF[ORD(ODD(BIT) AND ODD(4088))
                 + (7 - ORD(ODD(BIT) AND ODD(7)))] <> FLAG THEN
            BEGIN
              ALLBITS := FALSE;
              EXIT(ALLBITS)
            END;
          COUNT := COUNT + 1
        END;
      QUOTH := QUOTH - 1;
      TOP := 4095
    UNTIL ULESS(QUOTH, QUOTL)
  END;

  FUNCTION BADVOL(N: INTEGER): BOOLEAN;
  BEGIN
    IF (N >= 1) AND (N <= DRVS[0].EXTRA) THEN
      BADVOL := FALSE
    ELSE
      BEGIN
        BADVOL := TRUE;
        ERRNUM := 1
      END
  END;

  FUNCTION HASPPM(N: INTEGER): BOOLEAN;
  VAR SLOT, BLK, OFF: INTEGER;
  BEGIN
    HASPPM := FALSE;
    IF N IN UNITS THEN
      BEGIN
        (*$R-*)
        DRIVE := N;
        (*$R+*)
        PAKEY := FINDPAS(PABUF, SLOT, BLK, OFF);
        IF (ERRNUM <> 0) OR (IOERR <> 0) THEN
          EXIT(HASPPM);
        (*$I-*)
        UNITREAD(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(N));
        IF IORESULT <> 0 THEN
          EXIT(HASPPM);
        (*$I+*)
        IF DRVS[0].DNAME <> 'PPM' THEN
          EXIT(HASPPM);
        HASPPM := TRUE;
        EXIT(HASPPM)
      END
    ELSE
      EXIT(HASPPM)
  END;

  PROCEDURE VOLCELL(N: INTEGER);
  BEGIN
    IF (N <= 16) OR ((N >= 33) AND (N <= 48)) THEN
      XPOS := 2
    ELSE
      XPOS := 42;
    IF N <= 16 THEN
      YPOS := N - 1
    ELSE
      IF N <= 32 THEN
        YPOS := N - 17
      ELSE
        IF N <= 48 THEN
          YPOS := N - 33
        ELSE
          IF N <= 64 THEN
            YPOS := N - 49;
    YPOS := YPOS + 7
  END;

  FUNCTION CHOOSE;
  VAR CH: CHAR;
    PROCEDURE INVOFF;
    BEGIN
      WRITE(CHR(14))
    END;
    PROCEDURE INVON;
    BEGIN
      WRITE(CHR(15))
    END;
    PROCEDURE SHOWMARK;
    BEGIN
      VOLCELL(HILITE);
      GOTOXY(XPOS-2, YPOS);
      WRITE('->')
    END;
    PROCEDURE HIDEMARK;
    BEGIN
      VOLCELL(HILITE);
      GOTOXY(XPOS-2, YPOS);
      WRITE('  ')
    END;
  BEGIN
    CLRERR;
    CHOOSE := 0;
    IF DRVS[0].EXTRA = 0 THEN
      EXIT(CHOOSE);
    IF HILITE <> 0 THEN
      BEGIN
        HIDEMARK;
        INVOFF;
        DRAWROW(HILITE)
      END;
    HILITE := 1;
    CH := ' ';
    SHOWMARK;
    REPEAT
      READ(KEYBOARD, CH);
      IF CH IN SETA THEN
        BEGIN
          HIDEMARK;
          HILITE := HILITE + 1;
          IF HILITE > DRVS[0].EXTRA THEN
            HILITE := 1;
          SHOWMARK
        END
      ELSE
        IF CH IN SETB THEN
          BEGIN
            HIDEMARK;
            HILITE := HILITE - 1;
            IF HILITE < 1 THEN
              HILITE := DRVS[0].EXTRA;
            SHOWMARK
          END
        ELSE
          IF CH = CHR(ALTCH) THEN
            BEGIN
              HIDEMARK;
              HILITE := 0;
              CHOOSE := 0;
              EXIT(CHOOSE)
            END
          ELSE
            IF CH = CHR(RIGHTCH) THEN
              BEGIN
                INVON;
                DRAWROW(HILITE);
                INVOFF;
                CHOOSE := HILITE
              END
            ELSE
              WRITE(CHR(7))
    UNTIL CH = CHR(RIGHTCH)
  END;

  PROCEDURE SHOWERR;
  BEGIN
    IF ODD(ERRSW) THEN
      BEGIN
        IF ODD(ERRHERE) THEN
          GOTOXY(0, CURROW);
        CLRLINE;
        WRITE(CHR(7));
        IF ERRNUM <> 0 THEN
          BEGIN
            WRITE('ERROR: ');
            CASE ERRNUM OF
              1: BEGIN
                   WRITE('This Pascal volume cannot be found');

                 END;

                   WRITELN
                 END;

                   WRITE('The Pascal area volume directory is full');
                   WRITELN
                 END;
              4: BEGIN
                   WRITE('This volume name conflicts with another volume name');
                   WRITELN
                 END;
              5: BEGIN
                   WRITE('Illegal unit number [should be 4, 5, 9 - 20, 128 - 143]');
                   WRITELN
                 END;
              6: BEGIN
                   WRITE('This volume has not been assigned a unit number');
                   WRITELN
                 END;
              7: BEGIN
                   WRITE('This ProFile drive is not available for use');
                   WRITELN
                 END;
              8: BEGIN
                   WRITE('Illegal ProFile drive number [should be 0 - 7]');
                   WRITELN
                 END;
              9: BEGIN
                   WRITE('Illegal volume name');
                   WRITELN
                 END;
              10: BEGIN
                    WRITE('This volume is write-protected - action cannot occur');
                    WRITELN
                  END;
              11: BEGIN
                    WRITE('This ProFile does not have a Pascal area');
                    WRITELN
                  END;
              12: BEGIN
                    { Parked: shipped LSA has CHR(160) after Pascal. }

                    WRITE('Volume size must be greater than 6 blocks and less than 32768');

                    WRITE('ProDOS directory is full, cannot create Pascal area');

                    WRITE('The ProDOS directory has a ProDOS file named PASCAL.AREA');

                  END
            END
          END
        ELSE
          IF IOERR <> 0 THEN
            BEGIN
              WRITE('ERROR: I/O error #');
              WRITE(IOERR);
              WRITE(' has occurred');
              WRITELN
            END
      END
  END;

  PROCEDURE TRIMEND(VAR S: STRING);
  VAR I, N: INTEGER;
  BEGIN
    CLEANNM(S);
    (*$R-*)
    N := ORD(S[0]);
    (*$R+*)
    I := 1;
    { 1.3 LAND+FJP; 1.1 chained two FJPs. Same length. }
    WHILE (N >= I) AND ((S[N] = ' ') OR (S[N] = ':')) DO
      BEGIN
        DELETE(S, N, 1);
        N := N - 1
      END
  END;

  PROCEDURE RDVOLNM(VAR TBL: DRVTABLE; N: INTEGER;
                VAR NM: VOLNAME; DRV: DRVNUM);
  BEGIN
    UNITREAD(PFUNIT, PABUF, 25, TBL[N].MAPLO + 2, DRVCTL(DRV));
    IF IORESULT = 0 THEN
      BEGIN
        MOVELEFT(PABUF[6], NM, 8);
        IF PVVALID(NM) THEN
          CLEANNM(NM)
        ELSE
          NM := '<none>'
      END
    ELSE
      NM := 'I/O Err'
  END;
  PROCEDURE RDNAMES(VAR TBL: DRVTABLE; VAR NAMES: NAMETBL; DRV: DRVNUM);
  VAR LAST, I: INTEGER;
  BEGIN
    I := 1;
    LAST := TBL[0].EXTRA;
    WHILE I <= LAST DO
      BEGIN
        RDVOLNM(TBL, I, NAMES[I], DRV);
        IF IORESULT <> 0 THEN
          IOERR := IORESULT;
        I := I + 1
      END
  END;
  PROCEDURE WRVOLNM(NM: VOLNAME; N: INTEGER);
  BEGIN
    UNITREAD(PFUNIT, PADIR, 2028, DRVS[N].MAPLO + 2, DRVCTL(DRIVE));
    IF IORESULT = 0 THEN
      BEGIN
        MOVELEFT(NM, PADIR[0].NM, 8);
        UNITWRITE(PFUNIT, PADIR, 2028, DRVS[N].MAPLO + 2, DRVCTL(DRIVE));
        IF IORESULT <> 0 THEN
          BEGIN
            IOERR := IORESULT;
            EXIT(WRVOLNM)
          END
        ELSE
          VNAMES[N] := NM
      END
    ELSE
      IOERR := IORESULT
  END;

  FUNCTION FINDBYNM;
  VAR I: INTEGER;
      N: INTEGER;
      LAST: INTEGER;
  BEGIN
    CLRERR;
    N := 0;
    I := 1;
    LAST := DRVS[0].EXTRA;
    WHILE I <= LAST DO
      BEGIN
        IF VNAMES[I] = NM THEN
          BEGIN
            N := N + 1;
            FINDBYNM := I;
            IF DESCS[I] = DS THEN
              EXIT(FINDBYNM)
          END;
        I := I + 1
      END;
    IF N <> 1 THEN
      BEGIN
        ERRNUM := 1;
        FINDBYNM := 0
      END
  END;

  PROCEDURE RENAVOL;
  VAR
    SD: VOLDESC;
    SN: VOLNAME;
    HIT: INTEGER;
  BEGIN
    SN := NM;
    SD := DS;
    CLRERR;
    IF BADVOL(N) THEN
      EXIT(RENAVOL);
    IF DRVS[N].ATTRIB.WPROT THEN
      BEGIN
        ERRNUM := 10;
        EXIT(RENAVOL)
      END;
    IF SN = '' THEN
      IF SD = '' THEN
        EXIT(RENAVOL)
      ELSE
        BEGIN
          HIT := FINDBYNM(VNAMES[N], SD);
          IF ERRNUM = 1 THEN
            ERRNUM := 0;
          IF (HIT <> 0) AND (HIT <> N) THEN
            BEGIN
              ERRNUM := 4;
              EXIT(RENAVOL)
            END;
          DESCS[N] := SD;
          UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
          IOERR := IORESULT;
          EXIT(RENAVOL)
        END
    ELSE
      IF PVVALID(SN) THEN
        BEGIN
          IF SD = '' THEN
            HIT := FINDBYNM(SN, DESCS[N])
          ELSE
            HIT := FINDBYNM(SN, SD);
          IF ERRNUM = 1 THEN
            ERRNUM := 0;
          IF HIT <> 0 THEN
            BEGIN
              ERRNUM := 4;
              EXIT(RENAVOL)
            END;
          WRVOLNM(SN, N);
          IF IOFAIL THEN
            EXIT(RENAVOL);
          IF SD <> '' THEN
            BEGIN
              DESCS[N] := SD;
              UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
              IOERR := IORESULT;
              EXIT(RENAVOL)
            END
        END
      ELSE
        BEGIN
          ERRNUM := 9;
          EXIT(RENAVOL)
        END
  END;

  PROCEDURE RDUNIT(N: INTEGER; VAR DEST: INTEGER);
  VAR
    T: RECORD
         CASE BOOLEAN OF
           FALSE: (I: INTEGER);
           TRUE: (P: ^IAP)
       END;
    B: RECORD
         CASE BOOLEAN OF
           FALSE: (I: INTEGER);
           TRUE: (P: ^BAP)
       END;
  BEGIN
    IF N IN [4, 5, 9..20] THEN
      BEGIN
        T.I := 234;
        DEST := T.P^^[N] + 1
      END
    ELSE
      IF N IN [128..143] THEN
        BEGIN
          B.I := 232;
          MOVELEFT(B.P^^[3 * (N - 128) + 1], DEST, 2)
        END
  END;
  PROCEDURE WRUNIT(N: INTEGER; SRC: INTEGER);
  VAR
    T: RECORD
         CASE BOOLEAN OF
           FALSE: (I: INTEGER);
           TRUE: (P: ^IAP)
       END;
    B: RECORD
         CASE BOOLEAN OF
           FALSE: (I: INTEGER);
           TRUE: (P: ^BAP)
       END;
  BEGIN
    IF N IN [4, 5, 9..20] THEN
      BEGIN
        T.I := 234;
        T.P^^[N] := SRC - 1
      END
    ELSE
      IF N IN [128..143] THEN
        BEGIN
          B.I := 232;
          MOVELEFT(SRC, B.P^^[3 * (N - 128) + 1], 2)
        END
  END;

  PROCEDURE INITDRV;
  VAR
    I: INTEGER;
    U: INTEGER;
    VID: INTEGER;
    PUN: RECORD
           CASE BOOLEAN OF
             FALSE: (A: INTEGER);
             TRUE: (Q: ^NAME7)
         END;
    PROF: NAME7;
    NM: STRING;
    PROCEDURE COPYNM(VAR SRC: NAME7; N: INTEGER; VAR DEST: NAME7);
    BEGIN
      MOVELEFT(SRC, DEST, N)
    END;
    PROCEDURE GETVNAME(DRV: INTEGER; VAR NM: STRING);
    VAR IMG: RECORD
               CASE INTEGER OF
                 0: (BLK: DIRBLOCK);
                 1: (SKIP: PACKED ARRAY [0..3] OF CHAR;
                     VOL: STRING)
             END;
    BEGIN
      NM := '<no name>';
      (*$I-*)
      UNITREAD(PFUNIT, IMG.BLK, 512, 2, DRVCTL(DRV));
      (*$I+*)
      IF IOFAIL THEN
        EXIT(GETVNAME);
      (*$R-*)
      IF ORD(IMG.VOL[0]) DIV 16 = 15 THEN
        BEGIN
          IMG.VOL[0] := CHR(ORD(IMG.VOL[0]) MOD 16);
          IF PDVALID(IMG.VOL) THEN
            NM := IMG.VOL
        END
      (*$R+*)
    END;
  BEGIN
    CLRERR;
    TBLLEN := 256 + 512;
    CURROW := 3;
    ERRHERE := 0;
    ERRSW := 1;
    LOADCRT;
    HILITE := 0;
    DRVADR := 0;
    U := 128;
    REPEAT
      RDUNIT(U, VID);
      PUN.A := VID + 3;
      COPYNM(PUN.Q^, 7, PROF);
      IF PROF = 'PROFILE' THEN
        BEGIN
          DRVADR := VID;
          PFUNIT := U
        END
      ELSE
        U := U + 1
    UNTIL (DRVADR <> 0) OR (U > 143);
    IF DRVADR = 0 THEN
      BEGIN
        ERRNUM := 12;
        EXIT(INITDRV)
      END;
    (*$I-*)
    UNITSTATUS(U, VOLS, 16384);
    (*$I+*)
    IF IOFAIL THEN
      BEGIN
        ERRNUM := 12;
        EXIT(INITDRV)
      END;
    UNITS := [];
    FOR I := 0 TO 7 DO
      BEGIN
        (*$I-*)
        UNITREAD(U, PABUF, 2, 2, DRVCTL(I));
        (*$I+*)
        IF IORESULT = 0 THEN
          UNITS := UNITS + [I]
      END;
    (*$I-*)
    WRITE('');
    (*$I+*)
    MARKED := [];
    FOR I := 0 TO 7 DO
      IF I IN UNITS THEN
        BEGIN
          GETVNAME(I, NM);
          IF NM = '<no name>' THEN
            UNITS := UNITS - [I]
          ELSE
            IF HASPPM(I) THEN
              MARKED := MARKED + [I];
          ERRNUM := 0
        END;
    DRIVE := VOLS[1].STAT;
    IF NOT (DRIVE IN UNITS) THEN
      BEGIN
        I := 0;
        WHILE I <= 7 DO
          IF I IN UNITS THEN
            BEGIN
              (*$R-*)
              DRIVE := I;
              (*$R+*)
              VOLS[1].STAT := DRIVE;
              (*$I-*)
              UNITSTATUS(U, VOLS, 16386);
              (*$I+*)
              I := 32767
            END
          ELSE
            I := I + 1
      END
  END;

  { 1.3 match-arm UJP is +7; 1.1 omitted skip-else after
    GOTO and used +5. Same length, one displacement byte. }
  FUNCTION KEYVOL(KEY: INTEGER): INTEGER;
  LABEL 1, 2;
  VAR FOUND, I: INTEGER;
  BEGIN
    FOUND := 0;
    I := 1;
    REPEAT
      WITH VOLS[I] DO
        IF BUSY THEN
          BEGIN
            IF (DV = DRIVE) AND (B = KEY) THEN
              BEGIN
                FOUND := UNITNUM(I);
                GOTO 2
              END;
            I := I + 1;
            GOTO 1
          END;
    2:
      I := I + 1;
    1:
    UNTIL (FOUND <> 0) OR (I > 30);
    KEYVOL := FOUND
  END;

  (*$I-*)
  PROCEDURE MOUNTDRV;
  VAR
    K, U: INTEGER;
    DIRTY, I: INTEGER;
    PAD: ARRAY [1..4] OF INTEGER;
    SLOT, BLK, OFF: INTEGER;
    LAST: INTEGER;
  BEGIN
    CLRERR;
    IF NOT (N IN [0..7]) THEN
      BEGIN
        ERRNUM := 8;
        EXIT(MOUNTDRV)
      END;
    IF NOT (N IN MARKED) THEN
      BEGIN
        ERRNUM := 7;
        EXIT(MOUNTDRV)
      END;
    (*$R-*)
    DRIVE := N;
    (*$R+*)
    HILITE := 0;
    VOLS[1].STAT := DRIVE;
    UNITSTATUS(PFUNIT, VOLS, 16386);
    PAKEY := FINDPAS(PABUF, SLOT, BLK, OFF);
    IF IOFAIL THEN
      EXIT(MOUNTDRV);
    UNITREAD(PFUNIT, PABUF, 512, 2, DRVCTL(N));
    IF IOFAIL THEN
      EXIT(MOUNTDRV);
    DRVMAP := 256 * PABUF[40] + PABUF[39];
    UNITREAD(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(N));
    IF IOFAIL THEN
      EXIT(MOUNTDRV);
    RDNAMES(DRVS, VNAMES, DRIVE);
    DIRTY := 0;
    I := 1;
    LAST := DRVS[0].EXTRA;
    WHILE I <= LAST DO
      BEGIN
        U := DRVS[I].ATTRIB.UNITNO;
        IF U IN [4, 5, 9..20, 128..143] THEN
          WITH VOLS[VOLIDX(U)] DO
            IF (NOT BUSY) OR (DV <> DRIVE) OR (B <> DRVS[I].MAPLO) THEN
              BEGIN
                DRVS[I].ATTRIB.UNITNO := 0;
                DIRTY := 1
              END;
        K := KEYVOL(DRVS[I].MAPLO);
        IF K <> 0 THEN
          IF K <> U THEN
            BEGIN
              DRVS[I].ATTRIB.UNITNO := K;
              DRVS[I].AUX := DRVADR;
              DIRTY := 1
            END;
        I := I + 1
      END;
    IF ODD(DIRTY) THEN
      BEGIN
        UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(N));
        IF IOFAIL THEN
          EXIT(MOUNTDRV)
      END
  END;

  FUNCTION BYUNIT(U: INTEGER): INTEGER;
  VAR LAST, I: INTEGER;
  BEGIN
    I := 1;
    LAST := DRVS[0].EXTRA;
    WHILE I <= LAST DO
      BEGIN
        IF DRVS[I].ATTRIB.UNITNO = U THEN
          BEGIN
            BYUNIT := I;
            EXIT(BYUNIT)
          END;
        I := I + 1
      END;
    BYUNIT := -1
  END;

  (*$I-*)
  PROCEDURE FREEUNIT;
  VAR V, DRV: INTEGER;
    PROCEDURE FIXREL;
    BEGIN
      IF N = 12 THEN
        IF VOLS[VOLIDX(4)].BUSY THEN
          BEGIN
            RDUNIT(12, SAVU12);
            WRUNIT(12, DRVS[BYUNIT(4)].AUX);
            UNITSTATUS(PFUNIT, VOLS, 16386)
          END;
      IF N = 4 THEN
        IF NOT VOLS[VOLIDX(12)].BUSY THEN
          WRUNIT(12, SAVU12)
    END;
  BEGIN
    CLRERR;
    IF NOT (N IN [4, 5, 9..20, 128..143]) THEN
      BEGIN
        ERRNUM := 5;
        EXIT(FREEUNIT)
      END;
    DRV := BYUNIT(N);
    V := VOLIDX(N);
    IF VOLS[V].BUSY THEN
      BEGIN
        VOLS[V].BUSY := FALSE;
        VOLS[V].B := 0;
        VOLS[V].A := 0
      END
    ELSE
      BEGIN
        ERRNUM := 6;
        EXIT(FREEUNIT)
      END;
    WRUNIT(N, DRVS[DRV].AUX);
    DRVS[DRV].ATTRIB.UNITNO := 0;
    DRVS[DRV].AUX := 0;
    FIXREL;
    UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
    IF IOFAIL THEN
      EXIT(FREEUNIT);
    UNITSTATUS(PFUNIT, VOLS, 16386)
  END;

  PROCEDURE BINDUNIT;
  LABEL 99;
  VAR V, T: INTEGER;
    PROCEDURE FIXASN;
    BEGIN
      IF B = 12 THEN
        IF VOLS[VOLIDX(4)].BUSY THEN
          DRVS[BYUNIT(12)].AUX := SAVU12;
      IF B = 4 THEN
        IF NOT VOLS[VOLIDX(12)].BUSY THEN
          BEGIN
            RDUNIT(12, SAVU12);
            WRUNIT(12, DRVS[BYUNIT(4)].AUX);
            UNITSTATUS(PFUNIT, VOLS, 16386)
          END
    END;
    FUNCTION CLEARASN: BOOLEAN;
    VAR OK: BOOLEAN;
        OLDDRV: INTEGER;
        V: INTEGER;
    BEGIN
      OK := TRUE;
      OLDDRV := DRIVE;
      V := VOLIDX(B);
      IF VOLS[V].BUSY THEN
        BEGIN
          IF VOLS[V].DV <> DRIVE THEN
            BEGIN
              OLDDRV := DRIVE;
              MOUNTDRV(VOLS[V].DV);
              IF IOFAIL THEN
                EXIT(BINDUNIT)
            END;
          FREEUNIT(B);
          OK := ERRNUM = 0;
          IF OLDDRV <> DRIVE THEN
            BEGIN
              MOUNTDRV(OLDDRV);
              IF IOFAIL THEN
                EXIT(BINDUNIT)
            END
        END;
      IF OK THEN
        IF DRVS[A].ATTRIB.UNITNO IN [4, 5, 9..20, 128..143] THEN
          BEGIN
            FREEUNIT(DRVS[A].ATTRIB.UNITNO);
            OK := ERRNUM = 0
          END;
      CLEARASN := OK
    END;
  BEGIN
    CLRERR;
    IF NOT (B IN [4, 5, 9..20, 128..143]) THEN
      BEGIN
        ERRNUM := 5;
        EXIT(BINDUNIT)
      END;
    IF B = PFUNIT THEN
      BEGIN
        ERRNUM := 16;
        EXIT(BINDUNIT)
      END;
    IF A > DRVS[0].EXTRA THEN
      BEGIN
        ERRNUM := 1;
        EXIT(BINDUNIT)
      END;
    IF CLEARASN THEN
      BEGIN
        V := VOLIDX(B);
        VOLS[V].DV := DRIVE;
        VOLS[V].BUSY := TRUE;
        VOLS[V].WPVOL := DRVS[A].ATTRIB.WPROT;
        VOLS[V].B := DRVS[A].MAPLO;
        VOLS[V].A := DRVS[A].EXTRA;
        RDUNIT(B, T);
        WRUNIT(B, DRVADR);
        DRVS[A].AUX := T;
        DRVS[A].ATTRIB.UNITNO := B;
        FIXASN;
        UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
        IF IOFAIL THEN
          EXIT(BINDUNIT);
        UNITSTATUS(PFUNIT, VOLS, 16386);
        GOTO 99
      END;
    IF IOFAIL THEN
      EXIT(BINDUNIT);
  99:
  END;
  (*$I+*)

  FUNCTION ADJAREA(N: INTEGER): BOOLEAN;
  VAR
    NEWEND, OLDEND: INTEGER;
    ENT: FILEENT;
    BLK: DIRBLOCK;
    SLOT, BLKNO, OFF: INTEGER;
    KEY: INTEGER;
    PROCEDURE FAILADJ;
    BEGIN
      IF (IORESULT <> 0) OR (ERRNUM <> 0) THEN
        EXIT(ADJAREA)
    END;
  BEGIN
    ADJAREA := FALSE;
    KEY := FINDPAS(BLK, SLOT, BLKNO, OFF);
    FAILADJ;
    MOVELEFT(BLK[OFF], ENT, 40);
    NEWEND := ENT.USEDHI * 256 + ENT.USEDLO;
    NEWEND := NEWEND + N;
    ENT.USEDHI := SAFEDIV(NEWEND, 256);
    ENT.USEDLO := SAFEMOD(NEWEND, 256);
    OLDEND := KEY - N;
    ENT.KEYHI := SAFEDIV(OLDEND, 256);
    ENT.KEYLO := SAFEMOD(OLDEND, 256);
    MOVELEFT(ENT, BLK[OFF], 40);
    (*$I-*)
    UNITWRITE(PFUNIT, BLK, 512, BLKNO, DRVCTL(DRIVE));
    (*$I+*)
    FAILADJ;
    ADJAREA := TRUE
  END;

  FUNCTION ALLOCBLK(NEED: INTEGER; VAR GOT: INTEGER): BOOLEAN;
    PROCEDURE FAILALC;
    BEGIN
      IF IOFAIL THEN
        EXIT(ALLOCBLK)
    END;
  BEGIN
    CLRERR;
    ALLOCBLK := FALSE;
    IF ALLBITS(PAKEY - 1, NEED, GOT, TRUE) THEN
      BEGIN
        SETBITS(PAKEY - 1, PAKEY - NEED, FALSE);
        FAILALC;
        IF ADJAREA(NEED) THEN
          ALLOCBLK := TRUE
      END;
    FAILALC
  END;

  FUNCTION FREEBLK(N: INTEGER): INTEGER;
    PROCEDURE FAILFRE;
    BEGIN
      IF IOFAIL THEN
        EXIT(FREEBLK)
    END;
  BEGIN
    CLRERR;
    FREEBLK := 0;
    IF ADJAREA(-N) THEN
      SETBITS(PAKEY + N - 1, PAKEY, TRUE);
    FAILFRE;
    FREEBLK := N
  END;

  PROCEDURE KRUNCHPA;
    { Last jump-table word is IC-JTAB; 1.1's self-rel is
      off by data_size 24. Instruction stream matches. }
    PROCEDURE SLIDE;
    VAR
      ENDPT: INTEGER;
      BASE: INTEGER;
      LEN: INTEGER;
      MAP: INTEGER;
      GAP: INTEGER;
      I: INTEGER;
      U: INTEGER;
      BYTES: INTEGER;
      HEAP: ^INTEGER;
      NBLK: INTEGER;
      MARKP: ^INTEGER;
      LIM: INTEGER;
      PROCEDURE FILLHEAP;
      VAR P: ^HEAPBLK;
      BEGIN
        NEW(HEAP);
        NBLK := 0;
        WHILE (MEMAVAIL > 556) AND (NBLK < 64) DO
          BEGIN
            NEW(P);
            NBLK := NBLK + 1
          END
      END;
    BEGIN
      MARK(MARKP);
      FILLHEAP;
      IF NBLK > 0 THEN
        BEGIN
          BASE := PAKEY + DRVS[0].MAPLO;
          I := DRVS[0].EXTRA;
          LIM := 1;
          WHILE I >= LIM DO
            BEGIN
              MAP := DRVS[I].MAPLO;
              LEN := DRVS[I].EXTRA;
              ENDPT := MAP + LEN;
              GAP := BASE - ENDPT;
              IF GAP > 0 THEN
                BEGIN
                  BYTES := NBLK * 512;
                  ENDPT := ENDPT - NBLK;
                  IF ENDPT < MAP THEN
                    BEGIN
                      BYTES := BYTES - 512 * (MAP - ENDPT);
                      ENDPT := MAP
                    END;
                  (*$I-*)
                  REPEAT
                    UNITREAD(PFUNIT, HEAP^, BYTES, ENDPT, DRVCTL(DRIVE));
                    IF IOFAIL THEN
                      EXIT(KRUNCHPA);
                    UNITWRITE(PFUNIT, HEAP^, BYTES, ENDPT + GAP, DRVCTL(DRIVE));
                    IF IOFAIL THEN
                      EXIT(KRUNCHPA)
                  UNTIL ENDPT <= MAP;
                  BASE := MAP + GAP;
                  U := KEYVOL(MAP);
                  IF U <> 0 THEN
                    BEGIN
                      VOLS[VOLIDX(U)].B := BASE;
                      UNITSTATUS(PFUNIT, VOLS, 16386)
                    END;
                  DRVS[I].MAPLO := BASE;
                  UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
                  IF IOFAIL THEN
                    EXIT(KRUNCHPA)
                END
              ELSE
                BASE := MAP;
              I := I - 1
            END
        END;
      RELEASE(MARKP)
    END;
    PROCEDURE SHRINK;
    VAR
      FRE: INTEGER;
      HI: INTEGER;
      LO: INTEGER;
    BEGIN
      WITH DRVS[0] DO
        IF EXTRA > 0 THEN
          LO := DRVS[1].MAPLO - 2
        ELSE
          LO := PAKEY + MAPLO - 2;
      HI := LO - PAKEY;
      FRE := FREEBLK(HI);
      IF IOFAIL THEN
        EXIT(KRUNCHPA);
      WITH DRVS[0] DO
        MAPLO := MAPLO - FRE;
      PAKEY := LO;
      (*$I-*)
      UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
      (*$I+*)
      IF IOFAIL THEN
        EXIT(KRUNCHPA)
    END;
  BEGIN
    CLRERR;
    SLIDE;
    SHRINK
  END;

  (*$I-*)
  PROCEDURE REMVOL;
  VAR LAST, K, I: INTEGER;
  BEGIN
    CLRERR;
    IF N > DRVS[0].EXTRA THEN
      BEGIN
        ERRNUM := 1;
        EXIT(REMVOL)
      END;
    IF DRVS[N].ATTRIB.WPROT THEN
      BEGIN
        ERRNUM := 10;
        EXIT(REMVOL)
      END;
    UNITREAD(PFUNIT, PADIR, 25, DRVS[N].MAPLO + 2, DRVCTL(DRIVE));
    IF IOFAIL THEN
      EXIT(REMVOL);
    IF PADIR[0].NFILES <> 0 THEN
      BEGIN
        ERRNUM := 15;
        EXIT(REMVOL)
      END;
    K := KEYVOL(DRVS[N].MAPLO);
    IF K <> 0 THEN
      BEGIN
        FREEUNIT(K);
        IF IOFAIL THEN
          EXIT(REMVOL)
      END;
    I := N;
    LAST := DRVS[0].EXTRA - 1;
    WHILE I <= LAST DO
      BEGIN
        DRVS[I] := DRVS[I + 1];
        VNAMES[I] := VNAMES[I + 1];
        DESCS[I] := DESCS[I + 1];
        I := I + 1
      END;
    DRVS[0].EXTRA := DRVS[0].EXTRA - 1;
    UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
    IF IOFAIL THEN
      EXIT(REMVOL);
    IF FLAG THEN
      KRUNCHPA
  END;
  (*$I+*)

  FUNCTION ADDVOL;
  VAR
    SD: VOLDESC;
    SN: VOLNAME;
    GOT: INTEGER;
    LIM: INTEGER;
    PROCEDURE FINDGAP(NEED: INTEGER; VAR SLOT: INTEGER; VAR BEST: INTEGER);
    VAR ENDPT, BESTG, GAP: INTEGER;
    BEGIN
      SLOT := 0;
      GAP := 0;
      BESTG := 0;
      WHILE (SLOT <= DRVS[0].EXTRA) AND (GAP < NEED) DO
        BEGIN
          SLOT := SLOT + 1;
          IF SLOT > DRVS[0].EXTRA THEN
            BEGIN
              IF DRVS[0].EXTRA > 0 THEN
                BEGIN
                  ENDPT := DRVS[DRVS[0].EXTRA].MAPLO +
                           DRVS[DRVS[0].EXTRA].EXTRA;
                  GAP := (PAKEY + DRVS[0].MAPLO - 1) - ENDPT + 1
                END
              ELSE
                BEGIN
                  ENDPT := PAKEY + 2;
                  GAP := (PAKEY + DRVS[0].MAPLO - 1) - ENDPT + 1
                END
            END
          ELSE
            IF SLOT = 1 THEN
              BEGIN
                ENDPT := PAKEY + 2;
                GAP := DRVS[1].MAPLO - 1 - ENDPT + 1
              END
            ELSE
              BEGIN
                ENDPT := DRVS[SLOT - 1].MAPLO + DRVS[SLOT - 1].EXTRA;
                GAP := DRVS[SLOT].MAPLO - 1 - ENDPT + 1
              END;
          IF GAP > BESTG THEN
            BESTG := GAP
        END;
      IF GAP < NEED THEN
        BEGIN
          BEST := BESTG;
          SLOT := 0
        END
      ELSE
        BEST := ENDPT
    END;
    PROCEDURE NEWVDIR;
    BEGIN
      FILLCHAR(PADIR, 2028, CHR(0));
      PADIR[0].FIRSTB := 0;
      PADIR[0].LASTB := 6;
      MOVELEFT(SN, PADIR[0].NM, 8);
      PADIR[0].NBLKS := N;
      PADIR[0].NFILES := 0;
      PADIR[0].DAT.MONTH := THEDATE.MONTH;
      PADIR[0].DAT.DAY := THEDATE.DAY;
      PADIR[0].DAT.YEAR := THEDATE.YEAR;
      (*$I-*)
      UNITWRITE(PFUNIT, PADIR, 26, DRVS[NEWSLOT].MAPLO + 2, DRVCTL(DRIVE));
      (*$I+*)
      IF IOFAIL THEN
        EXIT(ADDVOL)
    END;
  BEGIN
    SN := NM;
    SD := DS;
    CLRERR;
    ADDVOL := 0;
    NEWSLOT := 0;
    IF N < 6 THEN
      BEGIN
        ERRNUM := 13;
        EXIT(ADDVOL)
      END;
    IF DRVS[0].EXTRA >= 31 THEN
      BEGIN
        ERRNUM := 3;
        EXIT(ADDVOL)
      END;
    IF NOT PVVALID(SN) THEN
      BEGIN
        ERRNUM := 9;
        EXIT(ADDVOL)
      END;
    NEWSLOT := FINDBYNM(SN, SD);
    IF ERRNUM = 1 THEN
      ERRNUM := 0;
    IF NEWSLOT <> 0 THEN
      IF DESCS[NEWSLOT] = SD THEN
        BEGIN
          ERRNUM := 4;
          NEWSLOT := 0;
          EXIT(ADDVOL)
        END;
    FINDGAP(N, NEWSLOT, GAPEND);
    IF NEWSLOT = 0 THEN
      IF ALLOCBLK(N, GOT) THEN
        BEGIN
          DRVS[0].MAPLO := DRVS[0].MAPLO + N;
          PAKEY := PAKEY - N;
          (*$I-*)
          UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
          (*$I+*)
          IF IOFAIL THEN
            EXIT(ADDVOL);
          FINDGAP(N, NEWSLOT, GAPEND)
        END;
    IF NEWSLOT = 0 THEN
      BEGIN
        ERRNUM := 2;
        EXIT(ADDVOL)
      END;
    SELIX := DRVS[0].EXTRA;
    LIM := NEWSLOT;
    WHILE SELIX >= LIM DO
      BEGIN
        DRVS[SELIX + 1] := DRVS[SELIX];
        VNAMES[SELIX + 1] := VNAMES[SELIX];
        DESCS[SELIX + 1] := DESCS[SELIX];
        SELIX := SELIX - 1
      END;
    DRVS[NEWSLOT].MAPLO := GAPEND;
    DRVS[NEWSLOT].EXTRA := N;
    DRVS[NEWSLOT].ATTRIB.UNITNO := 0;
    DRVS[NEWSLOT].ATTRIB.FILL7 := 0;
    DRVS[NEWSLOT].ATTRIB.WPROT := FALSE;
    VNAMES[NEWSLOT] := SN;
    DESCS[NEWSLOT] := SD;
    DRVS[0].EXTRA := DRVS[0].EXTRA + 1;
    NEWVDIR;
    (*$I-*)
    UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
    (*$I+*)
    IF IOFAIL THEN
      NEWSLOT := 0;
    ADDVOL := NEWSLOT
  END;

  (*$I-*)
  PROCEDURE SETWP;
  VAR U: INTEGER;
  BEGIN
    CLRERR;
    IF N > DRVS[0].EXTRA THEN
      BEGIN
        ERRNUM := 1;
        EXIT(SETWP)
      END;
    DRVS[N].ATTRIB.WPROT := FLAG;
    UNITWRITE(PFUNIT, DRVS, TBLLEN, PAKEY, DRVCTL(DRIVE));
    IF IOFAIL THEN
      EXIT(SETWP);
    U := DRVS[N].ATTRIB.UNITNO;
    IF U <> 0 THEN
      BEGIN
        VOLS[VOLIDX(U)].WPVOL := FLAG;
        UNITSTATUS(PFUNIT, VOLS, 16386)
      END
  END;
