{ UTILITY -- Pascal Backup, segment 7 of BACKUP.CODE.

  A plain unit. Screen helpers copy CRTCTRL out of SYSCOM
  (LOD 2,1) into interface arrays, then UNITWRITE unit 2.
  Compiled (*$U-*) with SYSCOM at level 0, same wrapper as
  VMGR. (*$NS 7*) keeps the unit on segment 7.

  Interface variables occupy the whole host global frame
  (host words 3..5766). BACKUP itself declares none.

  CRTPRE is the unit's copy of CRTCTRL.PREFIXED; the name
  avoids a clash with SYSCOM^.CRTCTRL.PREFIXED inside WITH.

  U43..U48 are EXTERNAL 6502 in ppm/src/UTILNAT.TEXT. }
(*$U-*)
(*$NS 7*)
(*$R+*)
(*$I+*)

  END;

  FUNCTION U15;
  VAR
    PREF: BOOLEAN;
    CODE: INTEGER;
  BEGIN
    IF CH = ESCAPE THEN
      BEGIN
        UNITREAD(2, CH, 1, 0, 12);
        PREF := TRUE
      END
    ELSE
      PREF := FALSE;
    CODE := KEYMAP[ORD(CH)];
    IF NOT (PREF = INFOPREF[CODE]) THEN
      CODE := 10;
    U15 := CODE
  END;

  PROCEDURE U16;
  VAR
    T: STR80;
  BEGIN
    FILLCHAR(T[1], N, CH);
    T[0] := CHR(N);
    MOVELEFT(S[1], T[(N - LENGTH(S)) + 1], LENGTH(S));
    S := T
  END;

  PROCEDURE U17;
  VAR
    T: STR80;
  BEGIN
    FILLCHAR(T[1], N, CH);
    T[0] := CHR(N);
    MOVELEFT(S[1], T[1], LENGTH(S));
    S := T
  END;

  PROCEDURE U18;
  VAR
    I: INTEGER;
  BEGIN
    I := SCAN(LENGTH(S), <>' ', S[1]);
    S[0] := CHR(LENGTH(S) - I);
    MOVELEFT(S[I + 1], S[1], LENGTH(S))
  END;

  PROCEDURE U19;
  VAR
    I: INTEGER;
  BEGIN
    I := SCAN(-LENGTH(S), <>' ', S[LENGTH(S)]);
    S[0] := CHR(LENGTH(S) + I)
  END;

  PROCEDURE U20;
  VAR
    I: INTEGER;
    J: INTEGER;
  BEGIN
    I := LENGTH(S);
    J := 1;
    WHILE I >= J DO
      BEGIN
        IF S[I] = ' ' THEN
          DELETE(S, I, 1);
        I := I - 1
      END
  END;

  PROCEDURE U21;
  VAR
    POS: INTEGER;
    NEG: BOOLEAN;
  BEGIN
    IF N = 0 THEN
      S := '0'
    ELSE
      BEGIN
        NEG := N < 0;
        IF NEG THEN
          N := -N;
        POS := 7;
        WHILE N > 0 DO
          BEGIN
            POS := POS - 1;
            S[POS] := CHR((N MOD 10) + 48);
            N := N DIV 10
          END;
        IF NEG THEN
          BEGIN
            POS := POS - 1;
            S[POS] := '-'
          END;
        IF POS > 1 THEN
          MOVELEFT(S[POS], S[1], 7 - POS);
        S[0] := CHR(7 - POS)
      END
  END;

  FUNCTION U22;
  VAR
    NEG: BOOLEAN;
    DIG: INTEGER;
  BEGIN
    NEG := FALSE;
    U22 := FALSE;
    N := 0;
    IF S[1] = '-' THEN
      BEGIN
        NEG := TRUE;
        DELETE(S, 1, 1)
      END;
    WHILE (N < 1000) AND (S <> '') DO
      BEGIN
        DIG := ORD(S[1]) - 48;
        N := N * 10 + DIG;
        DELETE(S, 1, 1)
      END;
    IF S <> '' THEN
      BEGIN
        IF N > 3276 THEN
          EXIT(U22);
        IF (N = 3276) AND (S[1] > '7') THEN
          EXIT(U22);
        DIG := ORD(S[1]) - 48;
        N := N * 10 + DIG
      END;
    IF NEG THEN
      N := -N;
    U22 := TRUE
  END;

  FUNCTION U25;

    CH: PACKED ARRAY [0..0] OF CHAR;

  FUNCTION U26;
  VAR
    CH: CHAR;
  BEGIN
    CH := U25;
    WHILE NOT (CH IN VALID) DO
      BEGIN
        U42;
        CH := U25
      END;
    IF ECHO THEN
      IF CH IN SET125 THEN
        UNITWRITE(1, CH, 1, 0, 12);
    U26 := CH
  END;

  FUNCTION U27;
  VAR
    CODE: INTEGER;
    OK: BOOLEAN;
  BEGIN
    REPEAT
      CH := U25;
      CODE := U15(CH);
      IF CH IN VALID THEN
        OK := TRUE
      ELSE
        IF U43(CH) IN VALID THEN
          BEGIN
            CH := U43(CH);
            OK := TRUE
          END
        ELSE
          IF CODE IN KEYS THEN
            OK := TRUE
          ELSE
            BEGIN
              OK := FALSE;
              U42
            END
    UNTIL OK;
    IF CODE = 10 THEN
      IF (CH IN SET125) AND ECHO THEN
        UNITWRITE(1, CH, 1, 0, 12);
    W106 := CODE;
    U27 := CODE
  END;

      U24(X, Y);

  PROCEDURE U34;
  VAR
    PAD: INTEGER;
  BEGIN
    PAD := 0;
    U02(0, Y);
    IF LENGTH(S) > (W108 + 1) THEN
      S := COPY(S, 1, W108 + 1);
    IF LENGTH(S) < (W108 + 1) THEN
      PAD := ((W108 + 1) - LENGTH(S)) DIV 2;
    IF PAD > 0 THEN
      PAD := PAD - 1;
    U33(PAD, Y, S)
  END;

  PROCEDURE U36;
  VAR
    CH: CHAR;
    UNUSED: STR80;
    T: STR80;
    I: INTEGER;
    VALID: CHARSET;
    CODE: INTEGER;
  BEGIN
    IF X < 0 THEN
      U24(X, Y);
    T := '';
    I := 0;
    U35(X, Y, ' ', N);
    U23(X, Y);
    REPEAT
      IF I >= N THEN
        VALID := []
      ELSE
        VALID := SET125;
      VALID := VALID + [CHR(13)];
      CODE := U27(VALID, W105, CH, TRUE);
      CASE CODE OF
        0, 5, 8:
          IF CH <> CHR(13) THEN
            BEGIN
              I := I + 1;
              T[I] := CH
            END;
        4:
          BEGIN
            T := S;
            I := LENGTH(S)
          END;
        10:
          IF I > 0 THEN
            BEGIN
              I := I - 1;
              WRITE(CHR(8), ' ', CHR(8))
            END
      END
    UNTIL (CODE IN W5766) OR (CH = CHR(13));
    T[0] := CHR(I);
    U35(X, Y, T, N);
    S := T
  END;

  PROCEDURE U37;

  PROCEDURE U38;
  VAR
    CH: CHAR;
    SPARE: INTEGER;
    I: INTEGER;
    UNUSED: STR80;
    T: STR80;
    VALID: CHARSET;
    CODE: INTEGER;
    K: INTEGER;
  BEGIN
    IF X < 0 THEN
      U24(X, Y);
    T := '';
    I := 0;
    U35(X, Y, ' ', WID);
    U23(X, Y);
    REPEAT
      IF I = WID THEN
        VALID := []
      ELSE
        VALID := SETDIG;
      VALID := VALID + [CHR(13)];
      CODE := U27(VALID, W105, CH, TRUE);
      CASE CODE OF
        0, 5, 8:
          IF I > 0 THEN
            BEGIN
              I := I - 1;
              WRITE(CHR(8), ' ', CHR(8))
            END;
        10:
          IF CH <> CHR(13) THEN
            BEGIN
              I := I + 1;
              T[I] := CH
            END
          ELSE
            BEGIN
              NUM := 0;
              W5676 := 1;
              K := I;
              WHILE W5676 <= K DO
                BEGIN
                  NUM := (NUM * 10 + ORD(T[W5676])) - 48;
                  W5676 := W5676 + 1
                END
            END
      END
    UNTIL (CODE IN W5766) OR (CH = CHR(13));
    U37(X, Y, NUM, WID)
  END;

        INFOPREF[8] := PREFIXED[1];

        CRTINFOCH[11] := CHR(128 + 73);
        INFOPREF[11] := FALSE;
        CRTINFOCH[12] := CHR(128 + 68);
        INFOPREF[12] := FALSE;
        CRTINFOCH[10] := CHR(0);
        INFOPREF[10] := FALSE
      END;
    ESCAPE := SYSCOM^.CRTINFO.ALPHA_LOCK;
    SETCR := [CHR(13)] + [' '..CHR(127)];
    FOR I := 0 TO 255 DO
      KEYMAP[I] := 10;
    FOR J := 0 TO 12 DO
      KEYMAP[ORD(CRTINFOCH[J])] := J;
    KEYMAP[128 + 105] := 11;
    KEYMAP[128 + 100] := 12;
    CUR.B := 0;
    CUR.A := 0;

    CUR.WIDTH := 79;

    W5766 := [4];
    W107 := CUR.HEIGHT;
    W108 := CUR.WIDTH;
    W5756 := FALSE;
    W5668 := FALSE;
