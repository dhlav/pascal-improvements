{ UTILITY -- Pascal Backup, segment 7 of BACKUP.CODE.

  A plain unit. Screen helpers copy CRTCTRL out of SYSCOM
  (LOD 2,1) into interface arrays, then UNITWRITE unit 2.
  Compiled (*$U-*) with SYSCOM at level 0, same wrapper as
  VMGR. (*$NS 7*) keeps the unit on segment 7.

  Interface variables occupy the whole host global frame
  (host words 3..5766). BACKUP itself declares none.

  CRTPRE is the unit's copy of CRTCTRL.PREFIXED; the name
  avoids a clash with SYSCOM^.CRTCTRL.PREFIXED inside WITH.

  UPCHAR..RSTSTAT are EXTERNAL 6502 in ppm/src/UTILNAT.TEXT. }
(*$U-*)
(*$NS 7*)
(*$R+*)
(*$I+*)

  PROCEDURE ERASEOS;
  BEGIN
    MOVETO(X, Y);
    IF CRTPRE[2] THEN
      UNITWRITE(2, ESCAPE, 1, 0, 12);
    UNITWRITE(2, CRTCH[2], 1, 0, 12)
  END;

  PROCEDURE BACKSP;
  BEGIN
    IF CRTPRE[7] THEN
      UNITWRITE(2, ESCAPE, 1, 0, 12);
    UNITWRITE(2, CRTCH[7], 1, 0, 12)
  END;

  PROCEDURE GOHOME;
  BEGIN
    IF CRTPRE[8] THEN
      UNITWRITE(2, ESCAPE, 1, 0, 12);
    UNITWRITE(2, CRTCH[8], 1, 0, 12)
  END;

  PROCEDURE SENDESC;
  BEGIN
    IF CRTPRE[5] THEN
      UNITWRITE(2, ESCAPE, 1, 0, 12);
    UNITWRITE(2, CRTCH[5], 1, 0, 12)
  END;

  PROCEDURE LINEFEED;
  BEGIN
    UNITWRITE(2, CRTCH[6], 1, 0, 12)
  END;

  PROCEDURE GETKEY;
  BEGIN
    CH := ' ';
    REPEAT
      CH := INKEY;
      IF CH IN ['a'..'z'] THEN
        CH := CHR(ORD(CH) - 97 + 65);
      IF NOT (CH IN VALID) THEN
        BEGIN
          CH := CHR(7);
          UNITWRITE(2, CH, 1, 0, 12)
        END
    UNTIL CH IN VALID
  END;

  PROCEDURE CLEARLN;
  BEGIN
    IF CRTPRE[4] THEN
      UNITWRITE(2, ESCAPE, 1, 0, 12);
    UNITWRITE(2, CRTCH[4], 1, 0, 12)
  END;

  PROCEDURE CLEARY;
  BEGIN
    MOVETO(0, Y);
    IF CRTPRE[2] THEN
      UNITWRITE(2, ESCAPE, 1, 0, 12);
    UNITWRITE(2, CRTCH[2], 1, 0, 12)
  END;

  PROCEDURE NDFWD;
  BEGIN
    IF CRTPRE[0] THEN
      UNITWRITE(2, ESCAPE, 1, 0, 12);
    UNITWRITE(2, CRTCH[0], 1, 0, 12)
  END;

  PROCEDURE CLREOL;
  BEGIN
    MOVETO(X, Y);
    IF CRTPRE[1] THEN
      UNITWRITE(2, ESCAPE, 1, 0, 12);
    UNITWRITE(2, CRTCH[1], 1, 0, 12)
  END;
  (*$R+*)
  (*$I+*)

  (*$I-*)
  (*$R-*)
  PROCEDURE CLRSCR;
  BEGIN
    IF CRTCH[3] = '0' THEN
      BEGIN
        MOVETO(0, CUR.Y);
        UNITWRITE(2, CRTCH[2], 1, 0, 12)
      END
    ELSE
      BEGIN
        IF CRTPRE[3] THEN
          UNITWRITE(2, ESCAPE, 1, 0, 12);
        UNITWRITE(2, CRTCH[3], 1, 0, 12)
      END
  END;

  FUNCTION HASCRT;
  BEGIN
    HASCRT := CRTCH[N] <> CHR(0)
  END;

  FUNCTION HASINFO;
  BEGIN
    HASINFO := CRTINFOCH[N] <> CHR(0)
  END;

  FUNCTION KEYCODE;
  VAR
    PREF: BOOLEAN;
    CODE: INTEGER;
  BEGIN
    IF CH = ESCAPE THEN
      BEGIN
        UNITREAD(2, CH, 1, 0, 12);
        PREF := TRUE
      END
    ELSE
      PREF := FALSE;
    CODE := KEYMAP[ORD(CH)];
    IF NOT (PREF = INFOPREF[CODE]) THEN
      CODE := 10;
    KEYCODE := CODE
  END;

  PROCEDURE PADLEFT;
  VAR
    T: STR80;
  BEGIN
    FILLCHAR(T[1], N, CH);
    T[0] := CHR(N);
    MOVELEFT(S[1], T[(N - LENGTH(S)) + 1], LENGTH(S));
    S := T
  END;

  PROCEDURE PADRIGHT;
  VAR
    T: STR80;
  BEGIN
    FILLCHAR(T[1], N, CH);
    T[0] := CHR(N);
    MOVELEFT(S[1], T[1], LENGTH(S));
    S := T
  END;

  PROCEDURE TRIMLEAD;
  VAR
    I: INTEGER;
  BEGIN
    I := SCAN(LENGTH(S), <>' ', S[1]);
    S[0] := CHR(LENGTH(S) - I);
    MOVELEFT(S[I + 1], S[1], LENGTH(S))
  END;

  PROCEDURE TRIMTAIL;
  VAR
    I: INTEGER;
  BEGIN
    I := SCAN(-LENGTH(S), <>' ', S[LENGTH(S)]);
    S[0] := CHR(LENGTH(S) + I)
  END;

  PROCEDURE STRIPSP;
  VAR
    I: INTEGER;
    J: INTEGER;
  BEGIN
    I := LENGTH(S);
    J := 1;
    WHILE I >= J DO
      BEGIN
        IF S[I] = ' ' THEN
          DELETE(S, I, 1);
        I := I - 1
      END
  END;

  PROCEDURE NUMSTR;
  VAR
    POS: INTEGER;
    NEG: BOOLEAN;
  BEGIN
    IF N = 0 THEN
      S := '0'
    ELSE
      BEGIN
        NEG := N < 0;
        IF NEG THEN
          N := -N;
        POS := 7;
        WHILE N > 0 DO
          BEGIN
            POS := POS - 1;
            S[POS] := CHR((N MOD 10) + 48);
            N := N DIV 10
          END;
        IF NEG THEN
          BEGIN
            POS := POS - 1;
            S[POS] := '-'
          END;
        IF POS > 1 THEN
          MOVELEFT(S[POS], S[1], 7 - POS);
        S[0] := CHR(7 - POS)
      END
  END;

  FUNCTION STRNUM;
  VAR
    NEG: BOOLEAN;
    DIG: INTEGER;
  BEGIN
    NEG := FALSE;
    STRNUM := FALSE;
    N := 0;
    IF S[1] = '-' THEN
      BEGIN
        NEG := TRUE;
        DELETE(S, 1, 1)
      END;
    WHILE (N < 1000) AND (S <> '') DO
      BEGIN
        DIG := ORD(S[1]) - 48;
        N := N * 10 + DIG;
        DELETE(S, 1, 1)
      END;
    IF S <> '' THEN
      BEGIN
        IF N > 3276 THEN
          EXIT(STRNUM);
        IF (N = 3276) AND (S[1] > '7') THEN
          EXIT(STRNUM);
        DIG := ORD(S[1]) - 48;
        N := N * 10 + DIG
      END;
    IF NEG THEN
      N := -N;
    STRNUM := TRUE
  END;

  PROCEDURE MOVETO;
  BEGIN
    CUR.X := X;
    CUR.Y := Y;
    GOTOXY(X, Y)
  END;

  PROCEDURE GETXY;
  BEGIN
    X := CUR.X;
    Y := CUR.Y
  END;

  (*$R-*)
  FUNCTION INKEY;
  VAR
    CH: PACKED ARRAY [0..0] OF CHAR;
  BEGIN
    INKEY := ' ';
    UNITREAD(2, CH, 1, 0, 12);
    INKEY := CH[0];
    SAWESC := CH[0] = CHR(27)
  END;

  FUNCTION INSET;
  VAR
    CH: CHAR;
  BEGIN
    CH := INKEY;
    WHILE NOT (CH IN VALID) DO
      BEGIN
        BELL;
        CH := INKEY
      END;
    IF ECHO THEN
      IF CH IN ECHOSET THEN
        UNITWRITE(1, CH, 1, 0, 12);
    INSET := CH
  END;

  FUNCTION CMDKEY;
  VAR
    CODE: INTEGER;
    OK: BOOLEAN;
  BEGIN
    REPEAT
      CH := INKEY;
      CODE := KEYCODE(CH);
      IF CH IN VALID THEN
        OK := TRUE
      ELSE
        IF UPCHAR(CH) IN VALID THEN
          BEGIN
            CH := UPCHAR(CH);
            OK := TRUE
          END
        ELSE
          IF CODE IN KEYS THEN
            OK := TRUE
          ELSE
            BEGIN
              OK := FALSE;
              BELL
            END
    UNTIL OK;
    IF CODE = 10 THEN
      IF (CH IN ECHOSET) AND ECHO THEN
        UNITWRITE(1, CH, 1, 0, 12);
    LASTKEY := CODE;
    CMDKEY := CODE
  END;

  PROCEDURE DRAWYN;
  BEGIN
    IF FLAG THEN
      PUTSTR(X, Y, 'Yes')
    ELSE
      PUTSTR(X, Y, 'No ')
  END;

  (*$R-*)
  PROCEDURE ASKYN;
  VAR
    CH: CHAR;
  BEGIN
    DRAWYN(X, Y, FLAG);
    GETXY(X, Y);
    MOVETO(X - 3, Y);
    LASTKEY := CMDKEY([CHR(13), 'F', 'N', 'T', 'Y',
                 'f', 'n', 't', 'y'],
                [3, 4], CH, FALSE);
    CH := UPCHAR(CH);
    ANSOK := TRUE;
    CASE CH OF
      'T', 'Y':
        FLAG := TRUE;
      'F', 'N':
        FLAG := FALSE;
      OTHERWISE
        ANSOK := FALSE
    END;
    DRAWYN(X - 3, Y, FLAG)
  END;

  FUNCTION YESNO;
  BEGIN
    ASKYN(-1, 0, FLAG);
    YESNO := FLAG
  END;

  PROCEDURE PUTSTR;
  BEGIN
    IF X < 0 THEN
      GETXY(X, Y);
    CUR.X := X;
    CUR.Y := Y;
    MOVETO(X, Y);
    UNITWRITE(1, S[1], LENGTH(S), 0, 0);
    CUR.X := CUR.X + LENGTH(S);
    IF CUR.X > SCRWIDE THEN
      CUR.X := SCRWIDE
  END;

  PROCEDURE CENTER;
  VAR
    PAD: INTEGER;
  BEGIN
    PAD := 0;
    ERASEOS(0, Y);
    IF LENGTH(S) > (SCRWIDE + 1) THEN
      S := COPY(S, 1, SCRWIDE + 1);
    IF LENGTH(S) < (SCRWIDE + 1) THEN
      PAD := ((SCRWIDE + 1) - LENGTH(S)) DIV 2;
    IF PAD > 0 THEN
      PAD := PAD - 1;
    PUTSTR(PAD, Y, S)
  END;

  PROCEDURE PUTPAD;
  BEGIN
    PADRIGHT(S, N, ' ');
    PUTSTR(X, Y, S)
  END;

  PROCEDURE EDITSTR;
  VAR
    CH: CHAR;
    UNUSED: STR80;
    T: STR80;
    I: INTEGER;
    VALID: CHARSET;
    CODE: INTEGER;
  BEGIN
    IF X < 0 THEN
      GETXY(X, Y);
    T := '';
    I := 0;
    PUTPAD(X, Y, ' ', N);
    MOVETO(X, Y);
    REPEAT
      IF I >= N THEN
        VALID := []
      ELSE
        VALID := ECHOSET;
      VALID := VALID + [CHR(13)];
      CODE := CMDKEY(VALID, EDITKEYS, CH, TRUE);
      CASE CODE OF
        0, 5, 8:
          IF CH <> CHR(13) THEN
            BEGIN
              I := I + 1;
              T[I] := CH
            END;
        4:
          BEGIN
            T := S;
            I := LENGTH(S)
          END;
        10:
          IF I > 0 THEN
            BEGIN
              I := I - 1;
              WRITE(CHR(8), ' ', CHR(8))
            END
      END
    UNTIL (CODE IN STOPKEY) OR (CH = CHR(13));
    T[0] := CHR(I);
    PUTPAD(X, Y, T, N);
    S := T
  END;

  PROCEDURE PUTNUM;
  VAR
    T: STR80;
  BEGIN
    NUMSTR(VAL, T);
    PADLEFT(T, WID, ' ');
    PUTSTR(X, Y, T)
  END;

  PROCEDURE EDITNUM;
  VAR
    CH: CHAR;
    SPARE: INTEGER;
    I: INTEGER;
    UNUSED: STR80;
    T: STR80;
    VALID: CHARSET;
    CODE: INTEGER;
    K: INTEGER;
  BEGIN
    IF X < 0 THEN
      GETXY(X, Y);
    T := '';
    I := 0;
    PUTPAD(X, Y, ' ', WID);
    MOVETO(X, Y);
    REPEAT
      IF I = WID THEN
        VALID := []
      ELSE
        VALID := SETDIG;
      VALID := VALID + [CHR(13)];
      CODE := CMDKEY(VALID, EDITKEYS, CH, TRUE);
      CASE CODE OF
        0, 5, 8:
          IF I > 0 THEN
            BEGIN
              I := I - 1;
              WRITE(CHR(8), ' ', CHR(8))
            END;
        10:
          IF CH <> CHR(13) THEN
            BEGIN
              I := I + 1;
              T[I] := CH
            END
          ELSE
            BEGIN
              NUM := 0;
              DIGIX := 1;
              K := I;
              WHILE DIGIX <= K DO
                BEGIN
                  NUM := (NUM * 10 + ORD(T[DIGIX])) - 48;
                  DIGIX := DIGIX + 1
                END
            END
      END
    UNTIL (CODE IN STOPKEY) OR (CH = CHR(13));
    PUTNUM(X, Y, NUM, WID)
  END;

  FUNCTION WAITCODE;
  VAR
    CH: CHAR;
  BEGIN
    CENTER(SCRHIGH - 1, S);
    CENTER(SCRHIGH, 'Press SPACE to continue');
    WAITCODE := CMDKEY([CHR(13), ' '], [3, 4], CH, FALSE);

      WAITCODE := 3
  END;

  PROCEDURE HITSPACE;
  VAR
    CODE: INTEGER;
    CH: CHAR;
  BEGIN
    BELL;
    PUTSTR(0, SCRHIGH, 'Press SPACEBAR to continue');
    CODE := CMDKEY([CHR(13), ' '], [4], CH, FALSE)
  END;

  VAR
    CH: CHAR;
  BEGIN
    CH := CHR(7);
    UNITWRITE(2, CH, 1, 0, 12)

  PROCEDURE READDIRS;
  BEGIN
    PROGUNIT := SYSCOM^.SEGTABLE[1].CODEUNIT;
    BOOTUNIT := SYSCOM^.SEGTABLE[0].CODEUNIT;
    UNITREAD(PROGUNIT, PROGDIR, 26, 2, 0);
    DIRIO := IORESULT;
    IF DIRIO <> 0 THEN
      BEGIN
        WRITE('Unable to read directory block of program disk ');
        WRITELN
      END;
    IF BOOTUNIT <> PROGUNIT THEN
      BEGIN
        UNITREAD(BOOTUNIT, BOOTDIR, 26, 2, 0);
        DIRIO := IORESULT;
        IF DIRIO <> 0 THEN
          BEGIN
            WRITE('Unable to read directory block of boot disk');
            WRITELN
          END
      END
  END;

  PROCEDURE NEEDDISK;
  VAR
    D1: INTEGER;
    D2: INTEGER;
    D3: INTEGER;
    VID: STRING[7];
    REST: ARRAY [1..6] OF INTEGER;

    (*$I-*)
    PROCEDURE WAITVOL(VAR U: INTEGER; VAR BLK: DIR26);
    BEGIN
      REPEAT
        FILLCHAR(D1, 26, CHR(0));
        UNITREAD(U, D1, 26, 2, 0);
        DIRIO := IORESULT;
        IF (DIRIO <> 0) OR (VID <> BLK.VID) THEN
          BEGIN
            DIRIO := 1;
            CLEARY(22);
            WRITE('Insert ', BLK.VID, ' in unit #', U);
            HITSPACE;
            CLREOL(0, 22)
          END
      UNTIL DIRIO = 0
    END;
    (*$I+*)

  BEGIN
    WAITVOL(PROGUNIT, PROGDIR);
    IF PROGUNIT <> BOOTUNIT THEN
      WAITVOL(BOOTUNIT, BOOTDIR)
  END;
